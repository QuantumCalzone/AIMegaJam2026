; WBP_Earpiece.SetSpot(Spot) (T61): "end" sits beside the end screen's Play Again and Quit row (EndX, default -470 left of
; center, one Edge up), clear of the cards, the buttons and T60's ReasonText just above them; WBP_Stage is hidden there.
; Every other spot asks WBP_Stage.EarSpot, which knows the in-show screen's layout (Scene Prep or the scene's phases):
; an anchor on the screen's bottom edge (0 left, 0.5 center, 1 right), the offset from it, the height and the widest.
(fn SetSpot (Spot)
  (bind _e (Class|UIStyle|UISize :Role "Edge"))
  (set SpotA 0.5)
  (set SpotX (V EndX))
  (set SpotY _e)
  (set SpotW 720.0)
  (if (not (streq Spot "end"))
    (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
    (bind (_a _x _y _w) (Class|WBPStage|EarSpot :self (Class|BPShow|GetStage :self _show)))
    (set SpotA _a)
    (set SpotX _x)
    (set SpotY _y)
    (set SpotW _w))
  (Layout|CanvasSlot|SetAnchors :self (Slot|SlotasCanvasSlot (Variables|WBP_Earpiece|GetEarBox)) :InAnchors (Utilities|Struct|MakeAnchors :Minimum (Math|Vector2D|MakeVector2D (V SpotA) 1.0) :Maximum (Math|Vector2D|MakeVector2D (V SpotA) 1.0)))
  (Layout|CanvasSlot|SetAlignment :self (Slot|SlotasCanvasSlot (Variables|WBP_Earpiece|GetEarBox)) :InAlignment (Math|Vector2D|MakeVector2D (V SpotA) 1.0))
  (Layout|CanvasSlot|SetPosition :self (Slot|SlotasCanvasSlot (Variables|WBP_Earpiece|GetEarBox)) :InPosition (Math|Vector2D|MakeVector2D (V SpotX) (- 0.0 (V SpotY))))
  (Layout|SizeBox|SetMaxDesiredWidth :self (Variables|WBP_Earpiece|GetEarBox) :InMaxDesiredWidth (V SpotW)))
