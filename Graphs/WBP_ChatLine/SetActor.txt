; WBP_ChatLine.SetActor(Name, Color, Action, Speech, IsMe). T32: every value comes from UIStyle.
; Name in bold in its seat color (UIStyle.SeatColor by provider name; Color is the fallback for an unknown name),
; the action in quiet italic, the speech in white. My lines: a teal YOU tag and a faint teal tint behind the line.
(fn SetActor (Name Color Action Speech IsMe)
  (bind seat (Class|UIStyle|SeatColor :Name Name :Fallback Color))
  (bind fName (Class|UIStyle|UIFont :Role "Name"))
  (bind fAct (Class|UIStyle|UIFont :Role "Action"))
  (bind fSay (Class|UIStyle|UIFont :Role "Speech"))
  (bind quiet (Class|UIStyle|UIColor :Role "Quiet"))
  (bind white (Class|UIStyle|UIColor :Role "Text"))
  (bind ink (Class|UIStyle|UIColor :Role "InkSolid"))
  (bind s2 (Class|UIStyle|UISize :Role "S2"))
  (bind s3 (Class|UIStyle|UISize :Role "S3"))
  (bind bar (Class|UIStyle|UISize :Role "SeatBar"))
  (bind tint (Class|UIStyle|UISize :Role "YouTint"))
  (bind tagR (Class|UIStyle|UISize :Role "TagRadius"))
  (Class|Border|SetPadding :self (Variables|WBP_ChatLine|GetBg) :Padding (Utilities|Struct|MakeMargin s3 s2 s3 s2))
  (Class|Spacer|SetSize :self (Variables|WBP_ChatLine|GetAccentSize) :Size (Math|Vector2D|MakeVector2D bar bar))
  (Class|Border|SetBrushColor :self (Variables|WBP_ChatLine|GetAccent) :BrushColor seat)
  (Class|Text|SetFont :self (Variables|WBP_ChatLine|GetNameText) :Font fName)
  (if IsMe
    (Class|Text|SetText :self (Variables|WBP_ChatLine|GetNameText) :Text (Utilities|Text|ToText(String) "YOU"))
    (Class|Text|SetColorandOpacity :self (Variables|WBP_ChatLine|GetNameText) :ColorAndOpacity (Utilities|Struct|MakeSlateColor :SpecifiedColor ink))
    (bind tag (Class|UIStyle|Brush :Fill seat :Outline seat :Width 0.0 :Radius tagR))
    (Appearance|SetBrush :self (Variables|WBP_ChatLine|GetNameTag) :InBrush tag)
    (Class|Border|SetBrushColor :self (Variables|WBP_ChatLine|GetNameTag) :BrushColor (Utilities|Struct|MakeLinearColor 1.0 1.0 1.0 1.0))
    (Class|Border|SetPadding :self (Variables|WBP_ChatLine|GetNameTag) :Padding (Utilities|Struct|MakeMargin s2 0.0 s2 0.0))
    (Class|Border|SetBrushColor :self (Variables|WBP_ChatLine|GetBg) :BrushColor (Math|Color|NewOpacity(LinearColor) seat tint))
    (else
      (Class|Text|SetText :self (Variables|WBP_ChatLine|GetNameText) :Text (Utilities|Text|ToText(String) Name))
      (Class|Text|SetColorandOpacity :self (Variables|WBP_ChatLine|GetNameText) :ColorAndOpacity (Utilities|Struct|MakeSlateColor :SpecifiedColor seat))
      (Class|Border|SetBrushColor :self (Variables|WBP_ChatLine|GetNameTag) :BrushColor (Utilities|Struct|MakeLinearColor 0.0 0.0 0.0 0.0))
      (Class|Border|SetPadding :self (Variables|WBP_ChatLine|GetNameTag) :Padding (Utilities|Struct|MakeMargin 0.0 0.0 0.0 0.0))
      (Class|Border|SetBrushColor :self (Variables|WBP_ChatLine|GetBg) :BrushColor (Utilities|Struct|MakeLinearColor 0.0 0.0 0.0 0.0))))
  (Class|Text|SetFont :self (Variables|WBP_ChatLine|GetActionText) :Font fAct)
  (Class|Text|SetColorandOpacity :self (Variables|WBP_ChatLine|GetActionText) :ColorAndOpacity (Utilities|Struct|MakeSlateColor :SpecifiedColor quiet))
  (Class|Text|SetText :self (Variables|WBP_ChatLine|GetActionText) :Text (Utilities|Text|ToText(String) Action))
  (if (Utilities|String|IsEmpty Action)
    (Widget|SetVisibility :self (Variables|WBP_ChatLine|GetActionText) :InVisibility "Collapsed"))
  (Class|Text|SetFont :self (Variables|WBP_ChatLine|GetSpeechText) :Font fSay)
  (Class|Text|SetColorandOpacity :self (Variables|WBP_ChatLine|GetSpeechText) :ColorAndOpacity (Utilities|Struct|MakeSlateColor :SpecifiedColor white))
  (Class|Text|SetText :self (Variables|WBP_ChatLine|GetSpeechText) :Text (Utilities|Text|ToText(String) Speech))
  (if (Utilities|String|IsEmpty Speech)
    (Widget|SetVisibility :self (Variables|WBP_ChatLine|GetSpeechText) :InVisibility "Collapsed")))
