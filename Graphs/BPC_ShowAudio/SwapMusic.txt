(fn SwapMusic (Key)
  (CallFunction|StopMusic)
  (set MusicKey Key)
  (print (cat "NHOS: SOUND music " Key))
  (bind _snd (CallFunction|TrackFor :Key Key))
  (bind _m (Audio|SpawnSound2D :Sound _snd))
  (set Music _m))
