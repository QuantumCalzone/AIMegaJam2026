(fn SwapMusic (Key)
  (if (not (Utilities|String|IsEmpty (V MusicKey)))
    (Audio|Components|Audio|StopDelayed :self (V Music) :DelayTime 0.0))
  (set MusicKey Key)
  (print (cat "NHOS: SOUND music " Key))
  (if (streq Key "title")
    (bind _t (Audio|SpawnSound2D :Sound "/Game/NHOS/Audio/Music/S_MusicTitle.S_MusicTitle"))
    (set Music _t)
    (return))
  (bind _s (Audio|SpawnSound2D :Sound "/Game/NHOS/Audio/Music/S_MusicShow.S_MusicShow"))
  (set Music _s))
