; T20: plays one effect by key. Volumes live on the sound waves and their sound classes.
; T102: the "blip" key is gone with S_TalkBlip: nothing played it since T73.
(fn Sfx (Key Pitch)
  (print (cat "NHOS: SOUND " Key))
  (if (streq Key "sting_human")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Show/S_StingHuman.S_StingHuman" :PitchMultiplier Pitch)
    (return))
  (if (streq Key "sting_ai")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Show/S_StingAI.S_StingAI" :PitchMultiplier Pitch)
    (return))
  (if (streq Key "applause")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Show/S_Applause.S_Applause" :PitchMultiplier Pitch)
    (return))
  (if (streq Key "applause_win")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Show/S_ApplauseWin.S_ApplauseWin" :PitchMultiplier Pitch)
    (return))
  (if (streq Key "laughs")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Show/S_Laughs.S_Laughs" :PitchMultiplier Pitch)
    (return))
  (print (cat "NHOS: SOUND unknown " Key)))
