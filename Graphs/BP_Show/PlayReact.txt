; One reaction after the Cut goes into the shared log and the line queue, then the next reactor.
; Written from the editor's read-back in T62, which adds CheckDoes with Max 0: a reaction has no actions,
; so any it sends become "stands still" (NHOS: EXTRA <id> <n> dropped).
(fn PlayReact (Seat Ok Speech Pass Verbs Targets Items)
  (bind _nm (get (V SeatName) Seat))
  (bind _out (CallFunction|CapWords :Text Speech))
  (set ReactPassed (or (not Ok) (or Pass (Utilities|String|IsEmpty _out))))
  (set ReactSays (select (V ReactPassed) "" _out))
  (bind _bd (CallFunction|BuildDoes :Verbs Verbs :Targets Targets :Items Items))
  (set DoesStr _bd)
  (if (== Seat (V HumanSeat))
    (bind _sc (CallFunction|ScrubLabels :Text (V DoesStr)))
    (set DoesStr _sc))
  (bind _ck (CallFunction|CheckDoes :Seat Seat :Does (V DoesStr) :Max 0))
  (set DoesStr _ck)
  (set ReactDoes (select (V ReactPassed) (V DoesPass) (V DoesStr)))
  (set ReactWhen (cat "scene " (str (V SceneNum)) ", reactions"))
  (bind _entry (CallFunction|MakeEntry :When (V ReactWhen) :Actor _nm :Does (V ReactDoes) :Says (V ReactSays)))
  (CallFunction|ActorLine :Seat Seat :Entry _entry)
  (CallFunction|ShowLine :Seat Seat :Does (V ReactDoes) :Says (V ReactSays))
  (print (cat "NHOS: LOG " (V ReactWhen) " | " _nm " | " (V ReactDoes) " | " (V ReactSays)))
  (set ReactCount (+ (V ReactCount) 1))
  (set ReactPos (+ (V ReactPos) 1))
  (CallFunction|NextReactor))
