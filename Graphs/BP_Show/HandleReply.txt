; T07: every reply (an AI's, or my line under its input-<n> tag) finds its pending ask and goes to its phase.
; T63: the input LLM's spelling reply (SpellTag) goes to SpellIn first; it was never a pending ask. A late one
; no longer matches SpellTag and is dropped as stale. Written from the editor's read-back in T63.
(fn HandleReply (Tag Ok Speech Pass Verbs Targets Items)
  (if (and (not (Utilities|String|IsEmpty (V SpellTag))) (streq Tag (V SpellTag)))
    (CallFunction|SpellIn :Ok Ok :Speech Speech :Pass Pass)
    (return))
  (set RIdx (Utilities|Array|FindItem (V PendingTags) Tag))
  (if (< (V RIdx) 0)
    (print (cat "NHOS: STALE reply " Tag))
    (return))
  (set RSeat (get (V PendingSeats) (V RIdx)))
  (Utilities|Array|RemoveIndex (V PendingTags) (V RIdx))
  (Utilities|Array|RemoveIndex (V PendingSeats) (V RIdx))
  (set RVote (Utilities|String|Replace (V RVote) "@" ""))
  (set RSpeech Speech)
  (if (== (V RSeat) (V HumanSeat))
    (bind _sc (CallFunction|ScrubLabels :Text Speech))
    (set RSpeech _sc))
  (if (streq (V Phase) "free talk")
    (CallFunction|PlayTalk :Seat (V RSeat) :Ok Ok :Speech (V RSpeech) :Pass Pass)
    (return))
  (if (streq (V Phase) "the cut")
    (CallFunction|StoreVote :Seat (V RSeat) :Ok Ok :Vote (V RVote) :Reason (V RReason) :Speech (V RSpeech))
    (return))
  (if (streq (V Phase) "reactions")
    (CallFunction|PlayReact :Seat (V RSeat) :Ok Ok :Speech (V RSpeech) :Pass Pass :Verbs Verbs :Targets Targets :Items Items)
    (return))
  (if (streq (V Phase) "closing")
    (CallFunction|StoreClosing :Seat (V RSeat) :Ok Ok :Vote (V RVote) :Reason (V RReason))
    (return))
  (CallFunction|StoreTurn :Seat (V RSeat) :Ok Ok :Speech (V RSpeech) :Verbs Verbs :Targets Targets :Items Items)
  (set TurnLeft (- (V TurnLeft) 1))
  (if (<= (V TurnLeft) 0)
    (CallFunction|PlayTurn)))
