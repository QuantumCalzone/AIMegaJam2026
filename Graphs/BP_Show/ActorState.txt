; The state of the show an AI actor reads, as a JSON string, for one ask.
; Written from the editor's read-back in T62 (traced from the function entry; T50's two old copies of the chain
; stay in the graph, cut off). T62 adds "holding": CastHolding fills it in each cast entry, HeldBy in "you",
; and HoldLine prints the state on turn 2 for the check.
(fn ActorState (Seat AskKey Now)
  (bind _show (Json|LoadJsonfromString "{}"))
  (Json|SetField _show "scene" (V SceneNum))
  (Json|SetField _show "scenes" (V SceneCount))
  (Json|SetField _show "now" Now)
  (bind _st (Json|LoadJsonfromString "{}"))
  (Json|SetField _st "show" _show)
  (Json|SetField _st "set" (V SetObj))
  (bind _card (Json|LoadJsonfromString "{}"))
  (Json|SetField _card "title" (V CardTitle))
  (Json|SetField _card "premise" (V CardPremise))
  (Json|SetField _card "note_category" (V NoteCategory))
  (Json|SetField _st "scene_card" _card)
  (CallFunction|CastHolding)
  (Json|SetField _st "cast" (V Cast))
  (bind _you (Json|LoadJsonfromString "{}"))
  (Json|SetField _you "name" (get (V SeatName) Seat))
  (Json|SetField _you "role" (get (V SeatRole) Seat))
  (Json|SetField _you "want" (get (V SeatWant) Seat))
  (Json|SetField _you "note" (get (V SeatNote) Seat))
  (bind _held (CallFunction|HeldBy :Seat Seat))
  (Json|SetField _you "holding" _held)
  (Json|SetField _st "you" _you)
  (Json|SetField _st "log" (V SharedLog))
  (Json|SetField _st "game_messages" (V GameMsgs))
  (Json|SetField _st "options" (V AskOpts))
  (bind _asks (Json|GetField (V Prompts) "asks"))
  (set TmpObj _asks)
  (bind _ask (Json|GetField (V TmpObj) AskKey))
  (set TmpStr _ask)
  (Json|SetField _st "ask" (V TmpStr))
  (bind _out (Json|GetJsonString _st))
  (bind _showstr (Json|GetJsonString _show))
  (CallFunction|StateLine :Seat Seat :AskKey AskKey :Show _showstr)
  (CallFunction|HoldLine :Seat Seat :AskKey AskKey :State _out)
  (return _out))
