; T68: after the switches, BPC_Analytics.Launch loads the player's save and sends launch (once per level load,
; the title and autoplay both pass here), before the title widget asks IsOn.
(fn ShowTitle ()
  (CallFunction|LoadSwitches)
  (Class|BPCAnalytics|Launch :self (V Analytics))
  (bind _rl (Class|BPCLLMClient|IsRelaunch :self (Variables|Default|GetLLMClient)))
  (if (or (Variables|Default|GetAutoplay) _rl)
    (if _rl
      (Development|PrintString "NHOS: TITLE skipped relaunch" false))
    (Class|BPCShowAudio|PlayMusic :self (V Audio) :Key "show")
    (CallFunction|SetupShow)
    (else
      (Class|BPCShowAudio|PlayMusic :self (V Audio) :Key "title")
      (bind _returnvalue (UserInterface|CreateWidget "/Game/NHOS/UI/WBP_Title.WBP_Title_C" (Game|GetPlayerController 0)))
      (Variables|Default|SetTitleScreen _returnvalue)
      (UserInterface|Viewport|AddtoViewport :self _returnvalue :ZOrder 20)
      (Class|WBPStage|SetPhase :self (Variables|Default|GetStage) :Phase "NO HUMANS ON SET")
      (Development|PrintString "NHOS: TITLE shown" false))))
