; T08: my Free Talk line (T58: and my last word). "pass" passes at once, like the Pass button.
; T63: any other line takes the spelling path (SpellAsk), speech only. Written from the editor's read-back in T63.
; T78: the face chip goes with the turn (TakeFace).
(fn SendHumanTalk (Line)
  (set HumanSent true)
  (CallFunction|TakeFace)
  (Class|WBPStage|SetPassEnabled :self (Variables|Default|GetStage) :Enabled false)
  (if (streq (Utilities|String|ToLower (Utilities|String|Trim (Utilities|String|TrimTrailing Line))) "pass")
    (CallFunction|PlayTalk :Seat (V HumanSeat) :Ok true :Speech "" :Pass true)
    (return))
  (CallFunction|SpellAsk :Line Line :InPhase "free talk"))
