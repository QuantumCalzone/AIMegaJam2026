; T30: the stage director call that was in ActorLine. BPC_LinePlayer calls it when the line plays.
; T77: first, the seat's robot shows the line's emotion (BP_Robot.Emote), read back from the entry's "feels" word
; (FeelsId; neutral when it has none). PlayCur calls this as the bubble shows and the line starts to type (T38, T73),
; so the face changes with them. Every line that plays gets it: scene turns, Free Talk, passes, last words, votes and
; reactions. A cut robot doesn't emote (T75).
; T87: the face part is EmoteSeat now, so the stage director call below runs again for every line (since T77 it sat
; in the seat check's else path and never ran).
(fn StageLine (Seat Entry)
  (set EmoTmp "")
  (bind _hf (Json|HasField Entry "feels"))
  (if _hf
    (bind _f (Json|GetField Entry "feels"))
    (set EmoTmp _f))
  (bind _id (CallFunction|FeelsId :Word (V EmoTmp)))
  (CallFunction|EmoteSeat :Seat Seat :Id _id)
  (bind _hd (Json|HasField Entry "does"))
  (if (not _hd)
    (return))
  (bind _dir (Actor|GetActorOfClass :ActorClass "/Game/NHOS/Stage/BP_StageDirector.BP_StageDirector_C"))
  (Utilities|IsValid _dir
    (:"Is Valid"
      (Class|BPStageDirector|PlayLine :self _dir :Seat Seat :Entry Entry :Robots (Variables|Default|GetRobots) :SeatNames (Variables|Default|GetSeatName) :Ids (Variables|Default|GetSeatId)))
    (:"Is Not Valid")))
