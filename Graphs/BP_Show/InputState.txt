; T07, T26: the input LLM's state. T63: it only fixes spelling, so it gets no places or props.
; Written from the editor's read-back in T63.
(fn InputState (Line InPhase)
  (bind _o (Json|LoadJsonfromString "{}"))
  (Json|SetField _o "actor" (get (V SeatName) (V HumanSeat)))
  (Json|SetField _o "phase" InPhase)
  (clear (V InputCast))
  (clear (V ScreenNames))
  (for _index (range (V SeatCount))
    (if (!= _index (V HumanSeat))
      (add (V InputCast) (get (V SeatName) _index))
      (add (V ScreenNames) (cat (get (V SeatLabel) _index) " is " (get (V SeatName) _index)))))
  (Json|SetField _o "cast" (V InputCast))
  (Json|SetField _o "screen_names" (V ScreenNames))
  (bind _typed (CallFunction|ResolveMentions :Text Line))
  (Json|SetField _o "typed" _typed)
  (bind _asks (Json|GetField (V Prompts) "asks"))
  (set TmpObj _asks)
  (bind _ask (Json|GetField (V TmpObj) "input"))
  (set TmpStr _ask)
  (Json|SetField _o "ask" (V TmpStr))
  (bind _s (Json|GetJsonString _o))
  (return _s))
