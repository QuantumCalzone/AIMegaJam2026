; WBP_UIStyleSheet.ShowKind: one button kind in its four states, side by side. Normal and Disabled are real buttons
; styled by UIStyle.StyleButton; Hover and Pressed are borders drawn with the same style's Hovered and Pressed brushes,
; because a screenshot can't hold a real button in those states.
(fn ShowKind (Kind BtnN LblN BrdH LblH BrdP LblP BtnD LblD)
  (bind s (Class|UIStyle|ButtonStyle :Kind Kind))
  (bind (bN bH bP bD fN fH fP fD padN padP sP sC sH) (Utilities|Struct|BreakButtonStyle s))
  (bind dark (or (Utilities|String|EqualExactly(String) Kind "Primary") (Utilities|String|EqualExactly(String) Kind "Paper")))
  (bind f (Class|UIStyle|UIFont :Role (select dark "ButtonInk" "Button")))
  (Class|UIStyle|StyleButton :Button BtnN :Label LblN :Kind Kind)
  (Class|UIStyle|StyleButton :Button BtnD :Label LblD :Kind Kind)
  (Widget|SetIsEnabled :self BtnD :bInIsEnabled false)
  (Class|Text|SetText :self LblN :Text (Utilities|Text|ToText(String) "Normal"))
  (Class|Text|SetText :self LblD :Text (Utilities|Text|ToText(String) "Disabled"))
  (Appearance|SetBrush :self BrdH :InBrush bH)
  (Class|Border|SetBrushColor :self BrdH :BrushColor (Utilities|Struct|MakeLinearColor 1.0 1.0 1.0 1.0))
  (Class|Border|SetPadding :self BrdH :Padding padN)
  (Class|Border|SetHorizontalAlignment :self BrdH :HorizontalAlignment "HAlign_Center")
  (Class|Border|SetVerticalAlignment :self BrdH :VerticalAlignment "VAlign_Center")
  (Class|Text|SetFont :self LblH :Font f)
  (Class|Text|SetColorandOpacity :self LblH :ColorAndOpacity fH)
  (Class|Text|SetText :self LblH :Text (Utilities|Text|ToText(String) "Hover"))
  (Appearance|SetBrush :self BrdP :InBrush bP)
  (Class|Border|SetBrushColor :self BrdP :BrushColor (Utilities|Struct|MakeLinearColor 1.0 1.0 1.0 1.0))
  (Class|Border|SetPadding :self BrdP :Padding padP)
  (Class|Border|SetHorizontalAlignment :self BrdP :HorizontalAlignment "HAlign_Center")
  (Class|Border|SetVerticalAlignment :self BrdP :VerticalAlignment "VAlign_Center")
  (Class|Text|SetFont :self LblP :Font f)
  (Class|Text|SetColorandOpacity :self LblP :ColorAndOpacity fP)
  (Class|Text|SetText :self LblP :Text (Utilities|Text|ToText(String) "Pressed")))
