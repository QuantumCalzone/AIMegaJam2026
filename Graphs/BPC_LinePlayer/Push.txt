; T30: BP_Show.ShowLine queues a chat line here instead of putting it on screen.
(fn Push (StageW Seat Name Color Action Speech IsMe Entry)
  (if (not (V Loaded))
    (CallFunction|LoadPacing))
  (CallFunction|FindShow)
  (set Stage StageW)
  (bind (_id _ty) (Class|BPShow|LineInfo :self (V Show) :Seat Seat :IsMe IsMe))
  (set TmpStr "line")
  (add (V QKind) (V TmpStr))
  (add (V LSeat) Seat)
  (add (V LName) Name)
  (add (V LColor) Color)
  (add (V LAction) Action)
  (add (V LSpeech) Speech)
  (add (V LIsMe) IsMe)
  (add (V LEntry) Entry)
  (add (V LId) _id)
  (add (V LTyping) _ty)
  (if (not (V Busy))
    (CallFunction|Pump)))
