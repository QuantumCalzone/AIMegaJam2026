; T30: empties the queue (BP_Show.RestartShow).
; T73: also stops a line that is still typing.
(fn Clear ()
  (Utilities|Time|ClearTimerbyFunctionName :Object self :FunctionName "Pump")
  (Utilities|Time|ClearTimerbyFunctionName :Object self :FunctionName "PlayCur")
  (clear (V QKind))
  (clear (V LSeat))
  (clear (V LName))
  (clear (V LColor))
  (clear (V LAction))
  (clear (V LSpeech))
  (clear (V LIsMe))
  (clear (V LEntry))
  (clear (V LId))
  (clear (V LTyping))
  (clear (V CText))
  (clear (V CNum))
  (CallFunction|StopType)
  ; T38: the last bubble fades too.
  (CallFunction|HushBubble)
  (set Busy false)
  (CallFunction|ClearTyping))
