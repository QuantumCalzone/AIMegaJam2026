; T30: reads the pacing block from Content/NHOS/Data/pacing.json and fast_lines from Local/switches.json. T73: the typing values.
(fn LoadPacing ()
  (set Loaded true)
  (set LineBase 0.6)
  (set LinePerWord 0.12)
  (set LineMin 1.5)
  (set LineMax 2.5)
  (set TypeMin 0.6)
  (set TypeMax 1.6)
  (set FastScale 0.1)
  (set TypeCps 30.0)
  (set PauseComma 0.15)
  (set PauseStop 0.3)
  (set SylRate 12.0)
  (set Fast false)
  (bind _empty (Json|LoadJsonfromString "{}"))
  (set CueHold _empty)
  (bind _cdir (Utilities|Paths|GetProjectContentDirectory))
  (bind (_o _ok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat _cdir "NHOS/Data/pacing.json"))))
  (if _ok
    (bind _hp (Json|HasField _o "pacing"))
    (if _hp
      (bind _p (Json|GetField _o "pacing"))
      (set Pace _p)
      (bind _f1 (Json|GetField (V Pace) "line_base"))
      (set LineBase _f1)
      (bind _f2 (Json|GetField (V Pace) "line_per_word"))
      (set LinePerWord _f2)
      (bind _f3 (Json|GetField (V Pace) "line_min"))
      (set LineMin _f3)
      (bind _f4 (Json|GetField (V Pace) "line_max"))
      (set LineMax _f4)
      (bind _f5 (Json|GetField (V Pace) "typing_min"))
      (set TypeMin _f5)
      (bind _f6 (Json|GetField (V Pace) "typing_max"))
      (set TypeMax _f6)
      (bind _f7 (Json|GetField (V Pace) "fast_scale"))
      (set FastScale _f7)
      (bind _f8 (Json|GetField (V Pace) "cue_hold"))
      (set CueHold _f8)
      (set TypeCps (Math|Float|Max(Float) (CallFunction|PaceF :Key "type_cps" :Def 30.0) 1.0))
      (set PauseComma (CallFunction|PaceF :Key "type_pause_comma" :Def 0.15))
      (set PauseStop (CallFunction|PaceF :Key "type_pause_stop" :Def 0.3))
      (set SylRate (Math|Float|Max(Float) (CallFunction|PaceF :Key "syl_max_rate" :Def 12.0) 1.0))))
  (bind _dir (Utilities|Paths|GetProjectDirectory))
  (bind (_so _sok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat _dir "Local/switches.json"))))
  (if _sok
    (bind _hf (Json|HasField _so "fast_lines"))
    (if _hf
      (bind _sf (Json|GetField _so "fast_lines"))
      (set Fast _sf)))
  (print (cat "NHOS: PACING file=" (Utilities|String|ToString(Boolean) _ok) " fast_lines=" (Utilities|String|ToString(Boolean) (V Fast)) " type_cps=" (Utilities|String|ToString(Float) (V TypeCps)))))
