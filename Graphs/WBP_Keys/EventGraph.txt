(event UserInterface|EventTick (MyGeometry InDeltaTime)
  (CallFunction|RowStatus :Key (Variables|WBP_Keys|GetKeyOpenAI) :Status (Variables|WBP_Keys|GetStatusOpenAI))
  (CallFunction|RowStatus :Key (Variables|WBP_Keys|GetKeyAnthropic) :Status (Variables|WBP_Keys|GetStatusAnthropic))
  (CallFunction|RowStatus :Key (Variables|WBP_Keys|GetKeySpaceXAI) :Status (Variables|WBP_Keys|GetStatusSpaceXAI))
  (CallFunction|RowStatus :Key (Variables|WBP_Keys|GetKeyMeta) :Status (Variables|WBP_Keys|GetStatusMeta))
  (CallFunction|ShowField :Field (Variables|WBP_Keys|GetFieldOpenAI) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Keys|GetKeyOpenAI)))
  (CallFunction|ShowField :Field (Variables|WBP_Keys|GetFieldAnthropic) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Keys|GetKeyAnthropic)))
  (CallFunction|ShowField :Field (Variables|WBP_Keys|GetFieldSpaceXAI) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Keys|GetKeySpaceXAI)))
  (CallFunction|ShowField :Field (Variables|WBP_Keys|GetFieldMeta) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Keys|GetKeyMeta)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Keys|GetSaveRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Keys|GetSaveButton)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Keys|GetTestRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Keys|GetTestButton)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Keys|GetBackRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Keys|GetBackButton)))
  (if (V Testing)
    (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
    (bind _cl (Class|BPShow|GetLLMClient :self _show))
    (bind (_done _ok _note) (Class|BPCLLMClient|KeyTestResult :self _cl))
    (if _done
      (set Testing false)
      (Widget|SetText(Text) :self (Variables|WBP_Keys|GetKeysNote) :InText (Utilities|Text|ToText(String) _note))
      (print (cat "NHOS: KEYS tested ok=" (Utilities|String|ToString(Boolean) _ok)))
      (if (and _ok (V StartAfter))
        (print "NHOS: KEYS start")
        (Widget|RemovefromParent :self self)
        (Class|BPCLLMClient|CloseUnavailable :self _cl)
        (bind _re (Class|BPCLLMClient|Relaunch :self _cl :NewMode "keys"))
        (if (not _re)
          (Class|BPShow|StartfromTitle :self _show))))))

(event OnClicked(SaveButton)
  (print "NHOS: KEYS click save")
  (if (not (V Testing))
    (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
    (bind _cl (Class|BPShow|GetLLMClient :self _show))
    (Class|BPCLLMClient|SetKey :self _cl :Prov "openai" :Key (Utilities|String|ToString(Text) (Widget|GetText(TextBox) :self (Variables|WBP_Keys|GetKeyOpenAI))))
    (Class|BPCLLMClient|SetKey :self _cl :Prov "anthropic" :Key (Utilities|String|ToString(Text) (Widget|GetText(TextBox) :self (Variables|WBP_Keys|GetKeyAnthropic))))
    (Class|BPCLLMClient|SetKey :self _cl :Prov "xai" :Key (Utilities|String|ToString(Text) (Widget|GetText(TextBox) :self (Variables|WBP_Keys|GetKeySpaceXAI))))
    (Class|BPCLLMClient|SetKey :self _cl :Prov "meta" :Key (Utilities|String|ToString(Text) (Widget|GetText(TextBox) :self (Variables|WBP_Keys|GetKeyMeta))))
    (bind _saved (Class|BPCLLMClient|SaveKeys :self _cl))
    (bind (_ok _err) (Class|BPCLLMClient|SetMode :self _cl :InMode "keys"))
    (if _ok
      (set StartAfter true)
      (set Testing true)
      (Widget|SetText(Text) :self (Variables|WBP_Keys|GetKeysNote) :InText (Utilities|Text|ToText(String) "Testing your keys: one short call to each lab..."))
      (Class|BPCLLMClient|TestKeys :self _cl)
      (else
        (Widget|SetText(Text) :self (Variables|WBP_Keys|GetKeysNote) :InText (Utilities|Text|ToText(String) _err))))))
