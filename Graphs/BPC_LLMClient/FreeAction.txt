(fn FreeAction ()
  (if (== (len (V FreeActions)) 0)
    (return))
  (set WTotal 0.0)
  (for _a (V FreeActions)
    (bind _w (Json|GetField _a "weight"))
    (set RandF _w)
    (set WTotal (+ (V WTotal) (V RandF))))
  (set WPick (Math|Random|RandomFloatinRange 0.0 (V WTotal)))
  (set TmpObj (get (V FreeActions) (- (len (V FreeActions)) 1)))
  (set WDone false)
  (for _a2 (V FreeActions)
    (if (not (V WDone))
      (bind _w2 (Json|GetField _a2 "weight"))
      (set RandF _w2)
      (set WPick (- (V WPick) (V RandF)))
      (if (<= (V WPick) 0.0)
        (set TmpObj _a2)
        (set WDone true))))
  (bind _vb (Json|GetField (V TmpObj) "verb"))
  (set AVerb _vb)
  (bind _tg (Json|GetField (V TmpObj) "target"))
  (set ATarget _tg)
  (bind _it (Json|GetField (V TmpObj) "item"))
  (set AItem _it)
  (if (and (Utilities|String|Contains :SearchIn (V ATarget) :Substring "{actor}") (== (len (V FOthers)) 0))
    (return))
  (if (and (Utilities|String|Contains :SearchIn (V AItem) :Substring "{prop}") (== (len (V FProps)) 0))
    (return))
  (if (and (Utilities|String|Contains :SearchIn (V ATarget) :Substring "{place}") (== (len (V FPlaces)) 0))
    (return))
  (bind _ft (CallFunction|FreeFill :Text (V ATarget)))
  (set TmpStr _ft)
  (bind _fi (CallFunction|FreeFill :Text (V AItem)))
  (set TmpStr2 _fi)
  (add (V RVerbs) (V AVerb))
  (add (V RTargets) (V TmpStr))
  (add (V RItems) (V TmpStr2)))
