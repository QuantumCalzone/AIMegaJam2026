(fn Relaunch (NewMode)
  (if (not (V ShowOn))
    (return false))
  (print (cat "NHOS: RELAUNCH mode=" NewMode))
  (if (Utilities|String|IsEmpty NewMode)
    (Game|OpenLevel(byName) :LevelName "L_Stage")
    (return true))
  (Game|OpenLevel(byName) :LevelName "L_Stage" :Options (cat "nhosmode=" NewMode))
  (return true))
