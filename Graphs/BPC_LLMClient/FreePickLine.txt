; T107: picks the index (FreeIdx) of a free line in TmpArr2 (the phase's 'lines', or a vote's 'reasons') that is none of the last FreeK
; lines this phase gave, FreeK = min(lines - 1, 3). A round has at most four AI speakers, so no two seats get the
; same line in one round of one phase while the phase has unused lines. FreeRecent holds '<phase id>#<index>' per pick,
; newest last, at most 40. The relay's free mode answers one request at a time and keeps no such list.
(fn FreePickLine ()
  (bind _id (Json|GetField (V FreePhase) "id"))
  (set FreeId _id)
  (bind _n (len (V TmpArr2)))
  (set FreeK (Math|Integer|Min(Integer) (- _n 1) 3))
  (clear (V FreeCands))
  (for _i (range _n)
    (set FreeHit false)
    (set FreeSeen 0)
    (for _r (range (len (V FreeRecent)))
      (bind _e (get (V FreeRecent) (- (- (len (V FreeRecent)) 1) _r)))
      (if (and (< (V FreeSeen) (V FreeK)) (Utilities|String|StartsWith :SourceString _e :InPrefix (cat (V FreeId) "#")))
        (set FreeSeen (+ (V FreeSeen) 1))
        (if (streq _e (cat (V FreeId) "#" (str _i)))
          (set FreeHit true))))
    (if (not (V FreeHit))
      (add (V FreeCands) _i)))
  (set FreeIdx 0)
  (if (> (len (V FreeCands)) 0)
    (set FreeIdx (get (V FreeCands) (Math|Random|RandomIntegerinRange 0 (- (len (V FreeCands)) 1)))))
  (add (V FreeRecent) (cat (V FreeId) "#" (str (V FreeIdx))))
  (if (> (len (V FreeRecent)) 40)
    (Utilities|Array|RemoveIndex (V FreeRecent) 0))
  (print (cat "NHOS: FREE LINE phase=" (V FreeId) " idx=" (str (V FreeIdx)))))
