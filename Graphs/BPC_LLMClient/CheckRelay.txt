(fn CheckRelay ()
  (set Health "wait")
  (set HealthMsg "")
  (if (Utilities|String|IsEmpty (V RelayUrl))
    (set Health "fail")
    (set HealthMsg "Play via Relay needs the relay address in llm.json.")
    (return))
  (print "NHOS: HEALTH check")
  (bind _req (Game|AddComponentbyClass (Components|GetOwner) "/Game/NHOS/Core/BPC_HttpGet.BPC_HttpGet_C"))
  (Class|BPCHttpGet|Setup :self _req :InTag "health" :InUrl (cat (V RelayUrl) "/health") :InHeaders (V FreeHeaders) :InBody "" :InClient self)
  (Class|BPCHttpGet|Send :self _req))
