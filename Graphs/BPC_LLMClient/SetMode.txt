(fn SetMode (InMode)
  (set Mode InMode)
  (if (streq InMode "relay")
    (if (Utilities|String|IsEmpty (V RelayUrl))
      (return false "Play via Relay needs the relay address in llm.json.")))
  (if (streq InMode "keys")
    (bind _n (CallFunction|KeyCount))
    (if (< _n 3)
      (return false "Enter keys for at least 3 of the 4 AI providers.")))
  (print (cat "NHOS: MODE " InMode))
  (return true ""))
