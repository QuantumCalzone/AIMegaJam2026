; T37: too few keys takes its line from errors.json.
; T68: a pick that succeeds reports the analytics mode event (BPC_Analytics.ModePicked).
(fn SetMode (InMode)
  (set Mode InMode)
  (if (streq InMode "relay")
    (if (Utilities|String|IsEmpty (V RelayUrl))
      (return false "Play needs the relay address in llm.json.")))
  (if (streq InMode "keys")
    (bind _n (CallFunction|KeyCount))
    (if (< _n 3)
      (print "NHOS: ERROR too_few_keys provider= where=keytest")
      (bind _fk (CallFunction|ErrorLine :Kind "keytest_too_few_keys" :Lab "" :N 0 :Type ""))
      (return false _fk)))
  (print (cat "NHOS: MODE " InMode))
  (if (V HaveStats)
    (Class|BPCAnalytics|ModePicked :self (V Stats)))
  (return true ""))
