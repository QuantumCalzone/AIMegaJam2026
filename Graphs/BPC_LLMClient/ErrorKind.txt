; T37: sorts a failed ask into a case of errors.json. Reads the reason HandleHttp built (Reason), the error's
; type and message (ErrType, ErrMsg, from ParseReply) and the relay's upstream_status (UpStatus).
; The provider's message is read only for keywords and never shown: some providers echo part of the key.
(fn ErrorKind (Reason)
  (if (Utilities|String|StartsWith :SourceString Reason :InPrefix "transport")
    (return "no_reply"))
  (if (or (streq (V ErrType) "session_limit") (or (streq (V ErrType) "player_limit") (or (streq (V ErrType) "budget_used_up") (streq (V ErrType) "provider_unavailable"))))
    (return (V ErrType)))
  (set EBody (or (Utilities|String|StartsWith :SourceString Reason :InPrefix "reply is not JSON") (or (Utilities|String|StartsWith :SourceString Reason :InPrefix "no text block") (or (Utilities|String|StartsWith :SourceString Reason :InPrefix "no choices") (Utilities|String|StartsWith :SourceString Reason :InPrefix "empty choices")))))
  (set EHay (select (V EBody) (cat (V ErrMsg) " " Reason) (V ErrMsg)))
  (set EHit (or (streq (V ErrType) "insufficient_quota") (streq (V ErrType) "billing_error")))
  (set EHit (or (V EHit) (or (Utilities|String|Contains :SearchIn (V EHay) :Substring "credit") (Utilities|String|Contains :SearchIn (V EHay) :Substring "quota"))))
  (set EHit (or (V EHit) (or (Utilities|String|Contains :SearchIn (V EHay) :Substring "billing") (or (Utilities|String|Contains :SearchIn (V EHay) :Substring "balance") (Utilities|String|Contains :SearchIn (V EHay) :Substring "spending limit")))))
  (if (V EHit)
    (return "out_of_credits"))
  (if (== (V UpStatus) 429)
    (return "relay_429"))
  (set EHit (or (streq (V ErrType) "rate_limit_error") (or (streq (V ErrType) "rate_limit_exceeded") (or (streq (V ErrType) "requests") (streq (V ErrType) "tokens")))))
  (set EHit (or (V EHit) (or (Utilities|String|Contains :SearchIn (V EHay) :Substring "rate limit") (Utilities|String|Contains :SearchIn (V EHay) :Substring "too many requests"))))
  (if (V EHit)
    (return "rate_limited"))
  (set EHit (or (streq (V ErrType) "overloaded_error") (or (== (V UpStatus) 503) (== (V UpStatus) 529))))
  (set EHit (or (V EHit) (or (Utilities|String|Contains :SearchIn (V EHay) :Substring "overloaded") (or (Utilities|String|Contains :SearchIn (V EHay) :Substring "capacity") (Utilities|String|Contains :SearchIn (V EHay) :Substring "high demand")))))
  (if (V EHit)
    (return "overloaded"))
  (set EHit (or (streq (V ErrType) "authentication_error") (or (streq (V ErrType) "permission_error") (or (streq (V ErrType) "invalid_api_key") (Utilities|String|Contains :SearchIn (V EHay) :Substring "api key")))))
  (if (V EHit)
    (return "key_rejected"))
  (if (or (Utilities|String|StartsWith :SourceString Reason :InPrefix "finish_reason length") (Utilities|String|StartsWith :SourceString Reason :InPrefix "stop_reason max_tokens"))
    (return "cut_off"))
  (if (or (Utilities|String|StartsWith :SourceString Reason :InPrefix "refusal") (or (Utilities|String|StartsWith :SourceString Reason :InPrefix "stop_reason refusal") (Utilities|String|StartsWith :SourceString Reason :InPrefix "finish_reason content_filter")))
    (return "refused"))
  (if (V EBody)
    (return "unreadable"))
  (if (streq (V ErrType) "provider_error")
    (return "relay_provider"))
  (return "unknown"))
