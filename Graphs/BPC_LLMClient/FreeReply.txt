(fn FreeReply (User)
  (bind (_st _ok) (Json|LoadJsonfromString User))
  (set FreeState _st)
  (set FreeMe "")
  (set FreeVote "")
  (set FreeAsk "")
  (bind _hk (Json|HasField (V FreeState) "ask"))
  (if _hk
    (bind _ak (Json|GetField (V FreeState) "ask"))
    (set FreeAsk _ak))
  (bind _found (CallFunction|FreeFindPhase))
  (bind _he (Json|HasField (V FreePhase) "echo_typed"))
  (if _he
    (bind _ty (Json|GetField (V FreeState) "typed"))
    (set TmpStr _ty)
    (set TmpArr (Utilities|String|ParseIntoArray :SourceString (V TmpStr) :Delimiter " " :CullEmptyStrings true))
    (clear (V TmpArr2))
    (for _wi (range (Math|Integer|Min(Integer) (len (V TmpArr)) 25))
      (add (V TmpArr2) (get (V TmpArr) _wi)))
    (set RSpeech (Utilities|String|JoinStringArray (V TmpArr2) " "))
    (return))
  (bind _hy (Json|HasField (V FreeState) "you"))
  (if _hy
    (bind _y (Json|GetField (V FreeState) "you"))
    (set TmpObj _y)
    (bind _yn (Json|GetField (V TmpObj) "name"))
    (set FreeMe _yn))
  (clear (V FOthers))
  (bind _hc (Json|HasField (V FreeState) "cast"))
  (if _hc
    (bind _c (Json|GetField (V FreeState) "cast"))
    (set TmpObjs _c)
    (for _o (V TmpObjs)
      (bind _cn (Json|GetField _o "name"))
      (set TmpStr _cn)
      (if (not (streq (V TmpStr) (V FreeMe)))
        (add (V FOthers) (V TmpStr)))))
  (clear (V FOpts))
  (bind _ho (Json|HasField (V FreeState) "options"))
  (if _ho
    (bind _op (Json|GetField (V FreeState) "options"))
    (set FOpts _op))
  (clear (V FPlaces))
  (clear (V FProps))
  (bind _hs (Json|HasField (V FreeState) "set"))
  (if _hs
    (bind _s (Json|GetField (V FreeState) "set"))
    (set TmpObj _s)
    (bind _pl (Json|GetField (V TmpObj) "places"))
    (set TmpObjs _pl)
    (for _p (V TmpObjs)
      (bind _pn (Json|GetField _p "name"))
      (set TmpStr _pn)
      (add (V FPlaces) (V TmpStr)))
    (bind _pr (Json|GetField (V TmpObj) "props"))
    (set FProps _pr))
  (bind _hv (Json|HasField (V FreePhase) "vote"))
  (if _hv
    (clear (V TmpArr))
    (for _v (V FOpts)
      (if (not (streq _v (V FreeMe)))
        (add (V TmpArr) _v)))
    (if (== (len (V TmpArr)) 0)
      (set TmpArr (V FOpts)))
    (if (> (len (V TmpArr)) 0)
      (set FreeVote (get (V TmpArr) (Math|Random|RandomIntegerinRange 0 (- (len (V TmpArr)) 1)))))
    (set RVote (V FreeVote))
    (bind _rs (Json|GetField (V FreePhase) "reasons"))
    (set TmpArr2 _rs)
    (if (> (len (V TmpArr2)) 0)
      (set TmpStr (get (V TmpArr2) (Math|Random|RandomIntegerinRange 0 (- (len (V TmpArr2)) 1))))
      (bind _rf (CallFunction|FreeFill :Text (V TmpStr)))
      (set RReason _rf))
    (return))
  (bind _hpc (Json|HasField (V FreePhase) "pass_chance"))
  (if _hpc
    (bind _pc (Json|GetField (V FreePhase) "pass_chance"))
    (set RandF _pc)
    (if (< (Math|Random|RandomFloat) (V RandF))
      (set RPass true)
      (return)))
  (bind _hl (Json|HasField (V FreePhase) "lines"))
  (if _hl
    (bind _ls (Json|GetField (V FreePhase) "lines"))
    (set TmpArr2 _ls)
    (if (> (len (V TmpArr2)) 0)
      (set TmpStr (get (V TmpArr2) (Math|Random|RandomIntegerinRange 0 (- (len (V TmpArr2)) 1))))
      (bind _fl (CallFunction|FreeFill :Text (V TmpStr)))
      (set RSpeech _fl)))
  (set ALo 0)
  (set AHi 0)
  (bind _hac (Json|HasField (V FreePhase) "actions"))
  (if _hac
    (bind _ac (Json|GetField (V FreePhase) "actions"))
    (set TmpInts _ac)
    (if (>= (len (V TmpInts)) 2)
      (set ALo (get (V TmpInts) 0))
      (set AHi (get (V TmpInts) 1))))
  (set ACnt (Math|Random|RandomIntegerinRange (V ALo) (V AHi)))
  (for _k (range (V ACnt))
    (CallFunction|FreeAction)))
