; T37: one line of errors.json's case Kind, picked at random, with {lab}, {n} and {type} filled in.
; A screen's own case (health_, down_ or keytest_ + kind) falls back to the plain kind. "" if neither exists.
(fn ErrorLine (Kind Lab N Type)
  (set ELKey Kind)
  (bind _has (Json|HasField (V ErrCases) Kind))
  (if (not _has)
    (bind (_left _right _found) (Utilities|String|Split :SourceString Kind :InStr "_"))
    (set ELKey _right))
  (bind _has2 (Json|HasField (V ErrCases) (V ELKey)))
  (if (not _has2)
    (print (cat "NHOS: ERROR no line for " Kind))
    (return ""))
  (bind _ls (Json|GetField (V ErrCases) (V ELKey)))
  (set ELArr _ls)
  (if (== (len (V ELArr)) 0)
    (return ""))
  (set ELine (get (V ELArr) (Math|Random|RandomIntegerinRange 0 (- (len (V ELArr)) 1))))
  (set ELine (Utilities|String|Replace :SourceString (V ELine) :From "{lab}" :To Lab))
  (set ELine (Utilities|String|Replace :SourceString (V ELine) :From "{n}" :To (str N)))
  (set ELine (Utilities|String|Replace :SourceString (V ELine) :From "{type}" :To Type))
  (return (V ELine)))
