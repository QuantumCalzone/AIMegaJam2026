; T37: also loads errors.json (cases, test_replies) and the error_test switch.
; T68: also finds the analytics component (Stats, FindStats).
; T69: also input_schema (the input ask's reply shape) and the attempt_test switch.
; T77: also free_mode.json emotion_keywords, the table that picks my emotion from my line.
(fn LoadConfig ()
  (bind _dir (Utilities|Paths|GetProjectDirectory))
  (bind _cdir (Utilities|Paths|GetProjectContentDirectory))
  (bind _cfg (cat _cdir "NHOS/Data/llm.json"))
  (set Ready false)
  (set Mode "free")
  (bind (_o _ok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath _cfg)))
  (if (not _ok)
    (return false (cat "config file missing or not valid JSON: " _cfg)))
  (bind _f1 (Json|GetField _o "api_version"))
  (set ApiVersion _f1)
  (bind _f2 (Json|GetField _o "beta"))
  (set Beta _f2)
  (bind _f3 (Json|GetField _o "fallbacks"))
  (set Fallbacks _f3)
  (bind _f4 (Json|GetField _o "reply_schema"))
  (set ReplySchema _f4)
  (set InputSchema (V ReplySchema))
  (bind _his (Json|HasField _o "input_schema"))
  (if _his
    (bind _fis (Json|GetField _o "input_schema"))
    (set InputSchema _fis))
  (bind _f5 (Json|GetField _o "roles"))
  (set Roles _f5)
  (bind _hp (Json|HasField _o "providers"))
  (if (not _hp)
    (return false (cat "no providers in " _cfg)))
  (bind _f6 (Json|GetField _o "providers"))
  (set Providers _f6)
  (set RelayUrl "")
  (bind _hr (Json|HasField _o "relay"))
  (if _hr
    (bind _f7 (Json|GetField _o "relay"))
    (set RelayUrl _f7))
  (clear (V PIds))
  (clear (V PProv))
  (clear (V PNames))
  (bind (_po _pok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat _cdir "NHOS/Data/personas.json"))))
  (if (not _pok)
    (return false "personas.json missing or not valid JSON"))
  (bind _pa (Json|GetField _po "personas"))
  (set TmpObjs _pa)
  (for _p (V TmpObjs)
    (bind _pid (Json|GetField _p "id"))
    (set TmpStr _pid)
    (add (V PIds) (V TmpStr))
    (bind _ppr (Json|GetField _p "provider"))
    (set TmpStr _ppr)
    (add (V PProv) (V TmpStr))
    (bind _pnm (Json|GetField _p "name"))
    (set TmpStr _pnm)
    (add (V PNames) (V TmpStr)))
  (clear (V EmoKeys))
  (bind (_fo _fok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat _cdir "NHOS/Data/free_mode.json"))))
  (if _fok
    (bind _fph (Json|GetField _fo "phases"))
    (set FreePhases _fph)
    (bind _fac (Json|GetField _fo "scene_actions"))
    (set FreeActions _fac)
    (bind _hek (Json|HasField _fo "emotion_keywords"))
    (if _hek
      (bind _fek (Json|GetField _fo "emotion_keywords"))
      (set EmoKeys _fek)))
  (bind (_eo _eok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat _cdir "NHOS/Data/errors.json"))))
  (if _eok
    (bind _ec (Json|GetField _eo "cases"))
    (set ErrCases _ec)
    (bind _etr (Json|GetField _eo "test_replies"))
    (set ErrTests _etr))
  (bind (_ko _kok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat _dir "Local/llm_key.json"))))
  (if _kok
    (set KeyObj _ko)
    (else
      (bind _ke (Json|LoadJsonfromString "{}"))
      (set KeyObj _ke)))
  (set BurnTest false)
  (set ErrTest "")
  (set AttemptTest "")
  (bind (_so _sok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat _dir "Local/switches.json"))))
  (if _sok
    (bind _hm (Json|HasField _so "mode"))
    (if _hm
      (bind _sm (Json|GetField _so "mode"))
      (set Mode _sm))
    (bind _hb (Json|HasField _so "burn_test"))
    (if _hb
      (bind _sb (Json|GetField _so "burn_test"))
      (set BurnTest _sb))
    (bind _het (Json|HasField _so "error_test"))
    (if _het
      (bind _set (Json|GetField _so "error_test"))
      (set ErrTest _set))
    (bind _hat (Json|HasField _so "attempt_test"))
    (if _hat
      (bind _sat (Json|GetField _so "attempt_test"))
      (set AttemptTest _sat)))
  (set Relaunched false)
  (bind _gm (Game|GetGameMode))
  (bind _opt (GameOptions|ParseOption :Options (Class|GameModeBase|GetOptionsString :self _gm) :Key "nhosmode"))
  (set TmpStr _opt)
  (if (or (streq (V TmpStr) "free") (streq (V TmpStr) "keys"))
    (set Mode (V TmpStr))
    (set Relaunched true)
    (print (cat "NHOS: RELAUNCHED mode=" (V TmpStr))))
  (if (not (or (streq (V Mode) "relay") (streq (V Mode) "keys")))
    (set Mode "free"))
  (set FreeHeaders (V EmptyHeaders))
  (Http|AddHeader (V FreeHeaders) "content-type" "application/json")
  (set Session (Guid|ToString(Guid) (Guid|NewGuid)))
  (CallFunction|FindStats)
  (set Ready true)
  (bind _n (CallFunction|KeyCount))
  (print (cat "NHOS: CLIENT ready mode=" (V Mode) " keys=" (str _n) " burn_test=" (Utilities|String|ToString(Boolean) (V BurnTest)) " error_test=" (V ErrTest) " attempt_test=" (V AttemptTest) " emotion_keywords=" (str (len (V EmoKeys)))))
  (return true ""))
