(fn ParseReply (Response)
  (bind (_o _ok) (Json|LoadJsonfromString Response))
  (if (not _ok)
    (return false "" (cat "reply is not JSON: " Response)))
  (set TmpStr "")
  (bind _ht (Json|HasField _o "type"))
  (if _ht
    (bind _t (Json|GetField _o "type"))
    (set TmpStr _t))
  (bind _he (Json|HasField _o "error"))
  (bind _hc (Json|HasField _o "choices"))
  (if (or (streq (V TmpStr) "error") (and _he (not _hc)))
    (bind _e (Json|GetField _o "error"))
    (set TmpObj _e)
    (set TmpStr2 "")
    (bind _het (Json|HasField (V TmpObj) "type"))
    (if _het
      (bind _et (Json|GetField (V TmpObj) "type"))
      (set TmpStr2 _et))
    (bind _m (Json|GetField (V TmpObj) "message"))
    (set TmpStr _m)
    (return false "" (cat "API error " (V TmpStr2) ": " (V TmpStr))))
  (bind _hu (Json|HasField _o "usage"))
  (if (streq (V CurFmt) "openai_chat")
    (if _hu
      (bind _u (Json|GetField _o "usage"))
      (set TmpObj _u)
      (bind _pt (Json|GetField (V TmpObj) "prompt_tokens"))
      (set UsageIn _pt)
      (bind _ct (Json|GetField (V TmpObj) "completion_tokens"))
      (set UsageOut _ct))
    (if (not _hc)
      (return false "" (cat "no choices in reply: " Response)))
    (bind _ch (Json|GetField _o "choices"))
    (set TmpObjs _ch)
    (if (== (len (V TmpObjs)) 0)
      (return false "" "empty choices in reply"))
    (set TmpObj (get (V TmpObjs) 0))
    (bind _msg (Json|GetField (V TmpObj) "message"))
    (set TmpObj2 _msg)
    (set ChatText "")
    (bind _hct (Json|HasField (V TmpObj2) "content"))
    (if _hct
      (bind _cc0 (Json|GetField (V TmpObj2) "content"))
      (set ChatText _cc0))
    (set TmpStr "")
    (bind _hf (Json|HasField (V TmpObj) "finish_reason"))
    (if _hf
      (bind _fr (Json|GetField (V TmpObj) "finish_reason"))
      (set TmpStr _fr))
    (if (or (streq (V TmpStr) "length") (streq (V TmpStr) "content_filter"))
      (return false "" (cat "finish_reason " (V TmpStr) " out=" (V UsageOut) " chars=" (str (Utilities|String|Len (V ChatText))) " starts=" (Utilities|String|GetSubstring :SourceString (V ChatText) :StartIndex 0 :Length 120))))
    (set TmpStr2 "")
    (bind _hrf (Json|HasField (V TmpObj2) "refusal"))
    (if _hrf
      (bind _rf (Json|GetField (V TmpObj2) "refusal"))
      (set TmpStr2 _rf))
    (if (not (Utilities|String|IsEmpty (V TmpStr2)))
      (return false "" (cat "refusal " (V TmpStr2))))
    (return true (V ChatText) ""))
  (if _hu
    (bind _u2 (Json|GetField _o "usage"))
    (set TmpObj _u2)
    (bind _it (Json|GetField (V TmpObj) "input_tokens"))
    (set UsageIn _it)
    (bind _ot (Json|GetField (V TmpObj) "output_tokens"))
    (set UsageOut _ot))
  (bind _sr (Json|GetField _o "stop_reason"))
  (set TmpStr _sr)
  (if (or (streq (V TmpStr) "refusal") (streq (V TmpStr) "max_tokens"))
    (return false "" (cat "stop_reason " (V TmpStr))))
  (bind _cb (Json|GetField _o "content"))
  (set ContentBlocks _cb)
  (for _blk (V ContentBlocks)
    (bind _bt (Json|GetField _blk "type"))
    (set TmpStr _bt)
    (if (streq (V TmpStr) "text")
      (bind _tx (Json|GetField _blk "text"))
      (set TmpStr _tx)
      (return true (V TmpStr) "")))
  (return false "" (cat "no text block in reply: " Response)))
