; T37: the unavailable screen's line when the relay's health check fails (OnHealth).
; T68: also sends the analytics error event.
; T93: keeps the kind (HealthKind) for the title's line (TitleLine). The title checks every 60 s, so the
; analytics error goes out only when the kind changes (HealthSent; OnHealth clears it on a good answer).
(fn HealthFail (Kind N)
  (set HealthKind Kind)
  (bind _l (CallFunction|ErrorLine :Kind (cat "health_" Kind) :Lab "" :N N :Type ""))
  (set HealthMsg _l)
  (print (cat "NHOS: ERROR " Kind " provider= where=health"))
  (if (and (V HaveStats) (not (streq Kind (V HealthSent))))
    (set HealthSent Kind)
    (Class|BPCAnalytics|Error :self (V Stats) :Kind (cat "health_" Kind))))
