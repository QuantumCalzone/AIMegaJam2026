(fn AskProvider (Tag Role)
  (if (not (streq Role "actor"))
    (bind _rp (CallFunction|RoleProvider :Role Role))
    (return _rp))
  (bind (_left _right _found) (Utilities|String|Split :SourceString Tag :InStr "-" :SearchCase "IgnoreCase" :SearchDir "FromEnd"))
  (set APIdx (Utilities|Array|FindItem (V PIds) _left))
  (if (< (V APIdx) 0)
    (return ""))
  (return (get (V PProv) (V APIdx))))
