(fn Ask (Tag Role System User)
  (set CurProv "")
  (bind _hr (Json|HasField (V Roles) Role))
  (if (not (V Ready))
    (CallFunction|ClearReply)
    (CallFunction|Deliver :Tag Tag :Ok false :Error "AI service unavailable" :Seconds "0")
    (return))
  (if (not _hr)
    (CallFunction|ClearReply)
    (CallFunction|Deliver :Tag Tag :Ok false :Error (cat "unknown role " Role) :Seconds "0")
    (return))
  (bind _prov (CallFunction|AskProvider :Tag Tag :Role Role))
  (set CurProv _prov)
  (if (Utilities|String|IsEmpty (V CurProv))
    (CallFunction|ClearReply)
    (CallFunction|Deliver :Tag Tag :Ok false :Error "no provider for this ask" :Seconds "0")
    (return))
  (if (streq (V Mode) "free")
    (add (V LocalTags) Tag)
    (add (V LocalUsers) User)
    (add (V LocalProv) (V CurProv))
    (Utilities|Time|SetTimerbyFunctionName :Object self :FunctionName "FlushLocal" :Time 0.3 :bLooping false)
    (return))
  (bind _key (CallFunction|ProvKey :Prov (V CurProv)))
  (set AskKey _key)
  (if (and (streq (V Mode) "keys") (Utilities|String|IsEmpty (V AskKey)))
    (CallFunction|ClearReply)
    (CallFunction|Deliver :Tag Tag :Ok false :Error "no API key for this provider" :Seconds "0")
    (return))
  (bind _body (CallFunction|BuildBody :System System :User User :Role Role :Prov (V CurProv)))
  (set AskBody _body)
  (if (streq (V Mode) "relay")
    (bind _env (Json|LoadJsonfromString "{}"))
    (bind _bo (Json|LoadJsonfromString (V AskBody)))
    (Json|SetField _env "provider" (V CurProv))
    (Json|SetField _env "session" (V Session))
    (Json|SetField _env "body" _bo)
    (bind _envs (Json|GetJsonString _env))
    (set AskBody _envs)
    (set AskUrl (V RelayUrl))
    (else
      (bind _url (CallFunction|ProvField :Prov (V CurProv) :Field "endpoint"))
      (set AskUrl _url)))
  (CallFunction|SendReq :Tag Tag :Prov (V CurProv) :Url (V AskUrl) :Payload (V AskBody) :Try 0))
