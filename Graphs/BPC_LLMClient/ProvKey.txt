(fn ProvKey (Prov)
  (bind _kf (CallFunction|ProvField :Prov Prov :Field "key_field"))
  (set TmpStr2 _kf)
  (if (Utilities|String|IsEmpty (V TmpStr2))
    (return ""))
  (bind _h (Json|HasField (V KeyObj) (V TmpStr2)))
  (if (not _h)
    (return ""))
  (bind _v (Json|GetField (V KeyObj) (V TmpStr2)))
  (set KeyStr _v)
  (return (Utilities|String|Trim (V KeyStr))))
