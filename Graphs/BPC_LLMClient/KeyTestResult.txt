(fn KeyTestResult ()
  (if (not (V KTRun))
    (return false false ""))
  (if (> (V KTLeft) 0)
    (return false false ""))
  (set KTRun false)
  (if (== (V KTCount) 0)
    (return true false "Paste at least one key to test."))
  (if (> (len (V KTFails)) 0)
    (return true false (cat "Key test: " (Utilities|String|JoinStringArray (V KTFails) " "))))
  (return true true (cat "All " (str (V KTCount)) " keys work.")))
