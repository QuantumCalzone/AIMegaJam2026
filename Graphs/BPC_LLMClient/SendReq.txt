(fn SendReq (Tag Prov Url Payload Try)
  (add (V PendTags) Tag)
  (add (V PendProv) Prov)
  (add (V PendTry) Try)
  (add (V PendUrl) Url)
  (add (V PendBody) Payload)
  (bind _req (Game|AddComponentbyClass (Components|GetOwner) "/Game/NHOS/Core/BPC_LLMRequest.BPC_LLMRequest_C"))
  (if (streq (V Mode) "relay")
    (Class|BPCLLMRequest|Setup :self _req :InTag Tag :InUrl Url :InHeaders (V FreeHeaders) :InBody Payload :InClient self)
    (Class|BPCLLMRequest|Send :self _req)
    (return))
  (bind _key (CallFunction|ProvKey :Prov Prov))
  (set AskKey _key)
  (bind _fmt (CallFunction|ProvField :Prov Prov :Field "format"))
  (set AskFmt _fmt)
  (bind _ver (CallFunction|ProvField :Prov Prov :Field "api_version"))
  (set AskVer _ver)
  (set Headers (V EmptyHeaders))
  (Http|AddHeader (V Headers) "content-type" "application/json")
  (if (streq (V AskFmt) "anthropic")
    (Http|AddHeader (V Headers) "x-api-key" (V AskKey))
    (Http|AddHeader (V Headers) "anthropic-version" (V AskVer))
    (if (not (Utilities|String|IsEmpty (V Beta)))
      (Http|AddHeader (V Headers) "anthropic-beta" (V Beta)))
    (else
      (Http|AddHeader (V Headers) "Authorization" (cat "Bearer " (V AskKey)))))
  (Class|BPCLLMRequest|Setup :self _req :InTag Tag :InUrl Url :InHeaders (V Headers) :InBody Payload :InClient self)
  (Class|BPCLLMRequest|Send :self _req))
