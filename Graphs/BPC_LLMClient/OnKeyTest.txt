; T37: a lab's key-test note comes from errors.json (keytest_<kind>, else the plain kind), with the lab's
; name and the error's type, never the provider's text. A reply cut off or refused still proves the key works.
(fn OnKeyTest (TransportOk Response)
  (set KTLeft (- (V KTLeft) 1))
  (bind _lab (CallFunction|LabName :Prov (V CurProv)))
  (set KTName _lab)
  (if (not TransportOk)
    (bind _nl (CallFunction|ErrorLine :Kind "keytest_no_reply" :Lab (V KTName) :N 0 :Type ""))
    (add (V KTFails) _nl)
    (print (cat "NHOS: KEYTEST " (V CurProv) " fail transport"))
    (print (cat "NHOS: ERROR no_reply provider=" (V CurProv) " where=keytest"))
    (return))
  (bind (_ok _text _error) (CallFunction|ParseReply :Response Response))
  (if _ok
    (print (cat "NHOS: KEYTEST " (V CurProv) " ok"))
    (return))
  (bind _k (CallFunction|ErrorKind :Reason _error))
  (if (or (streq _k "cut_off") (streq _k "refused"))
    (print (cat "NHOS: KEYTEST " (V CurProv) " ok " _k))
    (return))
  (bind _ty (CallFunction|ErrTypeText))
  (bind _kl (CallFunction|ErrorLine :Kind (cat "keytest_" _k) :Lab (V KTName) :N 0 :Type _ty))
  (add (V KTFails) _kl)
  (print (cat "NHOS: KEYTEST " (V CurProv) " fail " _ty))
  (print (cat "NHOS: ERROR " _k " provider=" (V CurProv) " where=keytest")))
