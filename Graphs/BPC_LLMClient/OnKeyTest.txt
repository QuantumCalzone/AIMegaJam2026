(fn OnKeyTest (TransportOk Response)
  (set KTLeft (- (V KTLeft) 1))
  (set KTIdx (Utilities|Array|FindItem (V PProv) (V CurProv)))
  (set KTName (V CurProv))
  (if (>= (V KTIdx) 0)
    (set KTName (get (V PNames) (V KTIdx))))
  (if (not TransportOk)
    (add (V KTFails) (cat (V KTName) " did not answer."))
    (print (cat "NHOS: KEYTEST " (V CurProv) " fail transport"))
    (return))
  (bind (_ok _text _error) (CallFunction|ParseReply :Response Response))
  (set KTBad false)
  (if (not _ok)
    (set KTBad (not (or (Utilities|String|StartsWith :SourceString _error :InPrefix "finish_reason") (Utilities|String|StartsWith :SourceString _error :InPrefix "stop_reason")))))
  (if (V KTBad)
    (add (V KTFails) (cat (V KTName) ": the key didn't work (" (select (Utilities|String|IsEmpty (V ErrType)) "unreadable reply" (V ErrType)) ")."))
    (print (cat "NHOS: KEYTEST " (V CurProv) " fail " (select (Utilities|String|IsEmpty (V ErrType)) "unreadable reply" (V ErrType))))
    (return))
  (print (cat "NHOS: KEYTEST " (V CurProv) " ok")))
