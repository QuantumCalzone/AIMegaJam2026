(fn TestKeys ()
  (set Mode "keys")
  (set KTRun true)
  (set KTLeft 0)
  (set KTCount 0)
  (clear (V KTFails))
  (for _i (range (len (V PProv)))
    (set KTProv (get (V PProv) _i))
    (bind _k (CallFunction|ProvKey :Prov (V KTProv)))
    (if (not (Utilities|String|IsEmpty _k))
      (set KTLeft (+ (V KTLeft) 1))
      (set KTCount (+ (V KTCount) 1))
      (bind _body (CallFunction|BuildBody :System "This is a connection test. Reply with the JSON object only, with speech set to OK and every other field empty." :User "{\"ask\":\"Say OK.\"}" :Role "actor" :Prov (V KTProv)))
      (bind _url (CallFunction|ProvField :Prov (V KTProv) :Field "endpoint"))
      (CallFunction|SendReq :Tag (cat "keytest-" (V KTProv)) :Prov (V KTProv) :Url _url :Payload _body :Try 9)))
  (print (cat "NHOS: KEYTEST start keys=" (str (V KTCount)))))
