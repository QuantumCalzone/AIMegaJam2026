; T37: when a relay show goes down, the unavailable screen gets errors.json's line for the case (ErrorLine),
; with the relay's own error.message as the fallback for a relay type that has no line.
; T68: an event- tag goes to BPC_Analytics.OnSent; the down branch counts the ask and reports the error.
(fn HandleHttp (Tag SentAt TransportOk Response)
  ; T68: an analytics event's reply (BPC_Analytics.Send tags its request event-<event>) goes straight back,
  ; before anything of the ask state is touched.
  (if (Utilities|String|StartsWith :SourceString Tag :InPrefix "event-")
    (if (V HaveStats)
      (Class|BPCAnalytics|OnSent :self (V Stats) :Tag Tag :TransportOk TransportOk :Response Response))
    (return))
  (bind _now (Math|DateTime|UTCNow))
  (set Secs (Utilities|String|ToString(Float) (Math|Timespan|GetTotalSeconds (Math|DateTime|DateTime-DateTime _now SentAt))))
  (CallFunction|ClearReply)
  (set ErrType "")
  (set ErrMsg "")
  (set UpStatus 0)
  (set CurProv "")
  (set HTry 9)
  (set HUrl "")
  (set HBody "")
  (set HIdx (Utilities|Array|FindItem (V PendTags) Tag))
  (if (>= (V HIdx) 0)
    (set CurProv (get (V PendProv) (V HIdx)))
    (set HTry (get (V PendTry) (V HIdx)))
    (set HUrl (get (V PendUrl) (V HIdx)))
    (set HBody (get (V PendBody) (V HIdx)))
    (Utilities|Array|RemoveIndex (V PendTags) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendProv) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendTry) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendUrl) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendBody) (V HIdx)))
  (bind _fmt (CallFunction|ProvField :Prov (V CurProv) :Field "format"))
  (set CurFmt _fmt)
  (if (streq Tag "health")
    (CallFunction|OnHealth :TransportOk TransportOk :Response Response)
    (return))
  (if (Utilities|String|StartsWith :SourceString Tag :InPrefix "keytest-")
    (CallFunction|OnKeyTest :TransportOk TransportOk :Response Response)
    (return))
  (if (V Down)
    (print (cat "NHOS: LLM " Tag " dropped relay down"))
    (return))
  (set HErr "")
  (set HText "")
  (if (not TransportOk)
    (set HErr (cat "transport " Response))
    (else
      (bind (_ok _text _error) (CallFunction|ParseReply :Response Response))
      (CallFunction|BurnPaid :Tag Tag :ParseOk _ok :ParseErr _error)
      (if (not _ok)
        (set HErr _error)
        (else
          (set HText _text)
          (bind (_ok2 _err2) (CallFunction|ParseFields :Text (V HText)))
          (if (not _ok2)
            (set HErr _err2))))))
  (if (Utilities|String|IsEmpty (V HErr))
    (CallFunction|Deliver :Tag Tag :Ok true :Error "" :Seconds (V Secs))
    (return))
  (set HNoRetry (or (Utilities|String|Contains :SearchIn (V HErr) :Substring "limit") (or (Utilities|String|Contains :SearchIn (V HErr) :Substring "budget") (or (Utilities|String|Contains :SearchIn (V HErr) :Substring "bad_request") (Utilities|String|Contains :SearchIn (V HErr) :Substring "refusal")))))
  (if (and (< (V HTry) 1) (not (V HNoRetry)))
    (print (cat "NHOS: LLM " Tag " retry " (V HErr) " provider=" (V CurProv) " mode=" (V Mode)))
    (CallFunction|SendReq :Tag Tag :Prov (V CurProv) :Url (V HUrl) :Payload (V HBody) :Try (+ (V HTry) 1))
    (return))
  (set HDown (or (streq (V ErrType) "session_limit") (or (streq (V ErrType) "player_limit") (or (streq (V ErrType) "budget_used_up") (streq (V ErrType) "provider_unavailable")))))
  (if (and (streq (V Mode) "relay") (or (V HDown) (not TransportOk)))
    (print (cat "NHOS: LLM " Tag " fail " (V HErr) " provider=" (V CurProv) " mode=" (V Mode)))
    (CallFunction|CountCall :Tag Tag :Ok false)
    (bind _dk (CallFunction|ErrorKind :Reason (V HErr)))
    (print (cat "NHOS: ERROR " _dk " provider=" (V CurProv) " where=down"))
    (if (V HaveStats)
      (Class|BPCAnalytics|Error :self (V Stats) :Kind (cat "down_" _dk)))
    (bind _dlab (CallFunction|LabName :Prov (V CurProv)))
    (bind _dl (CallFunction|ErrorLine :Kind (cat "down_" _dk) :Lab _dlab :N 0 :Type (V ErrType)))
    (if (Utilities|String|IsEmpty _dl)
      (CallFunction|GoDown :Msg (V ErrMsg))
      (else
        (CallFunction|GoDown :Msg _dl)))
    (return))
  (CallFunction|ClearReply)
  (CallFunction|Deliver :Tag Tag :Ok false :Error (V HErr) :Seconds (V Secs)))
