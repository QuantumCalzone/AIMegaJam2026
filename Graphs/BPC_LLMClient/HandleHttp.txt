(fn HandleHttp (Tag SentAt TransportOk Response)
  (bind _now (Math|DateTime|UTCNow))
  (set Secs (Utilities|String|ToString(Float) (Math|Timespan|GetTotalSeconds (Math|DateTime|DateTime-DateTime _now SentAt))))
  (CallFunction|ClearReply)
  (set CurProv "")
  (set HTry 9)
  (set HUrl "")
  (set HBody "")
  (set HIdx (Utilities|Array|FindItem (V PendTags) Tag))
  (if (>= (V HIdx) 0)
    (set CurProv (get (V PendProv) (V HIdx)))
    (set HTry (get (V PendTry) (V HIdx)))
    (set HUrl (get (V PendUrl) (V HIdx)))
    (set HBody (get (V PendBody) (V HIdx)))
    (Utilities|Array|RemoveIndex (V PendTags) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendProv) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendTry) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendUrl) (V HIdx))
    (Utilities|Array|RemoveIndex (V PendBody) (V HIdx)))
  (bind _fmt (CallFunction|ProvField :Prov (V CurProv) :Field "format"))
  (set CurFmt _fmt)
  (set HErr "")
  (set HText "")
  (if (not TransportOk)
    (set HErr (cat "transport " Response))
    (else
      (bind (_ok _text _error) (CallFunction|ParseReply :Response Response))
      (if (not _ok)
        (set HErr _error)
        (else
          (set HText _text)
          (bind (_ok2 _err2) (CallFunction|ParseFields :Text (V HText)))
          (if (not _ok2)
            (set HErr _err2))))))
  (if (Utilities|String|IsEmpty (V HErr))
    (CallFunction|Deliver :Tag Tag :Ok true :Error "" :Seconds (V Secs))
    (return))
  (set HNoRetry (or (Utilities|String|Contains :SearchIn (V HErr) :Substring "limit") (or (Utilities|String|Contains :SearchIn (V HErr) :Substring "budget") (or (Utilities|String|Contains :SearchIn (V HErr) :Substring "bad_request") (Utilities|String|Contains :SearchIn (V HErr) :Substring "refusal")))))
  (if (and (< (V HTry) 1) (not (V HNoRetry)))
    (print (cat "NHOS: LLM " Tag " retry " (V HErr) " provider=" (V CurProv) " mode=" (V Mode)))
    (CallFunction|SendReq :Tag Tag :Prov (V CurProv) :Url (V HUrl) :Payload (V HBody) :Try (+ (V HTry) 1))
    (return))
  (CallFunction|ClearReply)
  (CallFunction|Deliver :Tag Tag :Ok false :Error (V HErr) :Seconds (V Secs)))
