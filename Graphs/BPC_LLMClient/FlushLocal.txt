; T37: each ask goes through FlushOne, which can fail it for error_test.
(fn FlushLocal ()
  (set FlushTags (V LocalTags))
  (set FlushUsers (V LocalUsers))
  (set FlushProv (V LocalProv))
  (clear (V LocalTags))
  (clear (V LocalUsers))
  (clear (V LocalProv))
  (for _i (range (len (V FlushTags)))
    (CallFunction|ClearReply)
    (set CurProv (get (V FlushProv) _i))
    (CallFunction|FlushOne :Tag (get (V FlushTags) _i) :User (get (V FlushUsers) _i))))
