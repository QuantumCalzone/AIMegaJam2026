(fn FlushLocal ()
  (set FlushTags (V LocalTags))
  (set FlushUsers (V LocalUsers))
  (set FlushProv (V LocalProv))
  (clear (V LocalTags))
  (clear (V LocalUsers))
  (clear (V LocalProv))
  (for _i (range (len (V FlushTags)))
    (CallFunction|ClearReply)
    (set CurProv (get (V FlushProv) _i))
    (CallFunction|FreeReply :User (get (V FlushUsers) _i))
    (set UsageIn "0")
    (set UsageOut "0")
    (CallFunction|Deliver :Tag (get (V FlushTags) _i) :Ok true :Error "" :Seconds "0.3")))
