(fn BuildBody (System User Role Prov)
  (bind _fmt (CallFunction|ProvField :Prov Prov :Field "format"))
  (set BFmt _fmt)
  (bind _tf (CallFunction|ProvField :Prov Prov :Field "max_tokens_field"))
  (set BTokField _tf)
  (bind _ef (CallFunction|ProvField :Prov Prov :Field "effort_field"))
  (set BEffField _ef)
  (bind _eff (CallFunction|ProvField :Prov Prov :Field "effort"))
  (set Effort _eff)
  (bind _model (CallFunction|ProvField :Prov Prov :Field "model"))
  (set Model _model)
  (bind _mt (Json|GetField (V ProvObj) "max_tokens"))
  (set TokCap _mt)
  (if (not (streq Role "actor"))
    (bind _r (Json|GetField (V Roles) Role))
    (set TmpObj _r)
    (bind _rp (Json|GetField (V TmpObj) "provider"))
    (set TmpStr _rp)
    (if (streq (V TmpStr) Prov)
      (bind _rm (Json|GetField (V TmpObj) "model"))
      (set Model _rm)
      (bind _re (Json|GetField (V TmpObj) "effort"))
      (set Effort _re)
      (bind _rt (Json|GetField (V TmpObj) "max_tokens"))
      (set TokCap _rt)))
  (bind _user (Json|LoadJsonfromString "{\"role\":\"user\"}"))
  (Json|SetField _user "content" User)
  (bind _b (Json|LoadJsonfromString "{}"))
  (Json|SetField _b "model" (V Model))
  (Json|SetField _b (V BTokField) (V TokCap))
  (if (streq (V BFmt) "openai_chat")
    (bind _sys (Json|LoadJsonfromString "{\"role\":\"system\"}"))
    (Json|SetField _sys "content" System)
    (clear (V MsgArr))
    (add (V MsgArr) _sys)
    (add (V MsgArr) _user)
    (Json|SetField _b "messages" (V MsgArr))
    (bind _js (Json|LoadJsonfromString "{\"name\":\"reply\",\"strict\":true}"))
    (Json|SetField _js "schema" (V ReplySchema))
    (bind _rf (Json|LoadJsonfromString "{\"type\":\"json_schema\"}"))
    (Json|SetField _rf "json_schema" _js)
    (Json|SetField _b "response_format" _rf)
    (Json|SetField _b (V BEffField) (V Effort))
    (bind _cs (Json|GetJsonString _b))
    (return _cs))
  (Json|SetField _b "system" System)
  (Json|SetField _b "messages" (Utilities|Array|MakeArray _user))
  (bind _fm (Json|LoadJsonfromString "{\"type\":\"json_schema\"}"))
  (Json|SetField _fm "schema" (V ReplySchema))
  (bind _oc (Json|LoadJsonfromString "{}"))
  (Json|SetField _oc "effort" (V Effort))
  (Json|SetField _oc "format" _fm)
  (Json|SetField _b "output_config" _oc)
  (if (not (Utilities|String|IsEmpty (V Fallbacks)))
    (Json|SetField _b "fallbacks" (V Fallbacks)))
  (bind _as (Json|GetJsonString _b))
  (return _as))
