(fn FreeFindPhase ()
  (for _ph (V FreePhases)
    (set FreePhase _ph)
    (bind _hw (Json|HasField (V FreePhase) "when_field"))
    (if _hw
      (bind _wf (Json|GetField (V FreePhase) "when_field"))
      (set TmpStr _wf)
      (bind _hsf (Json|HasField (V FreeState) (V TmpStr)))
      (if _hsf
        (return true)))
    (bind _hac (Json|HasField (V FreePhase) "ask_contains"))
    (if _hac
      (bind _acs (Json|GetField (V FreePhase) "ask_contains"))
      (set TmpArr3 _acs)
      (for _w (V TmpArr3)
        (if (Utilities|String|Contains :SearchIn (V FreeAsk) :Substring _w :bUseCase true)
          (return true)))))
  (for _ph2 (V FreePhases)
    (set FreePhase _ph2)
    (bind _id (Json|GetField (V FreePhase) "id"))
    (set TmpStr _id)
    (if (streq (V TmpStr) "scene_turn")
      (return true)))
  (return false))
