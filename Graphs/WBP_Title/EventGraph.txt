; WBP_Title.EventGraph (T33). The title is a paper playbill over the stage, with How to play in an ink panel beside it
; until T61's How to play screen replaces it. Every color, size and font comes from UIStyle.
; While the title is up, the stage's own widgets (header, card, input row) are hidden, and they come back when it closes.
; T68: the Play stats toggle (StatsButton) under the three ways to play. It finds BP_Show's analytics component by
; class (ShowStats), so while BP_Show has none the row stays collapsed and the button does nothing.
; T93: Play via Relay became Play. The title checks the relay when it opens (Play greyed out, "Checking the AI
; actors..."), every 60 s of real time while it is up (CheckAt;
; the widget's InDeltaTime is capped per frame, so a slow frame rate would stretch a sum of it) and on the click. Play turns on when the answer lists at least
; 3 working providers with budget left (BPC_LLMClient.OnHealth, the same floor as BP_Show.DrawCast); otherwise it
; stays greyed out with BPC_LLMClient.TitleLine under it (SetPlay). A click waits for a fresh answer (Checking):
; if the answer changed since the last check, the unavailable screen opens as before.
; T61: How to play became a screen of its own (WBP_HowToPlay, opened by HelpButton). The ink panel beside the playbill is
; now "Tonight's show": the three opening lines from stagehand.json (ShowOpening), How to play, and the Stagehand tips
; toggle (TipsButton; ShowTips says what the next show will do, from BPC_Stagehand.TipsNext). The earpiece is at Z 15,
; under the title (20), so a tip never shows over it.

  (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
  (Widget|SetRenderOpacity :self (Class|BPShow|GetStage :self _show) :InOpacity 0.0)

(event UserInterface|EventTick (MyGeometry InDeltaTime)
  (CallFunction|ShowRing :Ring (Variables|WBP_Title|GetPlayRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Title|GetPlayButton)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Title|GetFreeRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Title|GetFreeButton)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Title|GetKeysRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Title|GetKeysButton)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Title|GetStatsRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Title|GetStatsButton)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Title|GetHelpRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Title|GetHelpButton)))
  (CallFunction|ShowRing :Ring (Variables|WBP_Title|GetTipsRing) :On (Widget|HasKeyboardFocus :self (Variables|WBP_Title|GetTipsButton)))
  (if (V TitleCheck)
    (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
    (bind _cl (Class|BPShow|GetLLMClient :self _show))
    (bind (_done _ok _msg) (Class|BPCLLMClient|HealthResult :self _cl))
    (if _done
      (set TitleCheck false)
      (set CheckAt (+ (Utilities|Time|GetRealTimeSeconds) 60.0))
      (print (cat "NHOS: TITLE play " (select _ok "on" "off")))
      (if _ok
        (CallFunction|SetPlay :On true :Line "" :Quiet true)
        (else
          (CallFunction|SetPlay :On false :Line (Class|BPCLLMClient|TitleLine :self _cl) :Quiet false)))
      (if (V Checking)
        (set Checking false)
        (Widget|SetText(Text) :self (Variables|WBP_Title|GetModeNote) :InText (Utilities|Text|ToText(String) ""))
        (if _ok
          (Class|BPShow|StartfromTitle :self _show)
          (else
            (Class|BPCLLMClient|ShowUnavailable :self _cl :Msg _msg)))))
    (else
      (if (>= (Utilities|Time|GetRealTimeSeconds) (V CheckAt))
        (set TitleCheck true)
        (bind _show2 (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
        (Class|BPCLLMClient|CheckRelay :self (Class|BPShow|GetLLMClient :self _show2))))))

(event OnClicked(PlayButton)
  (print "NHOS: TITLE click relay")
  (if (not (V Checking))
    (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
    (bind _cl (Class|BPShow|GetLLMClient :self _show))
    (bind (_ok _err) (Class|BPCLLMClient|SetMode :self _cl :InMode "relay"))
    (if _ok
      (set Checking true)
      (Class|Text|SetColorandOpacity :self (Variables|WBP_Title|GetModeNote) :ColorAndOpacity (Utilities|Struct|MakeSlateColor :SpecifiedColor (Class|UIStyle|UIColor :Role "PaperText")))
      (Widget|SetText(Text) :self (Variables|WBP_Title|GetModeNote) :InText (Utilities|Text|ToText(String) "Checking the AI actors..."))
      (if (not (V TitleCheck))
        (set TitleCheck true)
        (Class|BPCLLMClient|CheckRelay :self _cl))
      (else
        (Class|Text|SetColorandOpacity :self (Variables|WBP_Title|GetModeNote) :ColorAndOpacity (Utilities|Struct|MakeSlateColor :SpecifiedColor (Class|UIStyle|UIColor :Role "Red")))
        (Widget|SetText(Text) :self (Variables|WBP_Title|GetModeNote) :InText (Utilities|Text|ToText(String) _err))))))

; T61: the Stagehand tips toggle turns the earpiece's tips on or off for the next show (Local/progress.json). It starts nothing.
(event OnClicked(TipsButton)
  (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
  (bind _sh (Class|BPShow|GetStagehand :self _show))
  (bind _on (Class|BPCStagehand|TipsNext :self _sh))
  (Class|BPCStagehand|SetTipsNext :self _sh :On (not _on))
  (print (cat "NHOS: TITLE click tips=" (select _on "off" "on")))
  (CallFunction|ShowTips))

; T68: the Play stats toggle. BPC_Analytics.SetOn sends one last analytics_off when it turns off, then nothing.
(event OnClicked(StatsButton)
  (bind _show (Actor|GetActorOfClass "/Game/NHOS/Core/BP_Show.BP_Show_C"))
  (bind _an (Actor|GetComponentbyClass :self _show :ComponentClass "/Game/NHOS/Core/BPC_Analytics.BPC_Analytics_C"))
  (Utilities|IsValid _an
    (:"Is Valid"
      (bind _on (Class|BPCAnalytics|IsOn :self _an))
      (Class|BPCAnalytics|SetOn :self _an :NewOn (not _on))
      (print (cat "NHOS: TITLE click stats=" (select _on "off" "on")))
      (CallFunction|ShowStats))
    (:"Is Not Valid"
      (print "NHOS: TITLE click stats, no analytics component"))))
