; T20: the robot's own sounds: step, grab, throw, land. Volumes live on the sound waves (SC_Effects).
(fn Snd (Key)
  (print (cat "NHOS: SOUND robot " Key))
  (if (streq Key "grab")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Grab.S_Grab")
    (return))
  (if (streq Key "throw")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Throw.S_Throw")
    (return))
  (if (streq Key "land")
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Land.S_Land")
    (return))
  (bind _i (Math|Random|RandomIntegerinRange 0 4))
  (if (== _i 0)
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Step0.S_Step0")
    (return))
  (if (== _i 1)
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Step1.S_Step1")
    (return))
  (if (== _i 2)
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Step2.S_Step2")
    (return))
  (if (== _i 3)
    (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Step3.S_Step3")
    (return))
  (Audio|PlaySound2D :Sound "/Game/NHOS/Audio/Robot/S_Step4.S_Step4"))
