(event EventBeginPlay
  (CallFunction|InitHome)
  ; T75: the emotion parts, then the emote_test switch.
  (CallFunction|FaceSetup)
  (CallFunction|EmoteTestInit)
  ; T104: the costume_test switch.
  (CallFunction|CostumeTestInit))

(event EventTick (DeltaSeconds)
  (Variables|Default|SetClock (+ (Variables|Default|GetClock) DeltaSeconds))
  (CallFunction|TickActs :Delta DeltaSeconds)
  ; T47: the headpieces go on and off with BP_Show's phase.
  (CallFunction|WatchShow)
  (bind _c (Variables|Default|GetClock))
  (if (Variables|Default|GetIsCut)
    (bind _s (- _c (Variables|Default|GetCutClock)))
    (bind _k (select (< _s 0.9) (/ _s 0.9) 1.0))
    (if (>= _s 0.9)
      (Variables|Default|SetTipRoll (Math|Interpolation|FInterpTo (Variables|Default|GetTipRoll) (Variables|Default|GetTipGoal) DeltaSeconds 3.0)))
    ; T79: OpenAI's hover sinks to the floor with the slump.
    (Transformation|SetRelativeLocation :self (Variables|Default|GetRig) :NewLocation (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z (+ (* (- 1.0 _k) (Variables|Default|GetHoverZ)) (* _k -10.0))))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetRig) :NewRotation (Math|Rotator|MakeRotator :Roll (Variables|Default|GetTipRoll) :Pitch (* _k -14.0) :Yaw 0.0))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotL) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotR) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    ; A cut OpenAI's elbows straighten too, and the hand sockets go back to their forearms' rotation.
    (Transformation|SetRelativeRotation :self (Variables|Default|GetElbowL) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetElbowR) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetHandSocket) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetHandSocketL) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    (else
      ; T87: a robot that rose for the reveal (Rise) eases from its fallen pose back to upright, SetCut's fall in
      ; reverse, at the fall's speed. Slump and TipRoll are 0 for a robot that was never cut, so nothing changes for it.
      (set Slump (Math|Interpolation|FInterpTo (V Slump) 0.0 DeltaSeconds 3.0))
      (Variables|Default|SetTipRoll (Math|Interpolation|FInterpTo (Variables|Default|GetTipRoll) 0.0 DeltaSeconds 3.0))
      (bind _talk (< _c (Variables|Default|GetTalkUntil)))
      (bind _cheer (< _c (Variables|Default|GetCheerUntil)))
      (bind _shake (< _c (Variables|Default|GetShakeUntil)))
      (bind _t (+ (* _c (Variables|Default|GetBobSpeed)) (Variables|Default|GetPhase)))
      (bind _st (Math|Trig|Sin(Radians) _t))
      (bind _w13 (Math|Trig|Sin(Radians) (* _c 13.0)))
      ; T79: the hover lifts the whole Rig.
      (bind _z (+ (Variables|Default|GetHoverZ) (+ (* _st (Variables|Default|GetBobAmp)) (+ (select _talk (* (Math|Float|Absolute(Float) _w13) 9.0) 0.0) (select _cheer (* (Math|Float|Absolute(Float) (Math|Trig|Sin(Radians) (* _c 9.0))) 26.0) 0.0)))))
      (bind _yaw (select _shake (* (Math|Trig|Sin(Radians) (* _c 45.0)) 14.0) 0.0))
      ; T75: the gesture's offsets (GestTick) ride on the hover, the bob, the talking bounce, the cheer and the shake.
      (Transformation|SetRelativeLocation :self (Variables|Default|GetRig) :NewLocation (Math|Vector|MakeVector :X (V GX) :Y 0.0 :Z (+ (* (- 1.0 (V Slump)) (+ _z (V GZ))) (* (V Slump) -10.0))))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetRig) :NewRotation (Math|Rotator|MakeRotator :Roll (+ (V GRoll) (Variables|Default|GetTipRoll)) :Pitch (+ (V GPitch) (* (V Slump) -14.0)) :Yaw (+ _yaw (V GYaw))))
      (bind _gs (+ 1.0 (V GScale)))
      (Transformation|SetRelativeScale3D :self (Variables|Default|GetRig) :NewScale3D (Math|Vector|MakeVector :X _gs :Y _gs :Z _gs))
      (bind _arm (select _cheer 150.0 (select _talk (+ 30.0 (* _w13 22.0)) (+ 6.0 (* _st 4.0)))))
      ; T79: OpenAI presses its hands together at rest; talking or cheering opens them as before. On these pivots a positive roll swings the right arm inward, so the flap and the cheer now roll each arm outward (T46 rolled them into the body).
      (Variables|Default|SetFold (Math|Interpolation|FInterpTo (Variables|Default|GetFold) (select (or _talk _cheer) 0.0 1.0) DeltaSeconds 6.0))
      (bind _fp (* (Variables|Default|GetFoldPitch) (Variables|Default|GetFold)))
      (bind _fr (* (Variables|Default|GetFoldRoll) (Variables|Default|GetFold)))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotL) :NewRotation (Math|Rotator|MakeRotator :Roll (- _arm _fr) :Pitch _fp :Yaw 0.0))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotR) :NewRotation (Math|Rotator|MakeRotator :Roll (- _fr _arm) :Pitch _fp :Yaw 0.0))
      ; OpenAI's fold also bends its elbows (FoldElbowPitch, FoldElbowYaw; 0 on the other robots): the forearms come forward and in,
      ; and the mittens press together in front of the chest. The rigid sleeve is two meshes, SM_OpenAI_UpperArm and SM_OpenAI_Forearm.
      (bind _ebp (* (Variables|Default|GetFoldElbowPitch) (Variables|Default|GetFold)))
      (bind _eby (* (Variables|Default|GetFoldElbowYaw) (Variables|Default|GetFold)))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetElbowL) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch _ebp :Yaw _eby))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetElbowR) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch _ebp :Yaw (- 0.0 _eby)))
      ; The hand sockets ride the forearms but keep the shoulder's rotation, so a held prop stays upright as before.
      (Transformation|SetWorldRotation :self (Variables|Default|GetHandSocket) :NewRotation (Transformation|GetWorldRotation :self (Variables|Default|GetArmPivotR)))
      (Transformation|SetWorldRotation :self (Variables|Default|GetHandSocketL) :NewRotation (Transformation|GetWorldRotation :self (Variables|Default|GetArmPivotL)))
      (bind _m (select _talk (Math|Float|Absolute(Float) _w13) 0.0))
      ; T75: the talking stretch rides on the emotion's mouth (MouthNow), in units of the resting mouth.
      ; With no emotion, MouthNow is MouthRest (0.02, 0.1, 0.02), so this is T46's stretch exactly.
      (bind _mw (+ (.y (V MouthNow)) (* _m (* 1.2 (.y (V MouthRest))))))
      (Transformation|SetRelativeScale3D :self (Variables|Default|GetMouth) :NewScale3D (Math|Vector|MakeVector :X (.x (V MouthNow)) :Y _mw :Z (+ (.z (V MouthNow)) (* _m (* 3.5 (.z (V MouthRest)))))))
      (CallFunction|PlaceCorners :W _mw)
      (if (Utilities|String|EqualExactly(String) (Variables|Default|GetPersonaId) "worrier")
        (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetWarnDome) :ParameterName "Glow" :ParameterValue (+ 7.0 (* (Math|Trig|Sin(Radians) (* _c 6.0)) 6.0)))))))
