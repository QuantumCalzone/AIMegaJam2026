(event EventBeginPlay
  (CallFunction|InitHome))

(event EventTick (DeltaSeconds)
  (Variables|Default|SetClock (+ (Variables|Default|GetClock) DeltaSeconds))
  (CallFunction|TickActs :Delta DeltaSeconds)
  (bind _c (Variables|Default|GetClock))
  (if (Variables|Default|GetIsCut)
    (bind _s (- _c (Variables|Default|GetCutClock)))
    (bind _k (select (< _s 0.9) (/ _s 0.9) 1.0))
    (if (>= _s 0.9)
      (Variables|Default|SetTipRoll (Math|Interpolation|FInterpTo (Variables|Default|GetTipRoll) (Variables|Default|GetTipGoal) DeltaSeconds 3.0)))
    ; T79: OpenAI's hover sinks to the floor with the slump.
    (Transformation|SetRelativeLocation :self (Variables|Default|GetRig) :NewLocation (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z (+ (* (- 1.0 _k) (Variables|Default|GetHoverZ)) (* _k -10.0))))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetRig) :NewRotation (Math|Rotator|MakeRotator :Roll (Variables|Default|GetTipRoll) :Pitch (* _k -14.0) :Yaw 0.0))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotL) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotR) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw 0.0))
    (else
      (bind _talk (< _c (Variables|Default|GetTalkUntil)))
      (bind _cheer (< _c (Variables|Default|GetCheerUntil)))
      (bind _shake (< _c (Variables|Default|GetShakeUntil)))
      (bind _t (+ (* _c (Variables|Default|GetBobSpeed)) (Variables|Default|GetPhase)))
      (bind _st (Math|Trig|Sin(Radians) _t))
      (bind _w13 (Math|Trig|Sin(Radians) (* _c 13.0)))
      ; T79: the hover lifts the whole Rig.
      (bind _z (+ (Variables|Default|GetHoverZ) (+ (* _st (Variables|Default|GetBobAmp)) (+ (select _talk (* (Math|Float|Absolute(Float) _w13) 9.0) 0.0) (select _cheer (* (Math|Float|Absolute(Float) (Math|Trig|Sin(Radians) (* _c 9.0))) 26.0) 0.0)))))
      (bind _yaw (select _shake (* (Math|Trig|Sin(Radians) (* _c 45.0)) 14.0) 0.0))
      (Transformation|SetRelativeLocation :self (Variables|Default|GetRig) :NewLocation (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z _z))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetRig) :NewRotation (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw _yaw))
      (bind _arm (select _cheer 150.0 (select _talk (+ 30.0 (* _w13 22.0)) (+ 6.0 (* _st 4.0)))))
      ; T79: OpenAI presses its hands together at rest; talking or cheering opens them as before. On these pivots a positive roll swings the right arm inward, so the flap and the cheer now roll each arm outward (T46 rolled them into the body).
      (Variables|Default|SetFold (Math|Interpolation|FInterpTo (Variables|Default|GetFold) (select (or _talk _cheer) 0.0 1.0) DeltaSeconds 6.0))
      (bind _fp (* (Variables|Default|GetFoldPitch) (Variables|Default|GetFold)))
      (bind _fr (* (Variables|Default|GetFoldRoll) (Variables|Default|GetFold)))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotL) :NewRotation (Math|Rotator|MakeRotator :Roll (- _arm _fr) :Pitch _fp :Yaw 0.0))
      (Transformation|SetRelativeRotation :self (Variables|Default|GetArmPivotR) :NewRotation (Math|Rotator|MakeRotator :Roll (- _fr _arm) :Pitch _fp :Yaw 0.0))
      (bind _m (select _talk (Math|Float|Absolute(Float) _w13) 0.0))
      (Transformation|SetRelativeScale3D :self (Variables|Default|GetMouth) :NewScale3D (Math|Vector|MakeVector :X 0.02 :Y (+ 0.1 (* _m 0.12)) :Z (+ 0.02 (* _m 0.07))))
      (if (Utilities|String|EqualExactly(String) (Variables|Default|GetPersonaId) "worrier")
        (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetWarnDome) :ParameterName "Glow" :ParameterValue (+ 7.0 (* (Math|Trig|Sin(Radians) (* _c 6.0)) 6.0)))))))
