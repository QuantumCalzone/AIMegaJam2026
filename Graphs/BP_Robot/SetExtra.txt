; T104: shows the costume's extra part (CEntry.extra: mesh, color, fit.<persona>) on Extra, or hides
; it when On is false or the costume has none. SetHeadpiece calls it last, since it sets CFit.
(fn SetExtra (On)
  (if (or (not On) (not (Json|HasField (V CEntry) "extra")))
    (Rendering|SetVisibility :self (V Extra) :bNewVisibility false :bPropagateToChildren false)
    (return))
  (bind _x (Json|GetField (V CEntry) "extra"))
  (set CExtra _x)
  (bind _xf (Json|GetField (V CExtra) "fit"))
  (set COn _xf)
  (if (not (Json|HasField (V COn) (Variables|Default|GetPersonaId)))
    (Rendering|SetVisibility :self (V Extra) :bNewVisibility false :bPropagateToChildren false)
    (return))
  (bind _pf (Json|GetField (V COn) (Variables|Default|GetPersonaId)))
  (set CFit _pf)
  (bind _xm (Json|GetField (V CExtra) "mesh"))
  (set XMesh _xm)
  (bind _xc (Json|GetField (V CExtra) "color"))
  (set XColor _xc)
  (set XS3 (CallFunction|FitScale :Def (CallFunction|FitNum :Key "scale" :Def 1.0)))
  (set XLoc (Math|Vector|MakeVector :X (CallFunction|FitNum :Key "x" :Def 0.0) :Y (CallFunction|FitNum :Key "y" :Def 0.0) :Z (CallFunction|FitNum :Key "z" :Def 0.0)))
  (set XRot (Math|Rotator|MakeRotator :Roll (CallFunction|FitNum :Key "roll" :Def 0.0) :Pitch 0.0 :Yaw (CallFunction|FitNum :Key "yaw" :Def 0.0)))
  (bind _obj (Utilities|LoadAssetBlocking :Asset (Utilities|ToSoftObjectReference (Utilities|MakeSoftObjectPath :PathString (V XMesh)))))
  (bind _mesh (Utilities|Casting|CastToStaticMesh :Object _obj)
    (:then
      (Components|StaticMesh|SetStaticMesh :self (V Extra) :NewMesh _mesh)
      (Transformation|SetRelativeScale3D :self (V Extra) :NewScale3D (V XS3))
      (Transformation|SetRelativeLocation :self (V Extra) :NewLocation (V XLoc))
      (Transformation|SetRelativeRotation :self (V Extra) :NewRotation (V XRot))
      (Rendering|Material|SetVectorParameterValueonMaterials :self (V Extra) :ParameterName "Color" :ParameterValue (Math|Conversions|ToVector(LinearColor) (CallFunction|HexColor :Hex (V XColor))))
      (Rendering|SetVisibility :self (V Extra) :bNewVisibility true :bPropagateToChildren false)
      (print (cat "NHOS: COSTUME " (cat (Variables|Default|GetPersonaId) (cat " extra " (V XColor))))))
    (:CastFailed
      (Rendering|SetVisibility :self (V Extra) :bNewVisibility false :bPropagateToChildren false)
      (print (cat "NHOS: COSTUME " (cat (Variables|Default|GetPersonaId) (cat " extra mesh missing " (V XMesh))))))))
