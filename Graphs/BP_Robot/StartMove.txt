(fn StartMove (Dest Dur)
  (bind _l (Transformation|GetActorLocation))
  (bind _d (Math|Vector|Distance2D(Vector) :V1 _l :V2 Dest))
  (bind _dz (Math|Float|Absolute(Float) (- (.z Dest) (.z _l))))
  (if (and (< _d 5.0) (< _dz 5.0))
    (return))
  (set MvFrom _l)
  (set MvTo Dest)
  (set MvDur Dur)
  (set MvT 0.0)
  (set MvHop (select (> _dz 5.0) 75.0 (select (> _d 260.0) 32.0 20.0)))
  (bind _y (CallFunction|YawTo :Target Dest))
  ; T20: a footstep as the move starts and another just past halfway.
  (CallFunction|Snd :Key "step")
  (Utilities|Time|SetTimerbyFunctionName :Object self :FunctionName "StepSnd" :Time (* Dur 0.6) :bLooping false)
  (if (> _d 10.0)
    (set FaceYaw _y)))
