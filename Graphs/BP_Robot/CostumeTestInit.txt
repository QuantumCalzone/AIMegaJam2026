; T104: the costume_test switch (Local/switches.json). On: every robot wears each costume in
; costumes.json in turn, CostTestStep (4) seconds each, all five in the same one at once, in its
; test_color. A show doesn't dress or undress them while it is on (WatchShow). BeginPlay calls it.
(fn CostumeTestInit ()
  (set CostTest false)
  (bind (_so _sok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath (cat (Utilities|Paths|GetProjectDirectory) "Local/switches.json"))))
  (if (not _sok)
    (return))
  (bind _hf (Json|HasField _so "costume_test"))
  (if (not _hf)
    (return))
  (bind _sf (Json|GetField _so "costume_test"))
  (set CostTest _sf)
  (if (not (V CostTest))
    (return))
  (CallFunction|LoadCostumes)
  (bind _ids (Json|GetFieldNames (V CostObj)))
  (set CostIds _ids)
  (if (== (len (V CostIds)) 0)
    (set CostTest false)
    (return))
  (set CostTestIdx 0)
  (set CostTestNext (+ (V Clock) 1.5))
  (print (cat "NHOS: COSTUME TEST on ids=" (str (len (V CostIds))))))
