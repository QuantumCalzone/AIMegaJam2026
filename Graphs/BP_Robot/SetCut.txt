(fn SetCut (Stamp)
  ; T75: a cut robot stops emoting, and its face is at rest before it goes dark.
  (CallFunction|EmoteStop)
  (CallFunction|DropAll)
  (clear (V QVerb))
  (clear (V QLoc))
  (clear (V QProp))
  (clear (V QBot))
  (set LookUntil 0.0)
  (set AroundUntil 0.0)
  (Variables|Default|SetIsCut true)
  (Variables|Default|SetCutClock (Variables|Default|GetClock))
  (CallFunction|SetLit :On false)
  (bind _gray (Math|Vector|MakeVector :X 0.16 :Y 0.16 :Z 0.16))
  (bind _dark (Math|Vector|MakeVector :X 0.03 :Y 0.03 :Z 0.03))
  ; T79: every seat-colored part goes gray; OpenAI's floor ring goes out.
  (for _c (Actor|GetComponentsByClass :ComponentClass "/Script/Engine.StaticMeshComponent")
    (if (Components|ComponentHasTag :self _c :Tag "seat")
      (Rendering|Material|SetVectorParameterValueonMaterials :self _c :ParameterName "Color" :ParameterValue _gray))
    (if (Components|ComponentHasTag :self _c :Tag "ring")
      (Rendering|SetVisibility :self _c :bNewVisibility false :bPropagateToChildren false)))
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetEyeL) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetEyeR) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetMouth) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetBrowWL) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetBrowWR) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetBrowAL) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetBrowAR) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Components|TextRender|SetText :self (Variables|Default|GetLabel) :Value (Utilities|Text|ToText(String) Stamp))
  (Rendering|Components|TextRender|SetTextRenderColor :self (Variables|Default|GetLabel) :Value (Math|Conversions|ToColor(LinearColor) :InLinearColor (Utilities|Struct|MakeLinearColor :R 1.0 :G 0.12 :B 0.08 :A 1.0) :InUseSRGB true))
  (Variables|Default|SetTipGoal (* 82.0 (Variables|Default|GetFallDir))))
