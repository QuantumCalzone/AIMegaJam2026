(fn SetCut (Stamp)
  (CallFunction|DropAll)
  (clear (V QVerb))
  (clear (V QLoc))
  (clear (V QProp))
  (clear (V QBot))
  (set LookUntil 0.0)
  (set AroundUntil 0.0)
  (Variables|Default|SetIsCut true)
  (Variables|Default|SetCutClock (Variables|Default|GetClock))
  (CallFunction|SetLit :On false)
  (bind _gray (Math|Vector|MakeVector :X 0.16 :Y 0.16 :Z 0.16))
  (bind _dark (Math|Vector|MakeVector :X 0.03 :Y 0.03 :Z 0.03))
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetBody) :ParameterName "Color" :ParameterValue _gray)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetHead) :ParameterName "Color" :ParameterValue _gray)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetHandL) :ParameterName "Color" :ParameterValue _gray)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetHandR) :ParameterName "Color" :ParameterValue _gray)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetEyeL) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Material|SetVectorParameterValueonMaterials :self (Variables|Default|GetEyeR) :ParameterName "Color" :ParameterValue _dark)
  (Rendering|Components|TextRender|SetText :self (Variables|Default|GetLabel) :Value (Utilities|Text|ToText(String) Stamp))
  (Rendering|Components|TextRender|SetTextRenderColor :self (Variables|Default|GetLabel) :Value (Math|Conversions|ToColor(LinearColor) :InLinearColor (Utilities|Struct|MakeLinearColor :R 1.0 :G 0.12 :B 0.08 :A 1.0) :InUseSRGB true))
  (Variables|Default|SetTipGoal (* 82.0 (Variables|Default|GetFallDir))))
