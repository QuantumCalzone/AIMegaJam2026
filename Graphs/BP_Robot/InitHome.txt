(fn InitHome ()
  (if (V HasHome)
    (return))
  (set HasHome true)
  (bind _l (Transformation|GetActorLocation))
  (set HomeLoc _l)
  (set GoalLoc _l)
  (set HomeYaw (.yaw (Transformation|GetActorRotation)))
  (set FaceYaw (V HomeYaw))
  (set MvT 1.0)
  (set MvDur 1.0)
  (set HitAt -1.0)
  (set CamLoc (+ _l (* (Transformation|GetActorForwardVector) 1000.0)))
  (Transformation|SetAbsolute :self (V Label) :bNewAbsoluteLocation false :bNewAbsoluteRotation true :bNewAbsoluteScale false)
  (bind _cam (Actor|GetActorOfClass :ActorClass "/Script/Engine.CameraActor"))
  (Utilities|IsValid _cam
    (:"Is Valid"
      (set CamLoc (Transformation|GetActorLocation :self _cam))
      (CallFunction|FaceCamera))
    (:"Is Not Valid")))
