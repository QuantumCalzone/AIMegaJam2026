(fn DoHanded (Verb Loc Prop Bot K)
  (bind _c (V Clock))
  (set BusyUntil _c)
  (if (== Prop (V NullProp))
    (return))
  (bind _s (Class|BPProp|SlotFor :self Prop :A self))
  (if (< _s 0)
    (Utilities|Array|Insert (V QVerb) Verb 0)
    (Utilities|Array|Insert (V QLoc) Loc 0)
    (Utilities|Array|Insert (V QProp) Prop 0)
    (Utilities|Array|Insert (V QBot) Bot 0)
    (CallFunction|DoGrab :Prop Prop :K K)
    (return))
  (bind _y (CallFunction|YawTo :Target Loc))
  (set FaceYaw _y)
  (set LookUntil (+ _c 1.6))
  (if (streq Verb "throw")
    (Class|BPProp|ThrowTo :self Prop :Loc Loc :Dur (* 0.9 K) :Height 170.0 :Spin 1.5)
    (set HitBot Bot)
    (set HitAt (+ _c (* 0.9 K)))
    (set TalkUntil (+ _c 0.5))
    (set BusyUntil (+ _c (* 0.95 K)))
    (return))
  (set GiveTo false)
  (if (streq Verb "give")
    (if (!= Bot (V NullBot))
      (bind _down (CallFunction|IsDown :self Bot))
      (set GiveTo (not _down))))
  (if (V GiveTo)
    (CallFunction|Receive :self Bot :Prop Prop :Dur (* 0.7 K) :H 55.0)
    (set BusyUntil (+ _c (* 0.75 K)))
    (return))
  (Class|BPProp|ThrowTo :self Prop :Loc Loc :Dur (* 0.6 K) :Height 45.0 :Spin 0.0)
  (set BusyUntil (+ _c (* 0.65 K))))
