; T75: the emotion's gesture from emotes.json: its held pose and up to two waves. "none" or a name
; not in gestures plays no gesture (the checker fails that name).
(fn ReadGesture ()
  (bind _g (Json|GetField (V EmEntry) "gesture"))
  (set EmGesture _g)
  (set GPoseX 0.0)
  (set GPoseZ 0.0)
  (set GPosePitch 0.0)
  (set GPoseRoll 0.0)
  (set GPoseYaw 0.0)
  (set GPoseScale 0.0)
  (set G1On false)
  (set G2On false)
  (bind _has (Json|HasField (V EmGest) (V EmGesture)))
  (if (or (streq (V EmGesture) "none") (not _has))
    (return))
  (bind _ge (Json|GetField (V EmGest) (V EmGesture)))
  (set GEntry _ge)
  (bind _p (Json|GetField (V GEntry) "pose"))
  (set GPoseObj _p)
  (bind _px (Json|GetField (V GPoseObj) "x"))
  (set GPoseX _px)
  (bind _pz (Json|GetField (V GPoseObj) "z"))
  (set GPoseZ _pz)
  (bind _pp (Json|GetField (V GPoseObj) "pitch"))
  (set GPosePitch _pp)
  (bind _pr (Json|GetField (V GPoseObj) "roll"))
  (set GPoseRoll _pr)
  (bind _py (Json|GetField (V GPoseObj) "yaw"))
  (set GPoseYaw _py)
  (bind _ps (Json|GetField (V GPoseObj) "scale"))
  (set GPoseScale _ps)
  (bind _w (Json|GetField (V GEntry) "waves"))
  (set GWaves _w)
  (if (> (len (V GWaves)) 0)
    (CallFunction|ReadWave :I 0)
    (set G1On (>= (V WCh) 0))
    (set G1Ch (V WCh))
    (set G1Amp (V WAmp))
    (set G1Hz (V WHz))
    (set G1Secs (V WSecs))
    (set G1Abs (V WAbs)))
  (if (> (len (V GWaves)) 1)
    (CallFunction|ReadWave :I 1)
    (set G2On (>= (V WCh) 0))
    (set G2Ch (V WCh))
    (set G2Amp (V WAmp))
    (set G2Hz (V WHz))
    (set G2Secs (V WSecs))
    (set G2Abs (V WAbs))))
