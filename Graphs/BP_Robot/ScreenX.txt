; T75: how far forward (x, on Face) the shared round head's surface is at height Z and side Y: the screen
; (Visor) or the shell (the model RShell, as an ellipsoid; T82), whichever is in front, plus BrowFwd. FaceApply puts the run-time brows there
; and turns them along it, so they never sink into the curved screen. FaceRest reads the two shapes.
(fn ScreenX (Y Z)
  (bind _vy (/ Y (.y (V ScrR))))
  (bind _vz (/ (- Z (.z (V ScrC))) (.z (V ScrR))))
  (bind _vq (Math|Float|Max(Float) 0.0 (- 1.0 (+ (* _vy _vy) (* _vz _vz)))))
  (bind _hy (/ Y (.y (V ShR))))
  (bind _hz (/ (- Z (.z (V ShC))) (.z (V ShR))))
  (bind _hq (Math|Float|Max(Float) 0.0 (- 1.0 (+ (* _hy _hy) (* _hz _hz)))))
  (return (+ (Math|Float|Max(Float) (+ (.x (V ScrC)) (* (.x (V ScrR)) (Math|Float|Sqrt _vq))) (+ (.x (V ShC)) (* (.x (V ShR)) (Math|Float|Sqrt _hq)))) (V BrowFwd))))
