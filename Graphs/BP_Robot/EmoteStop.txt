; T75: ends any emotion at once and puts the face back at rest. ResetLook, SetCut and SetPersona call it
; first; EmoteTick calls it once the face has eased back. In the construction script it only clears state.
(fn EmoteStop ()
  (set EmOn false)
  (set EmUntil 0.0)
  (set GW 0.0)
  (set GK 0.0)
  (set GX 0.0)
  (set GZ 0.0)
  (set GPitch 0.0)
  (set GRoll 0.0)
  (set GYaw 0.0)
  (set GScale 0.0)
  (if (V FaceReady)
    (Transformation|SetRelativeScale3D :self (V Rig) :NewScale3D (Math|Vector|MakeVector :X 1.0 :Y 1.0 :Z 1.0)))
  (if (not (and (V HasRest) (V FaceReady)))
    (return))
  (set FEyeW 1.0)
  (set FEyeH 1.0)
  (set FEyeDZ 0.0)
  (set FEyeDX 0.0)
  (set FEyeTilt 0.0)
  (set FBrowRL (V RestBrowRL))
  (set FBrowRR (V RestBrowRR))
  (set FBrowDZ 0.0)
  (set FBrowSkew 0.0)
  (set BrowK (select (V RestBrows) 1.0 0.0))
  (set FMouthW 1.0)
  (set FMouthH 1.0)
  (set FCurve 0.0)
  (set FMTilt 0.0)
  (set FTint 0.0)
  (CallFunction|FaceApply)
  (CallFunction|FaceRestVis))
