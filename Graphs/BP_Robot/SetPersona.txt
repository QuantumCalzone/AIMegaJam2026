(fn SetPersona (Id)
  ; T75: end any emotion on the old face first.
  (CallFunction|EmoteStop)
  (Variables|Default|SetPersonaId Id)
  (Variables|Default|SetPhase (* (Math|Random|RandomFloat) 6.28))
  (bind _tag (Utilities|String|StringToName Id))
  (for _c (Actor|GetComponentsByClass :ComponentClass "/Script/Engine.StaticMeshComponent")
    (if (Components|ComponentHasTag :self _c :Tag "acc")
      (Rendering|SetVisibility :self _c :bNewVisibility (Components|ComponentHasTag :self _c :Tag _tag) :bPropagateToChildren false)))
  (bind _v (Utilities|String|EqualExactly(String) Id "visionary"))
  (bind _w (Utilities|String|EqualExactly(String) Id "worrier"))
  (bind _o (Utilities|String|EqualExactly(String) Id "overclocker"))
  (bind _p (Utilities|String|EqualExactly(String) Id "copycat"))
  (bind _h (Utilities|String|EqualExactly(String) Id "human"))
  ; T79: OpenAI, Anthropic and SpaceXAI wear the shared round head. T80: Meta has its doll head and I wear a cardboard box.
  (bind _r (or _v (or _w _o)))
  (Variables|Default|SetBaseColor (select _v (Utilities|Struct|MakeLinearColor :R 0.392 :G 0.604 :B 1.0 :A 1.0) (select _w (Utilities|Struct|MakeLinearColor :R 0.807 :G 0.25 :B 0.068 :A 1.0) (select _o (Utilities|Struct|MakeLinearColor :R 0.687 :G 0.871 :B 0.068 :A 1.0) (select _p (Utilities|Struct|MakeLinearColor :R 0.578 :G 0.413 :B 0.006 :A 1.0) (select _h (Utilities|Struct|MakeLinearColor :R 0.05 :G 0.631 :B 0.584 :A 1.0) (Variables|Default|GetBaseColor)))))))
  ; T79: OpenAI's idle bob is slow, with its hands pressed together at rest. T103: it stands on its feet now (T95), so no robot hovers.
  (Variables|Default|SetBobSpeed (select _v 1.1 (select _w 6.0 (select _o 4.5 (select _p 0.7 2.0)))))
  (Variables|Default|SetBobAmp (select _v 3.5 (select _w 1.0 (select _o 4.0 (select _p 0.6 2.5)))))
  (Variables|Default|SetHoverZ 0.0)
  ; OpenAI's fold: the shoulders swing a little forward and in, and the elbows bend the forearms forward (85) and in (72), so the mittens meet thumbs up
  ; in front of the chest, as in its concept. Before the elbows, the straight sleeves swung 70 forward and jutted at the camera.
  (Variables|Default|SetFoldPitch (select _v 15.0 0.0))
  (Variables|Default|SetFoldRoll (select _v 8.0 0.0))
  (Variables|Default|SetFoldElbowPitch (select _v 85.0 0.0))
  (Variables|Default|SetFoldElbowYaw (select _v 72.0 0.0))
  ; T79: eyes: wide and round (OpenAI), half-closed (Anthropic), small dots (SpaceXAI). T80: round 10 cm eyes for Meta and me.
  ; T103: the new heads are bigger, so OpenAI's and Anthropic's eyes grow, and each head spaces its eyes for its own screen, glasses or slots.
  (bind _es (select _v (Math|Vector|MakeVector :X 0.1 :Y 0.14 :Z 0.14) (select _w (Math|Vector|MakeVector :X 0.1 :Y 0.12 :Z 0.09) (select _o (Math|Vector|MakeVector :X 0.1 :Y 0.09 :Z 0.09) (select _p (Math|Vector|MakeVector :X 0.1 :Y 0.11 :Z 0.11) (Math|Vector|MakeVector :X 0.1 :Y 0.1 :Z 0.1))))))
  (bind _ey (select _v 13.0 (select _w 16.0 (select _o 12.0 (select _p 16.0 12.3)))))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetEyeL) :NewLocation (Math|Vector|MakeVector :X 33.0 :Y (- _ey) :Z 135.0))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetEyeR) :NewLocation (Math|Vector|MakeVector :X 33.0 :Y _ey :Z 135.0))
  (Transformation|SetRelativeScale3D :self (Variables|Default|GetEyeL) :NewScale3D _es)
  (Transformation|SetRelativeScale3D :self (Variables|Default|GetEyeR) :NewScale3D _es)
  ; T103: each new head's face, measured on its model: the eyes 3 cm under the surface of OpenAI's and Anthropic's screens and of SpaceXAI's visor,
  ; Meta's behind its glasses (its head sits 7 cm lower, on its short neck), mine behind the box's slots. SpaceXAI's grin sits under its nose.
  (Transformation|SetRelativeLocation :self (Variables|Default|GetFace) :NewLocation (select _v (Math|Vector|MakeVector :X 3.2 :Y 0.0 :Z -16.0) (select _w (Math|Vector|MakeVector :X 1.0 :Y 0.0 :Z -11.0) (select _o (Math|Vector|MakeVector :X -9.9 :Y 0.0 :Z -8.0) (select _p (Math|Vector|MakeVector :X -11.0 :Y 0.0 :Z -31.0) (Math|Vector|MakeVector :X -11.9 :Y 0.0 :Z 3.0))))))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetMouth) :NewLocation (select _v (Math|Vector|MakeVector :X 36.7 :Y 0.0 :Z 122.0) (select _w (Math|Vector|MakeVector :X 38.0 :Y 0.0 :Z 121.0) (select _o (Math|Vector|MakeVector :X 27.4 :Y 0.0 :Z 107.0) (select _p (Math|Vector|MakeVector :X 37.2 :Y 0.0 :Z 124.5) (Math|Vector|MakeVector :X 35.9 :Y 0.0 :Z 119.0))))))
  ; T80: LED eyes on the shared head (light blue); my eyes are eyes. Only the shared head's mouth is an LED. T103: Meta's eyes are pale yellow with black pupils (T95), not LEDs.
  (Variables|Default|SetEyeColor (select _p (Utilities|Struct|MakeLinearColor :R 0.86 :G 0.78 :B 0.3 :A 1.0) (select _h (Utilities|Struct|MakeLinearColor :R 0.78 :G 0.76 :B 0.7 :A 1.0) (Utilities|Struct|MakeLinearColor :R 0.55 :G 0.85 :B 1.0 :A 1.0))))
  (Variables|Default|SetLedEyes (not (or _h _p)))
  (Variables|Default|SetLedMouth _r)
  ; T109: each shoulder pivot sits where the arm enters its body (T109-build.md): OpenAI's sleeves come out of the sloping shoulders, Anthropic's ball sinks half into the egg,
  ; SpaceXAI's spring goes into the body's side socket, Meta's stub goes up under the shirt's shoulder, my sleeves hug the box from its top corners. Each arm's tilt is on its component (xArmXL/xArmXR).
  (bind _sy (select _v 21.0 (select _w 31.0 (select _o 27.5 (select _p 22.0 27.0)))))
  (bind _sz (select _v 69.0 (select _w 75.5 (select _o 80.0 (select _p 63.0 98.0)))))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetArmPivotL) :NewLocation (Math|Vector|MakeVector :X 0.0 :Y (- _sy) :Z _sz))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetArmPivotR) :NewLocation (Math|Vector|MakeVector :X 0.0 :Y _sy :Z _sz))
  ; T80: each body's head socket, placed and scaled so T47's headpieces fit every head (T80's Findings has the final numbers), and its back socket. T103: Meta's head top is at 151 now.
  ; HandSocket rides on the right arm pivot, at the centre of each arm's hand; T109: HandSocketL is its mirror on the left pivot (Receive's slot 1).
  ; Each hand socket now sits on its elbow (ElbowR, ElbowL), which is at the pivot for every robot but OpenAI. OpenAI's elbow is mid-sleeve, so its sockets are measured from there.
  (Transformation|SetRelativeLocation :self (Variables|Default|GetElbowL) :NewLocation (select _v (Math|Vector|MakeVector :X 0.03 :Y -3.53 :Z -15.66) (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z 0.0)))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetElbowR) :NewLocation (select _v (Math|Vector|MakeVector :X 0.03 :Y 3.53 :Z -15.66) (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z 0.0)))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetHandSocket) :NewLocation (select _v (Math|Vector|MakeVector :X 1.29 :Y 6.49 :Z -22.84)(select _w (Math|Vector|MakeVector :X 0.7 :Y 16.1 :Z -42.9) (select _o (Math|Vector|MakeVector :X -0.1 :Y 23.2 :Z -35.6) (select _p (Math|Vector|MakeVector :X 3.9 :Y 5.4 :Z -32.1) (Math|Vector|MakeVector :X 7.8 :Y -0.7 :Z -60.3))))))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetHandSocketL) :NewLocation (select _v (Math|Vector|MakeVector :X 1.29 :Y -6.49 :Z -22.84)(select _w (Math|Vector|MakeVector :X 0.7 :Y -16.1 :Z -42.9) (select _o (Math|Vector|MakeVector :X -0.1 :Y -23.2 :Z -35.6) (select _p (Math|Vector|MakeVector :X 3.9 :Y -5.4 :Z -32.1) (Math|Vector|MakeVector :X 7.8 :Y 0.7 :Z -60.3))))))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetHeadSocket) :NewLocation (select _h (Math|Vector|MakeVector :X -2.0 :Y 0.0 :Z 162.0) (select _p (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z 151.0) (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z 158.0))))
  (bind _hs (select _p 0.9 (select _h 1.1 1.0)))
  (Transformation|SetRelativeScale3D :self (Variables|Default|GetHeadSocket) :NewScale3D (Math|Vector|MakeVector :X _hs :Y _hs :Z _hs))
  (Transformation|SetRelativeLocation :self (Variables|Default|GetBackSocket) :NewLocation (select _v (Math|Vector|MakeVector :X -24.0 :Y 0.0 :Z 55.0) (select _w (Math|Vector|MakeVector :X -37.0 :Y 0.0 :Z 52.0) (select _o (Math|Vector|MakeVector :X -31.0 :Y 0.0 :Z 62.0) (select _p (Math|Vector|MakeVector :X -17.0 :Y 0.0 :Z 55.0) (Math|Vector|MakeVector :X -28.0 :Y 0.0 :Z 75.0))))))
  ; T75: read the new persona's resting face, after the eyes are set.
  (CallFunction|FaceRest)
  (CallFunction|ResetLook))
