; T47: put on costume Id (an id in costumes.json) in Color (sRGB hex), or take it off ("").
; It hangs on HeadSocket; the stalk, the toppers and the label rise so the topper pokes out of it.
; WatchShow and DressFromShow call it: "" at Scene Prep for every robot still standing, the seat's
; costume and color when the scene starts. A cut robot keeps its costume as it slumps.
; T104: each costume has a mesh per version and a fit per persona (fit.<persona id>): the version
; (mesh), scale or scale3, x, y, z, yaw, roll and top, all in this robot's HeadSocket space.
(fn SetHeadpiece (Id Color)
  (CallFunction|InitLift)
  (CallFunction|LoadCostumes)
  (CallFunction|MakeHat)
  (CallFunction|MakeExtra)
  (bind _has (and (not (Utilities|String|IsEmpty Id)) (Json|HasField (V CostObj) Id)))
  (CallFunction|ShowGlasses :On (not (and _has (streq Id "dark_shades"))))
  (if (not _has)
    (Rendering|SetVisibility :self (V Hat) :bNewVisibility false :bPropagateToChildren false)
    (CallFunction|SetExtra :On false)
    (CallFunction|LiftTop :Lift 0.0)
    (print (cat "NHOS: COSTUME " (cat (Variables|Default|GetPersonaId) (select (Utilities|String|IsEmpty Id) " off" (cat " unknown " Id)))))
    (return))
  (bind _e (Json|GetField (V CostObj) Id))
  (set CEntry _e)
  ; T104: this robot's fit (fit.<persona>) and the mesh of the version it names.
  (bind _fs (Json|GetField (V CEntry) "fit"))
  (set COn _fs)
  (if (not (Json|HasField (V COn) (Variables|Default|GetPersonaId)))
    (print (cat "NHOS: COSTUME " (cat (Variables|Default|GetPersonaId) (cat " no fit for " Id))))
    (Rendering|SetVisibility :self (V Hat) :bNewVisibility false :bPropagateToChildren false)
    (CallFunction|SetExtra :On false)
    (CallFunction|LiftTop :Lift 0.0)
    (CallFunction|ShowGlasses :On true)
    (return))
  (bind _pf (Json|GetField (V COn) (Variables|Default|GetPersonaId)))
  (set CFit _pf)
  (bind _v (Json|GetField (V CFit) "mesh"))
  (set HVer _v)
  (bind _ms (Json|GetField (V CEntry) "meshes"))
  (set CMeshes _ms)
  (bind _m (Json|GetField (V CMeshes) (V HVer)))
  (set HMesh _m)
  (set HScale (CallFunction|FitNum :Key "scale" :Def 1.0))
  (set HX (CallFunction|FitNum :Key "x" :Def 0.0))
  (set HY (CallFunction|FitNum :Key "y" :Def 0.0))
  (set HZ (CallFunction|FitNum :Key "z" :Def 0.0))
  (set HYaw (CallFunction|FitNum :Key "yaw" :Def 0.0))
  (set HRoll (CallFunction|FitNum :Key "roll" :Def 0.0))
  (set HTop (CallFunction|FitNum :Key "top" :Def 0.0))
  (set HS3 (CallFunction|FitScale :Def (V HScale)))
  (bind _obj (Utilities|LoadAssetBlocking :Asset (Utilities|ToSoftObjectReference (Utilities|MakeSoftObjectPath :PathString (V HMesh)))))
  (bind _mesh (Utilities|Casting|CastToStaticMesh :Object _obj)
    (:then
      (Components|StaticMesh|SetStaticMesh :self (V Hat) :NewMesh _mesh)
      (Transformation|SetRelativeScale3D :self (V Hat) :NewScale3D (V HS3))
      (Transformation|SetRelativeLocation :self (V Hat) :NewLocation (Math|Vector|MakeVector :X (V HX) :Y (V HY) :Z (V HZ)))
      (Transformation|SetRelativeRotation :self (V Hat) :NewRotation (Math|Rotator|MakeRotator :Roll (V HRoll) :Pitch 0.0 :Yaw (V HYaw)))
      (Rendering|Material|SetVectorParameterValueonMaterials :self (V Hat) :ParameterName "Color" :ParameterValue (Math|Conversions|ToVector(LinearColor) (CallFunction|HexColor :Hex Color)))
      (Rendering|SetVisibility :self (V Hat) :bNewVisibility true :bPropagateToChildren false)
      ; T47: the stalk's base ends 4 cm under the hat's top, on any head: the hat's top in Rig is
      ; socket z + socket scale x top, and the stalk's base is Nub's bare z - 12 (it is 24 cm tall).
      ; On the round head this is top - 2. A top of 0 (the tiara, the shades) leaves everything put.
      (bind _sl (Class|SceneComponent|GetRelativeLocation :self (V HeadSocket)))
      (bind _ss (Class|SceneComponent|GetRelativeScale3D :self (V HeadSocket)))
      (bind _nb (get (V LiftBase) 0))
      (CallFunction|LiftTop :Lift (select (> (V HTop) 0.0) (Math|Float|Max(Float) 0.0 (+ (+ (.z _sl) (* (.z _ss) (V HTop))) (- 8.0 (.z _nb)))) 0.0))
      (print (cat "NHOS: COSTUME " (cat (Variables|Default|GetPersonaId) (cat " " (cat Id (cat " " (cat Color (cat " " (V HVer)))))))))
      ; T104: the second part (the dwarf's beard), last, since it reuses CFit.
      (CallFunction|SetExtra :On true))
    (:CastFailed
      (print (cat "NHOS: COSTUME " (cat (Variables|Default|GetPersonaId) (cat " mesh missing " (V HMesh)))))
      (Rendering|SetVisibility :self (V Hat) :bNewVisibility false :bPropagateToChildren false)
      (CallFunction|SetExtra :On false)
      (CallFunction|LiftTop :Lift 0.0)
      (CallFunction|ShowGlasses :On true))))
