; T75: reads this robot's resting face from its parts, so neutral is its face value for value.
; SetPersona calls it after it sets the persona's face; Emote calls it if the persona changed since.
; It only reads in the construction script; with FaceReady it also moves the new parts onto this head.
(fn FaceRest ()
  (CallFunction|FaceParts)
  ; The shared head's screen (Visor) and shell as ellipsoids on Face, for ScreenX. T103: each head is its own model
  ; now, with its screen flush in its shell, so the Visor sits inside the head and the shell is an ellipsoid fitted to
  ; the front of each model (centre z and radii in the Rig: OpenAI 116 and 41 x 46 x 44, Anthropic 119 and 40 x 44 x 42,
  ; SpaceXAI 125 and 28 x 34 x 34 without its nose, Meta 112 and 28 x 38 x 46).
  (bind _pw (streq (V PersonaId) "worrier"))
  (bind _po (streq (V PersonaId) "overclocker"))
  (bind _pm (streq (V PersonaId) "copycat"))
  (bind _fl (Class|SceneComponent|GetRelativeLocation :self (V Face)))
  (bind _vl (Class|SceneComponent|GetRelativeLocation :self (V Visor)))
  (bind _vs (Class|SceneComponent|GetRelativeScale3D :self (V Visor)))
  (set ScrC (Math|Vector|MakeVector :X (- (.x _vl) (.x _fl)) :Y 0.0 :Z (- (.z _vl) (.z _fl))))
  (set ScrR (Math|Vector|MakeVector :X (* (.x _vs) 50.0) :Y (* (.y _vs) 50.0) :Z (* (.z _vs) 50.0)))
  (set ShC (Math|Vector|MakeVector :X (- 0.0 (.x _fl)) :Y 0.0 :Z (- (select _pw 119.0 (select _po 125.0 (select _pm 112.0 116.0))) (.z _fl))))
  (set ShR (select _pw (Math|Vector|MakeVector :X 40.0 :Y 44.0 :Z 42.0) (select _po (Math|Vector|MakeVector :X 28.0 :Y 34.0 :Z 34.0) (select _pm (Math|Vector|MakeVector :X 28.0 :Y 38.0 :Z 46.0) (Math|Vector|MakeVector :X 41.0 :Y 46.0 :Z 44.0)))))
  (set RestEyeLocL (Class|SceneComponent|GetRelativeLocation :self (V PEyeL)))
  (set RestEyeRotL (Class|SceneComponent|GetRelativeRotation :self (V PEyeL)))
  (set RestEyeScaleL (Class|SceneComponent|GetRelativeScale3D :self (V PEyeL)))
  (set RestEyeLocR (Class|SceneComponent|GetRelativeLocation :self (V PEyeR)))
  (set RestEyeRotR (Class|SceneComponent|GetRelativeRotation :self (V PEyeR)))
  (set RestEyeScaleR (Class|SceneComponent|GetRelativeScale3D :self (V PEyeR)))
  (set SideEyeL (select (< (.y (V RestEyeLocL)) 0.0) -1.0 1.0))
  (set SideEyeR (select (< (.y (V RestEyeLocR)) 0.0) -1.0 1.0))
  (set RestMouthLoc (+ (Class|SceneComponent|GetRelativeLocation :self (V PMouth)) (Math|Vector|MakeVector :X (V MouthDX) :Y 0.0 :Z (V MouthDZ))))
  (set RestMouthRot (Class|SceneComponent|GetRelativeRotation :self (V PMouth)))
  (bind _mv (Rendering|IsVisible :self (V PMouth)))
  (set MouthVis _mv)
  (bind _bl (Class|SceneComponent|GetRelativeLocation :self (V PBrowL)))
  (bind _br (Class|SceneComponent|GetRelativeLocation :self (V PBrowR)))
  (set RestBrowScaleL (Class|SceneComponent|GetRelativeScale3D :self (V PBrowL)))
  (set RestBrowScaleR (Class|SceneComponent|GetRelativeScale3D :self (V PBrowR)))
  (if (V RestBrows)
    (set RestBrowLocL _bl)
    (set RestBrowLocR _br)
    (set RestBrowRotL (Class|SceneComponent|GetRelativeRotation :self (V PBrowL)))
    (set RestBrowRotR (Class|SceneComponent|GetRelativeRotation :self (V PBrowR)))
    (else
      (set RestBrowLocL (Math|Vector|MakeVector :X (select (V BrowFix) (V BrowFixX) (.x _bl)) :Y (.y (V RestEyeLocL)) :Z (select (V BrowFix) (V BrowFixZ) (+ (.z (V RestEyeLocL)) (+ (* (.z (V RestEyeScaleL)) 50.0) (V BrowGap))))))
      (set RestBrowLocR (Math|Vector|MakeVector :X (select (V BrowFix) (V BrowFixX) (.x _br)) :Y (.y (V RestEyeLocR)) :Z (select (V BrowFix) (V BrowFixZ) (+ (.z (V RestEyeLocR)) (+ (* (.z (V RestEyeScaleR)) 50.0) (V BrowGap))))))
      (set RestBrowRotL (Math|Rotator|MakeRotator :Roll 0.0 :Pitch (.pitch (V RestEyeRotL)) :Yaw (.yaw (V RestEyeRotL))))
      (set RestBrowRotR (Math|Rotator|MakeRotator :Roll 0.0 :Pitch (.pitch (V RestEyeRotR)) :Yaw (.yaw (V RestEyeRotR))))))
  (set RestBrowRL (.roll (V RestBrowRotL)))
  (set RestBrowRR (.roll (V RestBrowRotR)))
  (set SideBrowL (select (< (.y (V RestBrowLocL)) 0.0) -1.0 1.0))
  (set SideBrowR (select (< (.y (V RestBrowLocR)) 0.0) -1.0 1.0))
  (set FBrowRL (V RestBrowRL))
  (set FBrowRR (V RestBrowRR))
  (set BrowK (select (V RestBrows) 1.0 0.0))
  (set MouthNow (V MouthRest))
  (set TintDirty false)
  (set RestFor (V PersonaId))
  (set HasRest true)
  (if (V FaceReady)
    (bind _bp (Transformation|GetAttachParent :self (V PBrowL)))
    (bind _bq (Transformation|GetAttachParent :self (V PBrowR)))
    (bind _mp (Transformation|GetAttachParent :self (V PMouth)))
    (CallFunction|AttachPart :P (V EmBrowL) :To _bp)
    (CallFunction|AttachPart :P (V EmBrowR) :To _bq)
    (CallFunction|AttachPart :P (V MouthCL) :To _mp)
    (CallFunction|AttachPart :P (V MouthCR) :To _mp)
    (bind _bm (Class|StaticMeshComponent|GetStaticMesh :self (V PBrowL)))
    (bind _mm (Class|StaticMeshComponent|GetStaticMesh :self (V PMouth)))
    (bind _lg (select (V LitOn) 10.0 3.0))
    (CallFunction|PartLook :P (V EmBrowL) :Mat (V BrowMat) :Mesh _bm :Col (V LedColor) :SetCol (V BrowLed) :Glow (select (V BrowLed) _lg 0.0))
    (CallFunction|PartLook :P (V EmBrowR) :Mat (V BrowMat) :Mesh _bm :Col (V LedColor) :SetCol (V BrowLed) :Glow (select (V BrowLed) _lg 0.0))
    (CallFunction|PartLook :P (V MouthCL) :Mat (V LedMat) :Mesh _mm :Col (V CornerCol) :SetCol true :Glow (select (V CornerLed) _lg 0.0))
    (CallFunction|PartLook :P (V MouthCR) :Mat (V LedMat) :Mesh _mm :Col (V CornerCol) :SetCol true :Glow (select (V CornerLed) _lg 0.0))))
