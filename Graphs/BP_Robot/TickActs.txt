(fn TickActs (Delta)
  (if (not (V HasHome))
    (CallFunction|InitHome))
  (bind _c (V Clock))
  (if (< (V MvT) 1.0)
    (set MvT (Math|Float|Clamp(Float) :Value (+ (V MvT) (/ Delta (V MvDur))) :Min 0.0 :Max 1.0))
    (bind _a (V MvT))
    (bind _e (* _a (* _a (- 3.0 (* 2.0 _a)))))
    (bind _hop (* (V MvHop) (Math|Trig|Sin(Radians) (* _a 3.14159))))
    (Transformation|SetActorLocation :NewLocation (+ (Math|Vector|Lerp(Vector) :A (V MvFrom) :B (V MvTo) :Alpha _e) (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z _hop)))
    (if (>= (V MvT) 1.0)
      (CallFunction|FaceCamera)))
  (if (and (>= (V HitAt) 0.0) (>= _c (V HitAt)))
    (set HitAt -1.0)
    (if (!= (V HitBot) (V NullBot))
      (CallFunction|Shake :self (V HitBot))))
  (bind _front (CallFunction|FrontYaw))
  (bind _sweep (+ _front (* 65.0 (Math|Trig|Sin(Radians) (* (- _c (V AroundStart)) 5.0)))))
  (bind _goal (select (V IsCut) _front (select (< (V MvT) 1.0) (V FaceYaw) (select (< _c (V AroundUntil)) _sweep (select (< _c (V LookUntil)) (V FaceYaw) _front)))))
  (bind _cur (Transformation|GetActorRotation))
  (Transformation|SetActorRotation :NewRotation (Math|Interpolation|RInterpTo :Current _cur :Target (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw _goal) :DeltaTime Delta :InterpSpeed 8.0))
  (if (and (>= _c (V BusyUntil)) (> (len (V QVerb)) 0))
    (set CurVerb (get (V QVerb) 0))
    (set CurLoc (get (V QLoc) 0))
    (set CurProp (get (V QProp) 0))
    (set CurBot (get (V QBot) 0))
    (Utilities|Array|RemoveIndex (V QVerb) 0)
    (Utilities|Array|RemoveIndex (V QLoc) 0)
    (Utilities|Array|RemoveIndex (V QProp) 0)
    (Utilities|Array|RemoveIndex (V QBot) 0)
    (CallFunction|StartAct :Verb (V CurVerb) :Loc (V CurLoc) :Prop (V CurProp) :Bot (V CurBot)))
  ; T75: the emotion's face and gesture, and the emote_test cycle.
  (CallFunction|EmoteTick :Delta Delta))
