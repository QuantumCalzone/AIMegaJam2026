; T47: puts on this robot's seat's headpiece and color. It reads, from BP_Show, its seat (its index in
; Robots), that seat's dealt role (SeatRole, filled at Scene Prep) and the card in play (CardId in
; CardArr, as BP_StageDirector.CardFromShow does), then finds the card's seat with that role.
; A cut robot, or a seat with no role this scene, is left as it is.
(fn DressFromShow ()
  (if (V IsCut)
    (return))
  (bind _seat (Utilities|Array|FindItem (Class|BPShow|GetRobots :self (V ShowRef)) self))
  (bind _roles (Class|BPShow|GetSeatRole :self (V ShowRef)))
  (if (or (< _seat 0) (>= _seat (len _roles)))
    (return))
  (set DRole (get _roles _seat))
  (if (Utilities|String|IsEmpty (V DRole))
    (return))
  (set DCard (Class|BPShow|GetCardId :self (V ShowRef)))
  (set DHat "")
  (set DCol "")
  (for _c (Class|BPShow|GetCardArr :self (V ShowRef))
    (bind _id (Json|GetField _c "id"))
    (set DStr _id)
    (if (streq (V DStr) (V DCard))
      (bind _ss (Json|GetField _c "seats"))
      (set DSeats _ss)
      (for _st (V DSeats)
        (bind _r (Json|GetField _st "role"))
        (set DStr _r)
        (if (and (streq (V DStr) (V DRole)) (Json|HasField _st "headpiece"))
          (bind _h (Json|GetField _st "headpiece"))
          (set DHat _h)
          (bind _k (Json|GetField _st "color"))
          (set DCol _k)))))
  (if (Utilities|String|IsEmpty (V DHat))
    (print (cat "NHOS: COSTUME " (cat (Variables|Default|GetPersonaId) (cat " no headpiece for " (V DRole)))))
    (return))
  (CallFunction|SetHeadpiece :Id (V DHat) :Color (V DCol)))
