(fn DoGrab (Prop K)
  (bind _c (V Clock))
  (set BusyUntil (+ _c (* 0.15 K)))
  (if (== Prop (V NullProp))
    (return))
  (bind _s (Class|BPProp|SlotFor :self Prop :A self))
  (if (>= _s 0)
    (return))
  (bind _y (CallFunction|YawTo :Target (Transformation|GetActorLocation :self Prop)))
  (set FaceYaw _y)
  (set LookUntil (+ _c 0.9))
  (CallFunction|Receive :Prop Prop :Dur (* 0.6 K) :H 70.0)
  (set BusyUntil (+ _c (* 0.65 K))))
