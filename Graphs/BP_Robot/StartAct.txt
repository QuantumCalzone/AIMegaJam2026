(fn StartAct (Verb Loc Prop Bot)
  (if (and (V IsCut) (not (streq Verb "home")))
    (return))
  (bind _c (V Clock))
  (bind _k (/ 1.0 (+ 1.0 (* 0.5 (len (V QVerb))))))
  (if (streq Verb "move")
    (CallFunction|StartMove :Dest Loc :Dur _k)
    (set BusyUntil (+ _c (+ _k 0.1)))
    (return))
  (if (streq Verb "home")
    (CallFunction|ToTable)
    (CallFunction|StartMove :Dest Loc :Dur 1.0)
    (set LookUntil 0.0)
    (set AroundUntil 0.0)
    (set BusyUntil (+ _c 1.1))
    (return))
  (if (streq Verb "look")
    (bind _ly (CallFunction|YawTo :Target Loc))
    (set FaceYaw _ly)
    (set LookUntil (+ _c 2.5))
    (set AroundUntil 0.0)
    (set BusyUntil (+ _c (* 0.6 _k)))
    (return))
  (if (streq Verb "around")
    (set AroundStart _c)
    (set AroundUntil (+ _c (* 1.3 _k)))
    (set LookUntil 0.0)
    (set BusyUntil (+ _c (* 1.3 _k)))
    (return))
  (if (streq Verb "grab")
    (CallFunction|DoGrab :Prop Prop :K _k)
    (return))
  (CallFunction|DoHanded :Verb Verb :Loc Loc :Prop Prop :Bot Bot :K _k))
