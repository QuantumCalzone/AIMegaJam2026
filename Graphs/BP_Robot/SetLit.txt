(fn SetLit (On)
  (Rendering|SetVisibility :self (Variables|Default|GetSpot) :bNewVisibility On :bPropagateToChildren false)
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetBody) :ParameterName "Glow" :ParameterValue (select On 0.15 0.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetHead) :ParameterName "Glow" :ParameterValue (select On 0.15 0.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetEyeL) :ParameterName "Glow" :ParameterValue (select On 10.0 3.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetEyeR) :ParameterName "Glow" :ParameterValue (select On 10.0 3.0))
  (if On
    (Variables|Default|SetTalkUntil (+ (Variables|Default|GetClock) 2.6))))
