(fn SetLit (On)
  (Rendering|SetVisibility :self (Variables|Default|GetSpot) :bNewVisibility On :bPropagateToChildren false)
  ; T80: only LEDs glow: the shared head's eyes and mouth, and Meta's eyes.
  (bind _g (select On 10.0 3.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetEyeL) :ParameterName "Glow" :ParameterValue (select (Variables|Default|GetLedEyes) _g 0.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetEyeR) :ParameterName "Glow" :ParameterValue (select (Variables|Default|GetLedEyes) _g 0.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetMouth) :ParameterName "Glow" :ParameterValue (select (Variables|Default|GetLedMouth) _g 0.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetBrowWL) :ParameterName "Glow" :ParameterValue (select On 10.0 3.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetBrowWR) :ParameterName "Glow" :ParameterValue (select On 10.0 3.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetBrowAL) :ParameterName "Glow" :ParameterValue (select On 10.0 3.0))
  (Rendering|Material|SetScalarParameterValueonMaterials :self (Variables|Default|GetBrowAR) :ParameterName "Glow" :ParameterValue (select On 10.0 3.0))
  ; T75: the new brows and corner bars glow like the eyes where they are LEDs. LitOn lets PartLook match it later.
  (set LitOn On)
  (if (V FaceReady)
    (Rendering|Material|SetScalarParameterValueonMaterials :self (V EmBrowL) :ParameterName "Glow" :ParameterValue (select (V BrowLed) _g 0.0))
    (Rendering|Material|SetScalarParameterValueonMaterials :self (V EmBrowR) :ParameterName "Glow" :ParameterValue (select (V BrowLed) _g 0.0))
    (Rendering|Material|SetScalarParameterValueonMaterials :self (V MouthCL) :ParameterName "Glow" :ParameterValue (select (V CornerLed) _g 0.0))
    (Rendering|Material|SetScalarParameterValueonMaterials :self (V MouthCR) :ParameterName "Glow" :ParameterValue (select (V CornerLed) _g 0.0)))
  (if On
    (Variables|Default|SetTalkUntil (+ (Variables|Default|GetClock) 2.6))))
