(fn BotCands ()
  (clear (V Cands))
  (bind _g (Class|BPRobot|GetGoal :self (V TgtBot)))
  (bind _x (.x _g))
  (bind _y (.y _g))
  (add (V Cands) (Math|Vector|MakeVector :X _x :Y (- _y 115.0) :Z 0.0))
  (add (V Cands) (Math|Vector|MakeVector :X _x :Y (+ _y 115.0) :Z 0.0))
  (add (V Cands) (Math|Vector|MakeVector :X (- _x 150.0) :Y _y :Z 0.0))
  (add (V Cands) (Math|Vector|MakeVector :X _x :Y (- _y 230.0) :Z 0.0))
  (add (V Cands) (Math|Vector|MakeVector :X _x :Y (+ _y 230.0) :Z 0.0))
  (add (V Cands) (Math|Vector|MakeVector :X (+ _x 150.0) :Y _y :Z 0.0)))
