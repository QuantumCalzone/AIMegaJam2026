(fn FindMark (Name)
  (if (Utilities|String|IsEmpty Name)
    (return false))
  (for _m (V Marks)
    (bind _n (Class|BPPlaceMark|GetPlace :self _m))
    (if (or (Utilities|String|Equal,CaseInsensitive(String) :A _n :B Name) (Utilities|String|EndsWith :SourceString _n :InSuffix (cat " " Name)))
      (set TgtMark _m)
      (return true)))
  (return false))
