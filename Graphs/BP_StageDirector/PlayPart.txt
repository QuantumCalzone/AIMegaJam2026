(fn PlayPart (Part)
  (bind _p (Utilities|String|TrimTrailing (Utilities|String|Trim Part)))
  (set PVerb "")
  (set PTgt "")
  (set PItem "")
  (if (Utilities|String|StartsWith :SourceString _p :InPrefix "moves to ")
    (set PVerb "move")
    (set PTgt (Utilities|String|RightChop :SourceString _p :Count 9)))
  (if (Utilities|String|Equal,CaseInsensitive(String) :A _p :B "looks around")
    (set PVerb "around"))
  (if (Utilities|String|StartsWith :SourceString _p :InPrefix "looks at ")
    (set PVerb "look")
    (set PTgt (Utilities|String|RightChop :SourceString _p :Count 9)))
  (if (Utilities|String|StartsWith :SourceString _p :InPrefix "votes for ")
    (set PVerb "vote")
    (set PTgt (Utilities|String|RightChop :SourceString _p :Count 10)))
  (if (Utilities|String|StartsWith :SourceString _p :InPrefix "grabs the ")
    (set PVerb "grab")
    (set PItem (Utilities|String|RightChop :SourceString _p :Count 10)))
  (if (Utilities|String|StartsWith :SourceString _p :InPrefix "throws the ")
    (set PVerb "throw")
    (bind (_tl _tr _tok) (Utilities|String|Split :SourceString (Utilities|String|RightChop :SourceString _p :Count 11) :InStr " at " :SearchDir "FromEnd"))
    (set PItem _tl)
    (set PTgt _tr))
  (if (Utilities|String|StartsWith :SourceString _p :InPrefix "gives the ")
    (set PVerb "give")
    (bind (_gl _gr _gok) (Utilities|String|Split :SourceString (Utilities|String|RightChop :SourceString _p :Count 10) :InStr " to " :SearchDir "FromEnd"))
    (set PItem _gl)
    (set PTgt _gr))
  (if (Utilities|String|IsEmpty (V PVerb))
    (return))
  (bind _ct (CallFunction|Clean :S (V PTgt)))
  (set PTgt _ct)
  (bind _ci (CallFunction|Clean :S (V PItem)))
  (set PItem _ci)
  (CallFunction|DoVerb :Part _p))
