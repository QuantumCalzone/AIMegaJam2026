(fn PickSpot ()
  (for _c (V Cands)
    (bind _ok (CallFunction|InStage :P _c))
    (if _ok
      (set Clash false)
      (for _b (V AllBots)
        (if (and (!= _b (V NullBot)) (!= _b (V CurBot)))
          (bind _g (Class|BPRobot|GetGoal :self _b))
          (if (and (< (Math|Float|Absolute(Float) (- (.y _g) (.y _c))) 95.0) (< (Math|Float|Absolute(Float) (- (.x _g) (.x _c))) 140.0))
            (set Clash true))))
      (if (not (V Clash))
        (set Pick _c)
        (return true))))
  (return false))
