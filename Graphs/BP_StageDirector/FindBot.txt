(fn FindBot (Name)
  (if (Utilities|String|IsEmpty Name)
    (return false))
  (for _i (range (len (V Names)))
    (if (and (Utilities|String|Equal,CaseInsensitive(String) :A (get (V Names) _i) :B Name) (< _i (len (V Bots))))
      (bind _b (get (V Bots) _i))
      (if (and (!= _b (V NullBot)) (!= _b (V CurBot)))
        (set TgtBot _b)
        (return true))))
  (return false))
