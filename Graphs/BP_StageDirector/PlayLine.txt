(fn PlayLine (Seat Entry Robots SeatNames Ids)
  (if (not (V Ready))
    (CallFunction|Init))
  (if (or (< Seat 0) (or (>= Seat (len Robots)) (>= Seat (len Ids))))
    (return))
  (set Bots Robots)
  (set Names SeatNames)
  (set CurBot (get Robots Seat))
  (set Id (get Ids Seat))
  (if (== (V CurBot) (V NullBot))
    (return))
  (bind _down (Class|BPRobot|IsDown :self (V CurBot)))
  (if _down
    (return))
  (bind _does (Json|GetField Entry "does"))
  (set DoesStr _does)
  (bind _parts (Utilities|String|ParseIntoArray :SourceString (V DoesStr) :Delimiter "; " :CullEmptyStrings true))
  (for _p _parts
    (CallFunction|PlayPart :Part _p)))
