; T38's fit rule. In: the bubble's rect (L, T, R, B) with its tail tip at (AX, B) on the speaker, in viewport units; Place has put
; the window (ViewTL, ViewBR) and each visible panel (KTL, KBR) in the same space. Out: the shift (DX, DY), the tail's x offset
; from the bubble's center (TX), and Up when the bubble ended up under its anchor (the tail then points up).
; 1. Inside the window's Margin. 2. Clear of each panel (grown by Gap), in SetKeepOut's order: of the four moves that clear it
; (left, right, down, up), the shortest that stays inside the margin. 3. The tail points at the anchor, kept TailInset from a corner.
(fn Fit (L T R B AX)
  (set FX 0.0)
  (set FY 0.0)
  (bind vl (+ (.x (V ViewTL)) (V Margin)))
  (bind vt (+ (.y (V ViewTL)) (V Margin)))
  (bind vr (- (.x (V ViewBR)) (V Margin)))
  (bind vb (- (.y (V ViewBR)) (V Margin)))
  (if (< L vl)
    (set FX (- vl L)))
  (if (> R vr)
    (set FX (- vr R)))
  (if (< T vt)
    (set FY (- vt T)))
  (if (> (+ B (V FY)) vb)
    (set FY (- vb B)))
  (for i (range (len (V KTL)))
    (bind ktl (get (V KTL) i))
    (bind kbr (get (V KBR) i))
    (bind kl (- (.x ktl) (V Gap)))
    (bind kt (- (.y ktl) (V Gap)))
    (bind kr (+ (.x kbr) (V Gap)))
    (bind kb (+ (.y kbr) (V Gap)))
    (bind nl (+ L (V FX)))
    (bind nr (+ R (V FX)))
    (bind nt (+ T (V FY)))
    (bind nb (+ B (V FY)))
    (if (and (and (< nl kr) (> nr kl)) (and (< nt kb) (> nb kt)))
      (set MvD 100000.0)
      (set MvX 0.0)
      (set MvY 0.0)
      (bind sl (- kl nr))
      (bind sr (- kr nl))
      (bind sd (- kb nt))
      (bind su (- kt nb))
      (if (and (>= (+ nl sl) vl) (< (- 0.0 sl) (V MvD)))
        (set MvD (- 0.0 sl))
        (set MvX sl)
        (set MvY 0.0))
      (if (and (<= (+ nr sr) vr) (< sr (V MvD)))
        (set MvD sr)
        (set MvX sr)
        (set MvY 0.0))
      (if (and (<= (+ nb sd) vb) (< sd (V MvD)))
        (set MvD sd)
        (set MvX 0.0)
        (set MvY sd))
      (if (and (>= (+ nt su) vt) (< (- 0.0 su) (V MvD)))
        (set MvD (- 0.0 su))
        (set MvX 0.0)
        (set MvY su))
      (if (< (V MvD) 100000.0)
        (set FX (+ (V FX) (V MvX)))
        (set FY (+ (V FY) (V MvY))))))
  (bind lim (Math|Float|Max(Float) (- (* 0.5 (- R L)) (V TailInset)) 0.0))
  (bind tx (Math|Float|Clamp(Float) :Value (- AX (+ (* 0.5 (+ L R)) (V FX))) :Min (- 0.0 lim) :Max lim))
  (return (V FX) (V FY) tx (>= (+ T (V FY)) B)))
