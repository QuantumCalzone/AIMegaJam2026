; T38: every look comes from UIStyle, as WBP_ChatLine's do: the speech in 20 pt white, an action and the vote's lead in
; 18 pt quiet italic, the typing dots in the 32 pt display type (at 20 pt they read as specks), padding 16 x 12, the window margin (Edge, 24), the gap to a panel (S2, 8),
; the fade (Panel, 0.25 s, in 10 steps) and the tail's inset from a corner (PanelRadius + S3). WrapAt (500) and DotTime (0.35 s)
; are the widget's own variables: UIStyle has no wrap width or dot time.
(fn ApplyStyle ()
  (bind fSay (Class|UIStyle|UIFont :Role "Speech"))
  (bind fAct (Class|UIStyle|UIFont :Role "Action"))
  (bind fDot (Class|UIStyle|UIFont :Role "Header"))
  (bind white (Class|UIStyle|UIColor :Role "Text"))
  (bind quiet (Class|UIStyle|UIColor :Role "Quiet"))
  (bind s2 (Class|UIStyle|UISize :Role "S2"))
  (bind s3 (Class|UIStyle|UISize :Role "S3"))
  (bind s4 (Class|UIStyle|UISize :Role "S4"))
  (bind edge (Class|UIStyle|UISize :Role "Edge"))
  (bind r (Class|UIStyle|UISize :Role "PanelRadius"))
  (bind fade (Class|UIStyle|UITime :Role "Panel"))
  (set Margin edge)
  (set Gap s2)
  (set TailInset (+ r s3))
  (set FadeStepTime (/ fade 10.0))
  (bind cWhite (Utilities|Struct|MakeSlateColor :SpecifiedColor white))
  (bind cQuiet (Utilities|Struct|MakeSlateColor :SpecifiedColor quiet))
  (Class|Border|SetPadding :self (Variables|WBP_SpeechBubble|GetEdge) :Padding (Utilities|Struct|MakeMargin s4 s3 s4 s3))
  (Class|Text|SetFont :self (Variables|WBP_SpeechBubble|GetSpeechText) :Font fSay)
  (Class|Text|SetFont :self (Variables|WBP_SpeechBubble|GetSpeechShown) :Font fSay)
  (Class|Text|SetFont :self (Variables|WBP_SpeechBubble|GetActionText) :Font fAct)
  (Class|Text|SetFont :self (Variables|WBP_SpeechBubble|GetLeadText) :Font fAct)
  (Class|Text|SetFont :self (Variables|WBP_SpeechBubble|GetDotsText) :Font fDot)
  (Class|Text|SetColorandOpacity :self (Variables|WBP_SpeechBubble|GetSpeechText) :ColorAndOpacity cWhite)
  (Class|Text|SetColorandOpacity :self (Variables|WBP_SpeechBubble|GetSpeechShown) :ColorAndOpacity cWhite)
  (Class|Text|SetColorandOpacity :self (Variables|WBP_SpeechBubble|GetDotsText) :ColorAndOpacity cWhite)
  (Class|Text|SetColorandOpacity :self (Variables|WBP_SpeechBubble|GetActionText) :ColorAndOpacity cQuiet)
  (Class|Text|SetColorandOpacity :self (Variables|WBP_SpeechBubble|GetLeadText) :ColorAndOpacity cQuiet)
  (Class|TextLayoutWidget|SetWrapTextAt :self (Variables|WBP_SpeechBubble|GetSpeechText) :WrapTextAt (V WrapAt))
  (Class|TextLayoutWidget|SetWrapTextAt :self (Variables|WBP_SpeechBubble|GetSpeechShown) :WrapTextAt (V WrapAt))
  (Class|TextLayoutWidget|SetWrapTextAt :self (Variables|WBP_SpeechBubble|GetActionText) :WrapTextAt (V WrapAt))
  (Class|TextLayoutWidget|SetWrapTextAt :self (Variables|WBP_SpeechBubble|GetLeadText) :WrapTextAt (V WrapAt)))
