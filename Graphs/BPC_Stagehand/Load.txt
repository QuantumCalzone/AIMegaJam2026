; BPC_Stagehand.Load (T61): reads Content/NHOS/Data/stagehand.json once per run. A missing or broken file leaves
; Loaded false, and every Say does nothing (the show plays on without tips).
(fn Load ()
  (set Loaded false)
  (set HoldBase 2.0)
  (set HoldPerWord 0.3)
  (set HoldMin 4.0)
  (set HoldMax 9.0)
  (set Gap 0.5)
  (bind _p (cat (Utilities|Paths|GetProjectContentDirectory) "NHOS/Data/stagehand.json"))
  (bind (_o _ok) (Json|LoadJsonfromFile (Utilities|Struct|MakeFilePath _p)))
  (if (not _ok)
    (print "NHOS: STAGEHAND file missing or not valid JSON")
    (return))
  (set Data _o)
  (bind _m (Json|GetField (V Data) "moments"))
  (set Moments _m)
  (bind _s (Json|GetField (V Data) "style"))
  (set Style _s)
  (CallFunction|ReadHold)
  (set Loaded true)
  (print "NHOS: STAGEHAND loaded"))
