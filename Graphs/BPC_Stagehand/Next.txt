; BPC_Stagehand.Next (T61): puts the next queued line in the earpiece, one at a time, then waits for its fade in,
; its hold (base + per_word x words, kept between min and max), its fade out and the gap.
; The earpiece is its own overlay at Z 15, made on first use: over WBP_Stage, under the title (20), over the end screen (10).
(fn Next ()
  (if (== (len (V QKey)) 0)
    (set Busy false)
    (return))
  (set Busy true)
  (bind _k (get (V QKey) 0))
  (bind _l (get (V QLine) 0))
  (bind _s (get (V QSpot) 0))
  (set TmpS _l)
  (set TmpKey _k)
  (set TmpSpot _s)
  (Utilities|Array|RemoveIndex (V QKey) 0)
  (Utilities|Array|RemoveIndex (V QLine) 0)
  (Utilities|Array|RemoveIndex (V QSpot) 0)
  (CallFunction|MakeEar)
  (bind _words (Utilities|String|ParseIntoArray :SourceString (V TmpS) :Delimiter " " :CullEmptyStrings true))
  (bind _n (len _words))
  (bind _hold (Math|Float|Clamp(Float) (+ (V HoldBase) (* (V HoldPerWord) _n)) (V HoldMin) (V HoldMax)))
  (bind _fade (Class|UIStyle|UITime :Role "Panel"))
  (print (cat (cat (cat "NHOS: TIP " (V TmpKey)) (cat " words=" (str _n))) (cat " hold=" (Utilities|String|ToString(Float) _hold))))
  (Class|WBPEarpiece|SetSpot :self (V Ear) :Spot (V TmpSpot))
  (Class|WBPEarpiece|Play :self (V Ear) :Line (V TmpS) :Hold _hold)
  (Utilities|Time|SetTimerbyFunctionName :Object self :FunctionName "Next" :Time (+ (+ _hold (* 2.0 _fade)) (V Gap)) :bLooping false))
