; T34 (row 2): slot S's tag. Its words (TagWords): my seat reads YOU (the doc), a passed robot gets "(passed)" over its
; name, a cut robot that rose on a win (T87's Rise: PassedOn, and BP_Show's SeatCut) gets "(cut)"; a cut robot lying on
; stage (IsCut) has no tag, since the stamp and the roster say it. Its place (TagX, TagY): the plate's top left in viewport
; units, centered over the label with its bottom TagLift cm above the label's bottom, inside the window's Edge.
(fn TagOne (S)
  (bind _robots (Class|BPShow|GetRobots :self (V Show)))
  (if (>= S (len _robots))
    (CallFunction|TagShow :S S :On false)
    (return))
  (set TagRobot (get _robots S))
  (if (== (V TagRobot) (V NullRobot))
    (CallFunction|TagShow :S S :On false)
    (return))
  (bind _labels (Class|BPShow|GetSeatLabel :self (V Show)))
  (if (>= S (len _labels))
    (CallFunction|TagShow :S S :On false)
    (return))
  (bind _label (get _labels S))
  (if (or (Utilities|String|IsEmpty _label) (Class|BPRobot|GetIsCut :self (V TagRobot)))
    (CallFunction|TagShow :S S :On false)
    (return))
  (bind _cut (Class|BPShow|GetSeatCut :self (V Show)))
  (bind _wasCut (and (< S (len _cut)) (get _cut (Math|Integer|Min(Integer) S (- (len _cut) 1)))))
  (bind _passed (Class|BPRobot|GetPassedOn :self (V TagRobot)))
  (bind _mine (== S (Class|BPShow|GetHumanSeat :self (V Show))))
  (if _mine
    (set TagMine S))
  (CallFunction|TagWords :S S :Name (select _mine "YOU" _label) :Mark (select _passed (select _wasCut "(cut)" "(passed)") "") :ColorKey _label)
  (bind _lab (Transformation|GetWorldLocation :self (Class|BPRobot|GetLabel :self (V TagRobot))))
  (bind _at (+ _lab (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z (V TagLift))))
  (bind (_p _ok) (Viewport|ProjectWorldLocationtoWidgetPosition :PlayerController (V TagPC) :WorldLocation _at :bPlayerViewportRelative true))
  (if (not _ok)
    (CallFunction|TagShow :S S :On false)
    (return))
  (bind _m (Class|UIStyle|UISize :Role "Edge"))
  (bind _w (get (V TagW) S))
  (bind _h (get (V TagH) S))
  (bind _x (Math|Float|Clamp(Float) :Value (- (.x _p) (* 0.5 _w)) :Min (+ (.x (V TagVTL)) _m) :Max (- (- (.x (V TagVBR)) _m) _w)))
  (bind _y (Math|Float|Max(Float) (- (.y _p) _h) (+ (.y (V TagVTL)) _m)))
  (Utilities|Array|SetArrayElem (V TagX) S _x)
  (Utilities|Array|SetArrayElem (V TagY) S _y)
  (CallFunction|TagShow :S S :On true))
