; T34 (row 4): three dots under the header in a scene turn (the doc's Header: "turn as pips"): done in Gold, now in Gold
; with a white outline, to come in Raised with a Quiet outline (10.7:1 on ink, so they read as dots, not gaps). Detail is
; the header's right part; the pips show only when it holds "TURN n".
(fn SetPips (Detail)
  (if (not (V PipReady))
    (set PipReady true)
    (add (V PipArr) (Variables|WBP_Stage|GetPip0))
    (add (V PipArr) (Variables|WBP_Stage|GetPip1))
    (add (V PipArr) (Variables|WBP_Stage|GetPip2)))
  (bind _at (Utilities|String|FindSubstring :SearchIn Detail :Substring "TURN " :bUseCase false :bSearchFromEnd false :StartPosition -1))
  (if (< _at 0)
    (Widget|SetVisibility :self (Variables|WBP_Stage|GetTurnPips) :InVisibility "Collapsed")
    (return))
  (bind _n (Utilities|String|StringToInteger :InString (Utilities|String|Mid :SourceString Detail :Start (+ _at 5) :Count 1)))
  (bind r (Class|UIStyle|UISize :Role "S2"))
  (bind w (Class|UIStyle|UISize :Role "InputOutline"))
  (bind gold (Class|UIStyle|UIColor :Role "Gold"))
  (bind text (Class|UIStyle|UIColor :Role "Text"))
  (bind raised (Class|UIStyle|UIColor :Role "Raised"))
  (bind quiet (Class|UIStyle|UIColor :Role "Quiet"))
  (for _i (range 3)
    (bind _done (< (+ _i 1) _n))
    (bind _now (== (+ _i 1) _n))
    (bind _p (get (V PipArr) _i))
    (Class|Border|SetBrushColor :self _p :BrushColor (Utilities|Struct|MakeLinearColor :R 1.0 :G 1.0 :B 1.0 :A 1.0))
    (Appearance|SetBrush :self _p :InBrush (Class|UIStyle|Brush :Fill (select (or _done _now) gold raised) :Outline (select _now text (select _done gold quiet)) :Width w :Radius r)))
  (Widget|SetVisibility :self (Variables|WBP_Stage|GetTurnPips) :InVisibility "HitTestInvisible"))
