; T85: the seat whose card shows, or -1. TipTry tests each robot slot; the winner's box is left in BestL/T/R/B for
; TipPlace. No card while the mouse is over a panel (TipInPanel): the robots are never behind one (T27), so this only
; stops a card popping up while I read my card or type. The mouse, the controller and the camera are read once here.
; GetMousePositionOnViewport keeps the last position when the mouse leaves the window.
; T34: a card that the arrow keys forced (CastKey, TipKeyOn) goes when the mouse moves more than 4 units.
(fn TipPick ()
  (set TipInPanel false)
  (set TipBest -1)
  (set BestD 1.0e9)
  (set TipMouse (Viewport|GetMousePositiononViewport))
  (if (and (V TipKeyOn) (> (Math|Vector2D|Distance2D :V1 (V TipMouse) :V2 (V TipMouseKey)) 4.0))
    (set TipForce "")
    (set TipKeyOn false))
  (set TipForceOn (not (Utilities|String|IsEmpty (V TipForce))))
  (for i (range (len (V TipKTL)))
    (bind ktl (get (V TipKTL) i))
    (bind kbr (get (V TipKBR) i))
    (if (and (and (>= (.x (V TipMouse)) (.x ktl)) (<= (.x (V TipMouse)) (.x kbr))) (and (>= (.y (V TipMouse)) (.y ktl)) (<= (.y (V TipMouse)) (.y kbr))))
      (set TipInPanel true)))
  (set TipPC (Game|GetPlayerController 0))
  (bind _cam (Game|GetPlayerCameraManager :PlayerIndex 0))
  (set TipCL (Camera|GetCameraLocation :self _cam))
  (bind (_ax _ay _az) (Math|Rotator|GetAxes (Camera|GetCameraRotation :self _cam)))
  (set TipCR _ay)
  (for s (range (len (Class|BPShow|GetRobots :self (V Show))))
    (CallFunction|TipTry :S s))
  (return (V TipBest)))
