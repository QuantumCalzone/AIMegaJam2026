; T34 (C4): who is still in, on my card under my want: one chip per seat in seat order, in its seat color with its screen
; name (mine reads YOU), a cut seat at 40% with a strike through its name. TagTick calls this every frame; it redraws only
; when the cast or a cut changes (RosterN, RosterCut), so a cut shows at once and Play Again redraws it.
(fn DrawRoster ()
  (bind _labels (Class|BPShow|GetSeatLabel :self (V Show)))
  (bind _cut (Class|BPShow|GetSeatCut :self (V Show)))
  (set TagHit false)
  (if (!= (len _labels) (V RosterN))
    (set TagHit true))
  (if (!= (len _cut) (len (V RosterCut)))
    (set TagHit true))
  (for _i (range (Math|Integer|Min(Integer) (len _cut) (len (V RosterCut))))
    (if (xor (get _cut _i) (get (V RosterCut) _i))
      (set TagHit true)))
  (if (not (V TagHit))
    (return))
  (set RosterN (len _labels))
  (set RosterCut _cut)
  (if (== (len (V RosterChips)) 0)
    (add (V RosterChips) (Variables|WBP_Stage|GetRoster0))
    (add (V RosterChips) (Variables|WBP_Stage|GetRoster1))
    (add (V RosterChips) (Variables|WBP_Stage|GetRoster2))
    (add (V RosterChips) (Variables|WBP_Stage|GetRoster3))
    (add (V RosterChips) (Variables|WBP_Stage|GetRoster4))
    (add (V RosterNames) (Variables|WBP_Stage|GetRosterName0))
    (add (V RosterNames) (Variables|WBP_Stage|GetRosterName1))
    (add (V RosterNames) (Variables|WBP_Stage|GetRosterName2))
    (add (V RosterNames) (Variables|WBP_Stage|GetRosterName3))
    (add (V RosterNames) (Variables|WBP_Stage|GetRosterName4))
    (add (V RosterStrikes) (Variables|WBP_Stage|GetRosterStrikeSize0))
    (add (V RosterStrikes) (Variables|WBP_Stage|GetRosterStrikeSize1))
    (add (V RosterStrikes) (Variables|WBP_Stage|GetRosterStrikeSize2))
    (add (V RosterStrikes) (Variables|WBP_Stage|GetRosterStrikeSize3))
    (add (V RosterStrikes) (Variables|WBP_Stage|GetRosterStrikeSize4))
    (bind cInk (Class|UIStyle|UIColor :Role "PaperText"))
    (Class|Border|SetBrushColor :self (Variables|WBP_Stage|GetRosterStrike0) :BrushColor cInk)
    (Class|Border|SetBrushColor :self (Variables|WBP_Stage|GetRosterStrike1) :BrushColor cInk)
    (Class|Border|SetBrushColor :self (Variables|WBP_Stage|GetRosterStrike2) :BrushColor cInk)
    (Class|Border|SetBrushColor :self (Variables|WBP_Stage|GetRosterStrike3) :BrushColor cInk)
    (Class|Border|SetBrushColor :self (Variables|WBP_Stage|GetRosterStrike4) :BrushColor cInk))
  (for _k (range 5)
    (CallFunction|RosterOne :I _k)))
