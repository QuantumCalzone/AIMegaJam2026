; T34 (T33-1): the gold focus ring (UIStyle FocusRing) on the Border that wraps an in-show button. A ButtonStyle has no
; focused brush (T32, C7), so RingTick draws this on the button that has the keyboard focus.
(fn ShowRing (Ring On)
  (bind w (Class|UIStyle|UISize :Role "FocusRing"))
  (bind r (Class|UIStyle|UISize :Role "ButtonRadius"))
  (bind gap (Class|UIStyle|UISize :Role "S1"))
  (bind gold (Class|UIStyle|UIColor :Role "Gold"))
  (bind clear (Utilities|Struct|MakeLinearColor :R 0.0 :G 0.0 :B 0.0 :A 0.0))
  (Class|Border|SetBrushColor :self Ring :BrushColor (Utilities|Struct|MakeLinearColor :R 1.0 :G 1.0 :B 1.0 :A 1.0))
  (bind b (Class|UIStyle|Brush :Fill clear :Outline (select On gold clear) :Width w :Radius (+ r gap)))
  (Appearance|SetBrush :self Ring :InBrush b))
