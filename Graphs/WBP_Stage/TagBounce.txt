; T34 (row 2): my tag bounces on my move (the doc: "yours bounces on your move"; a small overshoot is allowed for the YOU
; tag). SetPrompt starts it when my input opens (TagBounceT 0); each step is a 0.025 s timer, ten steps: 1.25, then down
; through 0.95 to 1.0. It skips a step while my tag is the speaker's (already grown).
(fn TagBounce ()
  (if (or (< (V TagMine) 0) (>= (V TagBounceT) 1.0))
    (return))
  (set TagBounceT (Math|Float|Min(Float) (+ (V TagBounceT) 0.1) 1.0))
  (bind _t (V TagBounceT))
  (bind _k (select (< _t 0.5) (- 1.25 (* 0.6 _t)) (+ 0.95 (* 0.1 (- _t 0.5)))))
  (if (not (get (V TagSpkA) (V TagMine)))
    (Widget|Transform|SetRenderScale :self (get (V TagFrames) (V TagMine)) :Scale (Math|Vector2D|MakeVector2D :X _k :Y _k)))
  (if (< (V TagBounceT) 1.0)
    (Utilities|Time|SetTimerbyFunctionName :Object self :FunctionName "TagBounce" :Time 0.025 :bLooping false)))
