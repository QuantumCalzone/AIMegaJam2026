; T34: one step of CastKey: CastK moves by Step until it lands on an AI seat on stage (CastHit), or falls off the end (none).
(fn CastNext (Step N Me)
  (if (or (>= (V CastHit) 0) (< (V CastK) -1))
    (return))
  (set CastK (+ (V CastK) Step))
  (if (or (< (V CastK) 0) (>= (V CastK) N))
    (set CastK -5)
    (return))
  (bind _cut (Class|BPShow|GetSeatCut :self (V Show)))
  (if (== (V CastK) Me)
    (return))
  (if (and (< (V CastK) (len _cut)) (get _cut (Math|Integer|Min(Integer) (V CastK) (- (len _cut) 1))))
    (return))
  (set CastHit (V CastK)))
