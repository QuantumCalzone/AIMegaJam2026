; T34 (T85's keyboard finding): Left and Right in Scene Prep (the middle panel shows) step the cast's hover card to the
; previous or next AI robot on stage, through T85's TipForce (the cast_tip switch's path: the same card, place and
; CASTTIP line), wrapping through none. A mouse move clears it (TipPick, TipKeyOn). Handled is true for the two keys.
(fn CastKey (Key)
  (if (not (or (streq Key "Left") (streq Key "Right")))
    (return false))
  (if (not (Widget|IsVisible :self (Variables|WBP_Stage|GetPrepSize)))
    (return false))
  (CallFunction|ShowRef)
  (if (== (V Show) (V NullShow))
    (return false))
  (bind _ids (Class|BPShow|GetSeatId :self (V Show)))
  (bind _n (len _ids))
  (bind _step (select (streq Key "Right") 1 -1))
  (bind _from (select (V TipKeyOn) (Utilities|Array|FindItem _ids (V TipForce)) -1))
  (set CastK (select (< _from 0) (select (> _step 0) -1 _n) _from))
  (set CastHit -1)
  (for _t (range _n)
    (CallFunction|CastNext :Step _step :N _n :Me (Class|BPShow|GetHumanSeat :self (V Show))))
  (set TipForce (select (>= (V CastHit) 0) (get _ids (Math|Integer|Max(Integer) (V CastHit) 0)) ""))
  (set TipKeyOn (>= (V CastHit) 0))
  (set TipMouseKey (Viewport|GetMousePositiononViewport))
  (print (cat "NHOS: CASTTIP key " (select (V TipKeyOn) (V TipForce) "none")))
  (return true))
