; WBP_Stage.SubmitInput: Enter in the input box (OnTextCommitted) and Action! (OnClicked(SendButton)) both land here.
; It trims the typed text, prints NHOS: TYPED and hands it to BP_Show through OnSend, then clears the box and keeps
; the keyboard focus in it.
; T64: while a picked action still lacks a chip, it does nothing (step 7). While the builder is up, an empty line
; goes through too, so a No action turn with no words sends; with the builder down, an empty line still sends nothing.
; T65: a SetText from the game fires no OnTextChanged, so the clear runs the speech field's change chain itself
; (SpeechChanged): the counter goes back to 0/15 and the @ list closes.
; T66: a timeout (ForceSend) sends an unfinished action too; HandleSend gets it as a look around.
; T34 (T33-1): Enter on an empty field passes while Pass is enabled (Free Talk, a last word, a reaction), as the Pass
; button and T66's TimeUp do; with the builder up (a scene turn) an empty line is No action, as before.
(fn SubmitInput ()
  (bind _blk (Class|WBPActionBuilder|Blocks :self (Variables|WBP_Stage|GetActionBuilder)))
  (if (and _blk (not (V ForceSend)))
    (print "NHOS: BUILDER blocked")
    (return))
  (bind _bon (Class|WBPActionBuilder|IsOn :self (Variables|WBP_Stage|GetActionBuilder)))
  (bind _text (Utilities|String|Trim (Utilities|String|TrimTrailing (Utilities|String|ToString(Text) (Class|TextBox|GetText :self (Variables|WBP_Stage|GetInputBox))))))
  (if (and (and (Utilities|String|IsEmpty _text) (not _bon)) (and (Variables|Default|GetPassOn) (Variables|Default|GetLastCanType)))
    (print "NHOS: TYPED pass")
    (Default|CallOnSend self "pass")
    (return))
  (if (or (not (Utilities|String|IsEmpty _text)) _bon)
    (print (cat "NHOS: TYPED " _text))
    (Default|CallOnSend self _text))
  (Widget|SetText(TextBox) :self (Variables|WBP_Stage|GetInputBox) :InText (Utilities|Text|ToText(String) ""))
  (CallFunction|SpeechChanged :Text "")
  (Widget|SetKeyboardFocus :self (Variables|WBP_Stage|GetInputBox)))
