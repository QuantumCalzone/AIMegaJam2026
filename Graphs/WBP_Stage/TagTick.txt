; T34 (row 2): each frame from SetPrompt. Every robot on stage gets its name tag at its label (the Label component's place,
; so T47's hats still lift it), projected into the window, kept inside Edge, and pushed up over an earlier tag it would
; overlap (TagStack). The speaker's tag grows 20% (the doc's Name tags), a tag under another robot's bubble dims, and the
; tag of the robot under the mouse glows gold (TagDim). Nothing before the show is Ready. The roster on my card follows
; the cuts (DrawRoster, when SeatCut changes). TagOne does one slot (returns, no exec-branching node, T107's else trap).
(fn TagTick ()
  (CallFunction|TagInit)
  (CallFunction|ShowRef)
  (if (== (V Show) (V NullShow))
    (CallFunction|TagHideAll)
    (return))
  (if (not (Class|BPShow|GetReady :self (V Show)))
    (CallFunction|TagHideAll)
    (return))
  (bind _robots (Class|BPShow|GetRobots :self (V Show)))
  (if (!= (len _robots) (V TagN))
    (set TagN (len _robots))
    (CallFunction|HideLabels))
  (CallFunction|DrawRoster)
  (set TagPC (Game|GetPlayerController 0))
  (bind vg (Widget|GetCachedGeometry :self (Variables|WBP_Stage|GetRootBox)))
  (bind vs (UserInterface|Geometry|GetLocalSize :Geometry vg))
  (bind z (Math|Vector2D|MakeVector2D :X 0.0 :Y 0.0))
  (bind (vp0 vtl) (UserInterface|Geometry|LocaltoViewport :Geometry vg :LocalCoordinate z))
  (bind (vp1 vbr) (UserInterface|Geometry|LocaltoViewport :Geometry vg :LocalCoordinate vs))
  (set TagVTL vtl)
  (set TagVBR vbr)
  (for _s (range 5)
    (CallFunction|TagOne :S _s))
  (CallFunction|TagStack)
  (bind _lp (Class|BPShow|GetLinePlayer :self (V Show)))
  (bind _spk (select (Class|BPCLinePlayer|GetBubbleUp :self _lp) (Class|BPCLinePlayer|GetBubbleSeat :self _lp) -1))
  (for _s (range 5)
    (CallFunction|TagDim :S _s)
    (CallFunction|TagSpk :S _s :On (== _s _spk))))
