; T34 (row 2): sets TagHit when another robot J shows a bubble that overlaps S's tag. When S's own bubble overlaps it (a
; bubble pushed down by the header or the banner), it sets TagOwnHit and the bubble's bottom (TagOwnBot) instead, and
; TagDim moves the tag just under the bubble, so the speaker keeps its name (the cold review: a bubble with no tag). The bubble is BP_Robot's BubbleW (T38),
; made on first use (HasBubble); HideNow collapses its BubbleBox, and a bubble faded to nothing (FadeStep) doesn't count. Viewport units, as TipKeepRects measures.
(fn TagBubbleHit (S J)
  (if (V TagHit)
    (return))
  (set TagRobot (get (Class|BPShow|GetRobots :self (V Show)) J))
  (if (== (V TagRobot) (V NullRobot))
    (return))
  (if (not (Class|BPRobot|GetHasBubble :self (V TagRobot)))
    (return))
  (bind _box (Class|WBPSpeechBubble|GetBubbleBox :self (Class|BPRobot|GetBubbleW :self (V TagRobot))))
  (if (not (Widget|IsVisible :self _box))
    (return))
  (if (< (Widget|GetRenderOpacity :self _box) 0.05)
    (return))
  (bind z (Math|Vector2D|MakeVector2D :X 0.0 :Y 0.0))
  (bind g (Widget|GetCachedGeometry :self _box))
  (bind sz (UserInterface|Geometry|GetLocalSize :Geometry g))
  (bind (p0 tl) (UserInterface|Geometry|LocaltoViewport :Geometry g :LocalCoordinate z))
  (bind (p1 br) (UserInterface|Geometry|LocaltoViewport :Geometry g :LocalCoordinate sz))
  (bind _x (get (V TagX) S))
  (bind _y (get (V TagY) S))
  (bind _over (and (and (< _x (.x br)) (> (+ _x (get (V TagW) S)) (.x tl))) (and (< _y (.y br)) (> (+ _y (get (V TagH) S)) (.y tl)))))
  (if (not _over)
    (return))
  (if (== S J)
    (set TagOwnHit true)
    (set TagOwnBot (.y br))
    (return))
  (set TagHit true))
