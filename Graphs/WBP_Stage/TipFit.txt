; T85: WBP_SpeechBubble.Fit (T38) without the tail. In: the card's rect (L, T, R, B) in viewport units; TipKeepRects has put
; the window (TipVTL, TipVBR) and each visible panel (TipKTL, TipKBR) in the same space. Out: the shift (DX, DY).
; 1. Inside the window less Edge. 2. Clear of each panel grown by S2: of the four moves that clear it (left, right,
; down, up), the shortest that stays inside the margin.
(fn TipFit (L T R B)
  (bind _m (Class|UIStyle|UISize :Role "Edge"))
  (bind _g (Class|UIStyle|UISize :Role "S2"))
  (set TFX 0.0)
  (set TFY 0.0)
  (bind vl (+ (.x (V TipVTL)) _m))
  (bind vt (+ (.y (V TipVTL)) _m))
  (bind vr (- (.x (V TipVBR)) _m))
  (bind vb (- (.y (V TipVBR)) _m))
  (if (< L vl)
    (set TFX (- vl L)))
  (if (> R vr)
    (set TFX (- vr R)))
  (if (< T vt)
    (set TFY (- vt T)))
  (if (> (+ B (V TFY)) vb)
    (set TFY (- vb B)))
  (for i (range (len (V TipKTL)))
    (bind ktl (get (V TipKTL) i))
    (bind kbr (get (V TipKBR) i))
    (bind kl (- (.x ktl) _g))
    (bind kt (- (.y ktl) _g))
    (bind kr (+ (.x kbr) _g))
    (bind kb (+ (.y kbr) _g))
    (bind nl (+ L (V TFX)))
    (bind nr (+ R (V TFX)))
    (bind nt (+ T (V TFY)))
    (bind nb (+ B (V TFY)))
    (if (and (and (< nl kr) (> nr kl)) (and (< nt kb) (> nb kt)))
      (set TMvD 100000.0)
      (set TMvX 0.0)
      (set TMvY 0.0)
      (bind sl (- kl nr))
      (bind sr (- kr nl))
      (bind sd (- kb nt))
      (bind su (- kt nb))
      (if (and (>= (+ nl sl) vl) (< (- 0.0 sl) (V TMvD)))
        (set TMvD (- 0.0 sl))
        (set TMvX sl)
        (set TMvY 0.0))
      (if (and (<= (+ nr sr) vr) (< sr (V TMvD)))
        (set TMvD sr)
        (set TMvX sr)
        (set TMvY 0.0))
      (if (and (<= (+ nb sd) vb) (< sd (V TMvD)))
        (set TMvD sd)
        (set TMvX 0.0)
        (set TMvY sd))
      (if (and (>= (+ nt su) vt) (< (- 0.0 su) (V TMvD)))
        (set TMvD (- 0.0 su))
        (set TMvX 0.0)
        (set TMvY su))
      (if (< (V TMvD) 100000.0)
        (set TFX (+ (V TFX) (V TMvX)))
        (set TFY (+ (V TFY) (V TMvY))))))
  (return (V TFX) (V TFY)))
