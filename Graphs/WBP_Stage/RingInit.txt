; T34 (T33-1): the six focusable in-show buttons and their rings, in one order, once. Every ring starts clear.
(fn RingInit ()
  (if (V RingReady)
    (return))
  (set RingReady true)
  (clear (V RingBtns))
  (clear (V RingFrames))
  (clear (V RingOn))
  (add (V RingBtns) (Variables|WBP_Stage|GetStartButton))
  (add (V RingBtns) (Variables|WBP_Stage|GetVoteBtn0))
  (add (V RingBtns) (Variables|WBP_Stage|GetVoteBtn1))
  (add (V RingBtns) (Variables|WBP_Stage|GetVoteBtn2))
  (add (V RingBtns) (Variables|WBP_Stage|GetVoteBtn3))
  (add (V RingBtns) (Variables|WBP_Stage|GetContinueButton))
  (add (V RingFrames) (Variables|WBP_Stage|GetStartRing))
  (add (V RingFrames) (Variables|WBP_Stage|GetVoteRing0))
  (add (V RingFrames) (Variables|WBP_Stage|GetVoteRing1))
  (add (V RingFrames) (Variables|WBP_Stage|GetVoteRing2))
  (add (V RingFrames) (Variables|WBP_Stage|GetVoteRing3))
  (add (V RingFrames) (Variables|WBP_Stage|GetContinueRing))
  (for _i (range 6)
    (add (V RingOn) false)
    (CallFunction|ShowRing :Ring (get (V RingFrames) _i) :On false)))
