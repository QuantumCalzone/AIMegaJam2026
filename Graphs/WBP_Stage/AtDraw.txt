; T65: opens, redraws or closes the @ list. AtOpen is read before it is set: a bind of a variable is a pure node, so
; reading it after the set would give the new value (found in T65's check).
(fn AtDraw (Open)
  (if Open
    (CallFunction|AtShow :Fresh (not (V AtOpen)))
    (else
      (Widget|SetVisibility :self (Variables|WBP_Stage|GetAtPop) :InVisibility "Collapsed")
      (if (V AtOpen)
        (print "NHOS: MENTION close"))))
  (set AtOpen Open))
