(fn RosterOne (I)
  (bind _labels (Class|BPShow|GetSeatLabel :self (V Show)))
  (bind _chip (get (V RosterChips) I))
  (if (>= I (len _labels))
    (Widget|SetVisibility :self _chip :InVisibility "Collapsed")
    (return))
  (bind _label (get _labels I))
  (if (Utilities|String|IsEmpty _label)
    (Widget|SetVisibility :self _chip :InVisibility "Collapsed")
    (return))
  (bind _cut (and (< I (len (V RosterCut))) (get (V RosterCut) (Math|Integer|Min(Integer) I (- (len (V RosterCut)) 1)))))
  (bind _mine (== I (Class|BPShow|GetHumanSeat :self (V Show))))
  (bind r (Class|UIStyle|UISize :Role "TagRadius"))
  (bind clear (Utilities|Struct|MakeLinearColor :R 0.0 :G 0.0 :B 0.0 :A 0.0))
  (Class|Border|SetBrushColor :self _chip :BrushColor (Utilities|Struct|MakeLinearColor :R 1.0 :G 1.0 :B 1.0 :A 1.0))
  (bind _seat (CallFunction|TagColor :S I))
  (bind _b (Class|UIStyle|Brush :Fill _seat :Outline clear :Width 0.0 :Radius r))
  (Appearance|SetBrush :self _chip :InBrush _b)
  (bind _name (get (V RosterNames) I))
  (Class|Text|SetText :self _name :Text (Utilities|Text|ToText(String) (select _mine "YOU" _label)))
  (Class|Text|SetFont :self _name :Font (Class|UIStyle|UIFont :Role "Caption"))
  (Class|Text|SetColorandOpacity :self _name :ColorAndOpacity (Utilities|Struct|MakeSlateColor :SpecifiedColor (Class|UIStyle|UIColor :Role "PaperText")))
  (Widget|SetRenderOpacity :self _chip :InOpacity (select _cut 0.4 1.0))
  (if _cut
    (Widget|SetVisibility :self (get (V RosterStrikes) I) :InVisibility "HitTestInvisible")
    (else
      (Widget|SetVisibility :self (get (V RosterStrikes) I) :InVisibility "Collapsed")))
  (Widget|SetVisibility :self _chip :InVisibility "HitTestInvisible"))
