; T34 (row 2): the five name tags' widgets in seat order, once, with their type: the name in Name (20 bold) in PaperText
; ink on the seat-color plate (5.9:1 on Anthropic, the lowest), the mark ("(passed)", "(cut)") in Caption.
(fn TagInit ()
  (if (V TagReady)
    (return))
  (set TagReady true)
  (set TagLift 6.0)
  (set TagMine -1)
  (add (V TagFrames) (Variables|WBP_Stage|GetTagFrame0))
  (add (V TagFrames) (Variables|WBP_Stage|GetTagFrame1))
  (add (V TagFrames) (Variables|WBP_Stage|GetTagFrame2))
  (add (V TagFrames) (Variables|WBP_Stage|GetTagFrame3))
  (add (V TagFrames) (Variables|WBP_Stage|GetTagFrame4))
  (add (V TagMarks) (Variables|WBP_Stage|GetTagMark0))
  (add (V TagMarks) (Variables|WBP_Stage|GetTagMark1))
  (add (V TagMarks) (Variables|WBP_Stage|GetTagMark2))
  (add (V TagMarks) (Variables|WBP_Stage|GetTagMark3))
  (add (V TagMarks) (Variables|WBP_Stage|GetTagMark4))
  (add (V TagNames) (Variables|WBP_Stage|GetTagName0))
  (add (V TagNames) (Variables|WBP_Stage|GetTagName1))
  (add (V TagNames) (Variables|WBP_Stage|GetTagName2))
  (add (V TagNames) (Variables|WBP_Stage|GetTagName3))
  (add (V TagNames) (Variables|WBP_Stage|GetTagName4))
  (bind fName (Class|UIStyle|UIFont :Role "Name"))
  (bind fCap (Class|UIStyle|UIFont :Role "Caption"))
  (bind cInk (Utilities|Struct|MakeSlateColor :SpecifiedColor (Class|UIStyle|UIColor :Role "PaperText")))
  (for _i (range 5)
    (add (V TagTxt) "")
    (add (V TagMk) "")
    (add (V TagW) 0.0)
    (add (V TagH) 0.0)
    (add (V TagX) 0.0)
    (add (V TagY) 0.0)
    (add (V TagOnA) false)
    (add (V TagDimA) false)
    (add (V TagGlowA) false)
    (add (V TagSpkA) false)
    (Class|Text|SetFont :self (get (V TagNames) _i) :Font fName)
    (Class|Text|SetColorandOpacity :self (get (V TagNames) _i) :ColorAndOpacity cInk)
    (Class|Text|SetFont :self (get (V TagMarks) _i) :Font fCap)
    (Class|Text|SetColorandOpacity :self (get (V TagMarks) _i) :ColorAndOpacity cInk)))
