; T68: a phase starts on screen, from RunCue's "stat" branch and from BP_Show.OpenPrep (scene_prep).
; At each Scene Prep it first sends the call counts since the last one (BPC_LLMClient.FlushCalls; my
; person's answer 1), then the phase. At the first Scene Prep the counts are empty, so it sends no calls.
(fn SendPhase (Key Scene)
  (set CurPhase Key)
  (set CurScene Scene)
  (if (streq Key "scene_prep")
    (Class|BPCLLMClient|FlushCalls :self (V Client)))
  (bind _f (Json|LoadJsonfromString "{}"))
  (Json|SetField _f "phase" Key)
  (Json|SetField _f "scene" Scene)
  (CallFunction|Send :Event "phase" :Fields _f))
