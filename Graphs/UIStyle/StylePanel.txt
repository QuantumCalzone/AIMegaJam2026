; UIStyle.StylePanel(Panel, Look). Looks: Ink (plum-black at 92% with old-gold trim: the running talk),
; Paper (cream card with old-gold trim: what is yours or decisive), Note (the gold director's note). Unknown looks get Ink.
(fn StylePanel (Panel Look)
  (bind r (CallFunction|UISize :Role "PanelRadius"))
  (bind trim (CallFunction|UISize :Role "PanelTrim"))
  (bind ctrim (CallFunction|UISize :Role "CardTrim"))
  (bind gold (CallFunction|UIColor :Role "OldGold"))
  (Class|Border|SetBrushColor :self Panel :BrushColor (Utilities|Struct|MakeLinearColor 1.0 1.0 1.0 1.0))
  (if (Utilities|String|EqualExactly(String) Look "Paper")
    (bind paper (CallFunction|UIColor :Role "Paper"))
    (bind bP (CallFunction|Brush :Fill paper :Outline gold :Width ctrim :Radius r))
    (Appearance|SetBrush :self Panel :InBrush bP)
    (return))
  (if (Utilities|String|EqualExactly(String) Look "Note")
    (bind note (CallFunction|UIColor :Role "Gold"))
    (bind bN (CallFunction|Brush :Fill note :Outline gold :Width trim :Radius r))
    (Appearance|SetBrush :self Panel :InBrush bN)
    (return))
  (bind ink (CallFunction|UIColor :Role "Ink"))
  (bind bI (CallFunction|Brush :Fill ink :Outline gold :Width trim :Radius r))
  (Appearance|SetBrush :self Panel :InBrush bI))
