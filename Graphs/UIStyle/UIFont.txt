; UIStyle.UIFont(Role) -> Font. Display roles: F_UI Bold with a dark outline (the doc's fallback until a display face is imported);
; ButtonInk is the same face without the outline, for dark labels on gold and paper.
; Body roles: F_UI (Roboto): names and card titles bold, actions italic, speech and the rest regular. Sizes come from UISize.
(fn UIFont (Role)
  (bind size (CallFunction|UISize :Role Role))
  (bind ow (CallFunction|UISize :Role "TextOutline"))
  (bind ink (CallFunction|UIColor :Role "InkSolid"))
  (bind outline (Utilities|Struct|MakeFontOutlineSettings :OutlineSize (Math|Float|Truncate ow) :OutlineColor ink))
  (bind none (Utilities|Struct|MakeFontOutlineSettings :OutlineSize 0))
  (if (Utilities|String|EqualExactly(String) Role "Header")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings outline)))
  (if (Utilities|String|EqualExactly(String) Role "Banner")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings outline)))
  (if (Utilities|String|EqualExactly(String) Role "Button")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings outline)))
  (if (Utilities|String|EqualExactly(String) Role "Result")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings outline)))
  (if (Utilities|String|EqualExactly(String) Role "Stamp")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings outline)))
  (if (Utilities|String|EqualExactly(String) Role "ButtonInk")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings none)))
  (if (Utilities|String|EqualExactly(String) Role "Name")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings none)))
  (if (Utilities|String|EqualExactly(String) Role "CardTitle")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Bold" :Size size :OutlineSettings none)))
  (if (Utilities|String|EqualExactly(String) Role "Action")
    (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Italic" :Size size :OutlineSettings none)))
  (return (SlateFont|MakeSlateFontInfo :FontObject "/Game/NHOS/UI/Fonts/F_UI.F_UI" :TypefaceFontName "Regular" :Size size :OutlineSettings none)))
