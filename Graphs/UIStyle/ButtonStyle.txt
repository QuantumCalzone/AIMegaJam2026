; UIStyle.ButtonStyle(Kind) -> ButtonStyle. Kinds: Primary, Secondary, Danger, Paper. Unknown kinds get Secondary.
(fn ButtonStyle (Kind)
  ; Primary: act now: the one gold button on a screen
  (if (Utilities|String|EqualExactly(String) Kind "Primary")
    (bind prFill (CallFunction|UIColor :Role "Gold"))
    (bind prHover (CallFunction|UIColor :Role "Gold.Hover"))
    (bind prPress (CallFunction|UIColor :Role "Gold.Pressed"))
    (bind prLine (CallFunction|UIColor :Role "InkSolid"))
    (bind prHoverLine (CallFunction|UIColor :Role "InkSolid"))
    (bind prText (CallFunction|UIColor :Role "InkSolid"))
    (bind prS (CallFunction|MakeButton :Fill prFill :Hover prHover :Press prPress :Line prLine :HoverLine prHoverLine :Text prText))
    (return prS))
  ; Danger: the vote and the cut
  (if (Utilities|String|EqualExactly(String) Kind "Danger")
    (bind daFill (CallFunction|UIColor :Role "Red"))
    (bind daHover (CallFunction|UIColor :Role "Red.Hover"))
    (bind daPress (CallFunction|UIColor :Role "Red.Pressed"))
    (bind daLine (CallFunction|UIColor :Role "InkSolid"))
    (bind daHoverLine (CallFunction|UIColor :Role "Text"))
    (bind daText (CallFunction|UIColor :Role "Text"))
    (bind daS (CallFunction|MakeButton :Fill daFill :Hover daHover :Press daPress :Line daLine :HoverLine daHoverLine :Text daText))
    (return daS))
  ; Paper: paper pieces: ballot tickets, playbill choices
  (if (Utilities|String|EqualExactly(String) Kind "Paper")
    (bind paFill (CallFunction|UIColor :Role "Paper"))
    (bind paHover (CallFunction|UIColor :Role "Paper.Hover"))
    (bind paPress (CallFunction|UIColor :Role "Paper.Pressed"))
    (bind paLine (CallFunction|UIColor :Role "PaperText"))
    (bind paHoverLine (CallFunction|UIColor :Role "PaperText"))
    (bind paText (CallFunction|UIColor :Role "PaperText"))
    (bind paS (CallFunction|MakeButton :Fill paFill :Hover paHover :Press paPress :Line paLine :HoverLine paHoverLine :Text paText))
    (return paS))
  ; Secondary: every other button: Pass, the other ways to play
  (bind seFill (CallFunction|UIColor :Role "Raised"))
  (bind seHover (CallFunction|UIColor :Role "Raised.Hover"))
  (bind sePress (CallFunction|UIColor :Role "Raised.Pressed"))
  (bind seLine (CallFunction|UIColor :Role "Quiet"))
  (bind seHoverLine (CallFunction|UIColor :Role "Text"))
  (bind seText (CallFunction|UIColor :Role "Text"))
  (bind seS (CallFunction|MakeButton :Fill seFill :Hover seHover :Press sePress :Line seLine :HoverLine seHoverLine :Text seText))
  (return seS))
