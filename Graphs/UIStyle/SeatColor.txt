; UIStyle.SeatColor(Name, Fallback) -> Color. The seat colors by provider name (Art Inspiration tab, The bots).
; A bot, its name tag and its chat bar share this color. Unknown names return Fallback.
(fn SeatColor (Name Fallback)
  ; OpenAI: #A8CCFF
  (if (Utilities|String|EqualExactly(String) Name "OpenAI")
    (return (Utilities|Struct|MakeLinearColor 0.3916 0.6038 1.0000 1.0)))
  ; Anthropic: #E8894A
  (if (Utilities|String|EqualExactly(String) Name "Anthropic")
    (return (Utilities|Struct|MakeLinearColor 0.8070 0.2502 0.0685 1.0)))
  ; SpaceXAI: #D8F04A
  (if (Utilities|String|EqualExactly(String) Name "SpaceXAI")
    (return (Utilities|Struct|MakeLinearColor 0.6867 0.8714 0.0685 1.0)))
  ; Meta: #C8AC12 (mustard, T108)
  (if (Utilities|String|EqualExactly(String) Name "Meta")
    (return (Utilities|Struct|MakeLinearColor 0.5776 0.4125 0.0060 1.0)))
  ; You: #3FD0C9
  (if (Utilities|String|EqualExactly(String) Name "You")
    (return (Utilities|Struct|MakeLinearColor 0.0497 0.6308 0.5841 1.0)))
  (return Fallback))
