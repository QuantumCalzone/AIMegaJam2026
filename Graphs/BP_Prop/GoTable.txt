; T110: during a swap the prop is on its way to the table anyway: a grab or throw that waits is dropped.
(fn GoTable (Dur)
  (if (and (> (V Swap) 0) (< (V Swap) 4))
    (set Slot -1)
    (set PendGrab false)
    (set PendThrow false)
    (return))
  (if (== (V Swap) 4)
    (set Swap 0)
    (set PendThrow false))
  (bind _l (Transformation|GetActorLocation))
  (if (and (not (V Flying)) (and (< (Math|Vector|Distance2D(Vector) :V1 _l :V2 (V HomeLoc)) 2.0) (< (Math|Float|Absolute(Float) (- (.z _l) (.z (V HomeLoc)))) 2.0)))
    (set Slot -1)
    (return))
  (Transformation|DetachFromActor :LocationRule "KeepWorld" :RotationRule "KeepWorld" :ScaleRule "KeepWorld")
  (set FlyFrom _l)
  (set FromRot (Transformation|GetActorRotation))
  (set FlyTo (V HomeLoc))
  (set FlyRot (V HomeRot))
  (set Slot -1)
  (set ToComp false)
  (set FlySpin 0.0)
  (set FlyT 0.0)
  (set FlyDur Dur)
  (set FlyH 90.0)
  (set Flying true))
