; T110: the swap's steps, after TickFly each frame. Swap 1: flying out; at the wing the prop takes its new model.
; Swap 2: waits in the wing until SwapT runs out, then flies in (a prop with no name stays in the wing).
; Swap 3: flying in; on landing, a grab or throw that came during the swap plays now (RunPend).
; Swap 4: a deferred grab is on its way to the hand, and a throw waits for it to arrive.
(fn TickSwap (Delta)
  (if (== (V Swap) 0)
    (return))
  (if (< (V Swap) 3)
    (set SwapT (- (V SwapT) Delta)))
  (if (V Flying)
    (return))
  (if (== (V Swap) 1)
    (CallFunction|SetProp :Name (V SwapName))
    (set Swap 2)
    (return))
  (if (== (V Swap) 2)
    (if (> (V SwapT) 0.0)
      (return))
    (if (Utilities|String|IsEmpty (V SwapName))
      (set Swap 0)
      (return))
    (set FlyFrom (V WingLoc))
    (set FromRot (V HomeRot))
    (set FlyTo (V HomeLoc))
    (set FlyRot (V HomeRot))
    (set ToComp false)
    (set FlySpin 0.0)
    (set FlyT 0.0)
    (set FlyDur 0.6)
    (set FlyH 60.0)
    (set Flying true)
    (set Swap 3)
    (return))
  (if (== (V Swap) 3)
    (set Swap 0)
    (print (cat "NHOS: ACT prop in " (V PropName)))
    (CallFunction|RunPend)
    (return))
  (set Swap 0)
  (if (V PendThrow)
    (set PendThrow false)
    (CallFunction|ThrowTo :Loc (V PendLoc) :Dur (V PendTDur) :Height (V PendTH) :Spin (V PendSpin))))
