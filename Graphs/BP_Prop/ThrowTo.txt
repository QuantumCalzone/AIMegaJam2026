(fn ThrowTo (Loc Dur Height Spin)
  (Transformation|DetachFromActor :LocationRule "KeepWorld" :RotationRule "KeepWorld" :ScaleRule "KeepWorld")
  (set FlyFrom (Transformation|GetActorLocation))
  (set FromRot (Transformation|GetActorRotation))
  (set FlyTo (+ Loc (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z (V RestZ))))
  (set FlyRot (V RestRot))
  (set Slot -1)
  (set ToComp false)
  (set FlySpin Spin)
  (set FlyT 0.0)
  (set FlyDur Dur)
  (set FlyH Height)
  (set Flying true))
