; T110: during a swap (Swap 1 to 3) the grab waits: the prop is held at once (Holder, Slot), and flies to the
; hand when it has landed on the table (RunPend). A new flight ends a grab-then-throw that waits (Swap 4).
(fn FlyToComp (Comp Off NewHolder NewSlot Dur Height)
  (if (and (> (V Swap) 0) (< (V Swap) 4))
    (set Holder NewHolder)
    (set Slot NewSlot)
    (set PendComp Comp)
    (set PendOff Off)
    (set PendBot NewHolder)
    (set PendSlot NewSlot)
    (set PendDur Dur)
    (set PendH Height)
    (set PendGrab true)
    (set PendThrow false)
    (print (cat "NHOS: ACT prop waits " (V PropName)))
    (return))
  (if (== (V Swap) 4)
    (set Swap 0)
    (set PendThrow false))
  (Transformation|DetachFromActor :LocationRule "KeepWorld" :RotationRule "KeepWorld" :ScaleRule "KeepWorld")
  (set FlyFrom (Transformation|GetActorLocation))
  (set FromRot (Transformation|GetActorRotation))
  (set FlyComp Comp)
  (set FlyOff (+ Off (V GripLoc)))
  (set Holder NewHolder)
  (set Slot NewSlot)
  (set ToComp true)
  (set FlySpin 0.0)
  (set FlyT 0.0)
  (set FlyDur Dur)
  (set FlyH Height)
  (set Flying true))
