(fn TickFly (Delta)
  (if (not (V Flying))
    (return))
  (set FlyT (+ (V FlyT) (/ Delta (V FlyDur))))
  (bind _a (Math|Float|Clamp(Float) :Value (V FlyT) :Min 0.0 :Max 1.0))
  (bind _e (* _a (* _a (- 3.0 (* 2.0 _a)))))
  (if (V ToComp)
    (set Dest (Math|Transform|TransformLocation :T (Transformation|GetWorldTransform :self (V FlyComp)) :Location (V FlyOff)))
    (else
      (set Dest (V FlyTo))))
  (bind _arc (* (V FlyH) (* 4.0 (* _a (- 1.0 _a)))))
  (Transformation|SetActorLocation :NewLocation (+ (Math|Vector|Lerp(Vector) :A (V FlyFrom) :B (V Dest) :Alpha _e) (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z _arc)))
  (bind _fr (V FromRot))
  (Transformation|SetActorRotation :NewRotation (Math|Rotator|MakeRotator :Roll (.roll _fr) :Pitch (+ (.pitch _fr) (* (V FlySpin) (* _a 360.0))) :Yaw (+ (.yaw _fr) (* (V FlySpin) (* _a 140.0)))))
  (if (>= (V FlyT) 1.0)
    (set Flying false)
    (if (V ToComp)
      (Transformation|AttachActorToComponent :Parent (V FlyComp) :LocationRule "KeepWorld" :RotationRule "KeepWorld" :ScaleRule "KeepWorld" :bWeldSimulatedBodies false)
      (Transformation|SetActorRelativeLocation :NewRelativeLocation (V FlyOff))
      (Transformation|SetActorRelativeRotation :NewRelativeRotation (V GripRot))
      (else
        (Transformation|SetActorLocationAndRotation :NewLocation (V FlyTo) :NewRotation (V FlyRot))))))
