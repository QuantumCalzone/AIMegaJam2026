; BPC_NoteClue.Build(InKind, NoteId) -> String (T88): the clue's text exactly as the card shows it (and the bots read
; it), the kind's prefix included. Each builder call is bound before the return, so its node runs (an impure call
; inside a return's expression is pruned). The built kinds follow T88 step 4; the others read the note's field named in "from".
(fn Build (InKind NoteId)
  (bind _ni (Utilities|Array|FindItem (V NoteIds) NoteId))
  (bind _fi (Utilities|Array|FindItem (V KindIds) InKind))
  (if (or (< _ni 0) (< _fi 0))
    (print (cat "NHOS: CLUE cannot build " InKind " for " NoteId))
    (return ""))
  (set NoteObj (get (V NoteObjs) _ni))
  (bind _n (Json|GetField (V NoteObj) "note"))
  (set Note _n)
  (bind _key (Json|GetField (V NoteObj) "key"))
  (set Key _key)
  (set BPre (get (V KindPrefixes) _fi))
  (if (streq InKind "vowels")
    (bind _r1 (CallFunction|Blank :Src (V Note) :Vowels false))
    (return (cat (V BPre) _r1)))
  (if (streq InKind "consonants")
    (bind _r2 (CallFunction|Blank :Src (V Note) :Vowels true))
    (return (cat (V BPre) _r2)))
  (if (streq InKind "first_letters")
    (bind _r3 (CallFunction|FirstLetters :Src (V Note)))
    (return (cat (V BPre) _r3)))
  (if (streq InKind "scrambled")
    (bind _r4 (CallFunction|Scramble :Src (V Note)))
    (return (cat (V BPre) _r4)))
  (if (streq InKind "pig_latin")
    (bind _r5 (CallFunction|PigLatin :Src (V Note)))
    (return (cat (V BPre) _r5)))
  (if (streq InKind "key_blank")
    (bind _r6 (CallFunction|KeyMask :Src (V Note) :Key (V Key) :Hide true))
    (return (cat (V BPre) _r6)))
  (if (streq InKind "key_only")
    (bind _r7 (CallFunction|KeyMask :Src (V Note) :Key (V Key) :Hide false))
    (return (cat (V BPre) _r7)))
  (bind _from (get (V KindFroms) _fi))
  (bind _v (Json|GetField (V NoteObj) _from))
  (set TmpStr _v)
  (return (cat (V BPre) (V TmpStr))))
